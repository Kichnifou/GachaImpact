/*
CE QUE FAIT LE CODE :

- Gère la commande !coffre
- Affiche les objets collectionnables possédés par le viewer
- Lit la section viewer["coffre"] ajoutée par le système Event
- Ne crée PAS de viewer si le joueur n’existe pas
- Ne donne aucune récompense
- Ne modifie pas les quantités du coffre
- Ajoute seulement les sections manquantes via EnsureViewerDefaults si le viewer existe
- Affiche un message Twitch compact sur une ligne
- Split proprement les messages si le contenu dépasse 450 caractères
- Format :
  🏆 Coffre | 🏝️ Coquillage Doré (x2) | ⭐ Étoile Filante (x1)
- Si le coffre est vide, affiche une aide courte
- Lit et sauvegarde viewers_data.json en JObject
- Ne reconstruit jamais entièrement un viewer
- Compatible Streamer.bot
- Sans LINQ
*/

using System;
using System.IO;
using System.Collections.Generic;
using Newtonsoft.Json;
using Newtonsoft.Json.Linq;

public class CPHInline
{
    private const int MaxMessageLength = 450;

    public bool Execute()
    {
        string viewersPath = @"C:\Users\axeld\Documents\Kichnifou\Twitch\Streambot\Data\viewers_data.json";

        if (!args.ContainsKey("userName") || args["userName"] == null)
            return false;

        string username = NormalizeUsername(args["userName"].ToString());

        if (string.IsNullOrEmpty(username))
            return false;

        if (!File.Exists(viewersPath))
        {
            SendWarning("Aucune donnée viewer trouvée.");
            return true;
        }

        JObject viewers = LoadJObject(viewersPath);

        if (viewers == null)
        {
            SendError("Impossible de lire viewers_data.json.");
            return true;
        }

        JObject viewer = viewers[username] as JObject;

        if (viewer == null)
        {
            SendWarning(username + ", tu n'existes pas encore dans le jeu.");
            return true;
        }

        EnsureViewerDefaults(viewer, username);

        JObject coffre = GetObject(viewer, "coffre");

        if (coffre.Count == 0)
        {
            SaveJObject(viewersPath, viewers);
            CPH.SendMessage("ℹ️ " + username + ", ton coffre est vide. Les objets d'event s'obtiennent avec !event collection.");
            return true;
        }

        List<string> items = BuildCoffreItems(coffre);

        if (items.Count == 0)
        {
            SaveJObject(viewersPath, viewers);
            CPH.SendMessage("ℹ️ " + username + ", ton coffre est vide. Les objets d'event s'obtiennent avec !event collection.");
            return true;
        }

        SaveJObject(viewersPath, viewers);

        SendSplitCoffre(username, items);

        return true;
    }

    private List<string> BuildCoffreItems(JObject coffre)
    {
        List<string> items = new List<string>();

        foreach (KeyValuePair<string, JToken> entry in coffre)
        {
            string itemId = NormalizeText(entry.Key);
            int quantity = ParseInt(entry.Value);

            if (quantity <= 0)
                continue;

            string display = GetCollectionDisplay(itemId);

            if (string.IsNullOrEmpty(display))
                display = "❔ " + entry.Key;

            items.Add(display + " (x" + quantity + ")");
        }

        SortItemsByKnownOrder(items);

        return items;
    }

    private void SortItemsByKnownOrder(List<string> items)
    {
        // Tri très simple pour garder un ordre logique par mois sans LINQ.
        string[] order = new string[]
        {
            "🎆 Lanterne du Nouvel An",
            "💖 Cœur Cristallin",
            "🌱 Bourgeon Éternel",
            "🥚 Œuf Enchanté",
            "🌸 Fleur de Printemps",
            "🏝️ Coquillage Doré",
            "⭐ Étoile Filante",
            "🧭 Boussole Antique",
            "🌾 Gerbe de Récolte",
            "🎃 Citrouille Hantée",
            "🍁 Feuille Ancienne",
            "❄️ Flocon Enchanté"
        };

        List<string> sorted = new List<string>();

        for (int i = 0; i < order.Length; i++)
        {
            for (int j = 0; j < items.Count; j++)
            {
                if (items[j].StartsWith(order[i]))
                    sorted.Add(items[j]);
            }
        }

        for (int j = 0; j < items.Count; j++)
        {
            bool alreadyAdded = false;

            for (int k = 0; k < sorted.Count; k++)
            {
                if (sorted[k] == items[j])
                {
                    alreadyAdded = true;
                    break;
                }
            }

            if (!alreadyAdded)
                sorted.Add(items[j]);
        }

        items.Clear();

        for (int i = 0; i < sorted.Count; i++)
            items.Add(sorted[i]);
    }

    private string GetCollectionDisplay(string itemId)
    {
        if (itemId == "lanterne_nouvel_an")
            return "🎆 Lanterne du Nouvel An";

        if (itemId == "coeur_cristallin")
            return "💖 Cœur Cristallin";

        if (itemId == "bourgeon_eternel")
            return "🌱 Bourgeon Éternel";

        if (itemId == "oeuf_enchante")
            return "🥚 Œuf Enchanté";

        if (itemId == "fleur_de_printemps")
            return "🌸 Fleur de Printemps";

        if (itemId == "coquillage_dore")
            return "🏝️ Coquillage Doré";

        if (itemId == "etoile_filante")
            return "⭐ Étoile Filante";

        if (itemId == "boussole_antique")
            return "🧭 Boussole Antique";

        if (itemId == "gerbe_de_recolte")
            return "🌾 Gerbe de Récolte";

        if (itemId == "citrouille_hantee")
            return "🎃 Citrouille Hantée";

        if (itemId == "feuille_ancienne")
            return "🍁 Feuille Ancienne";

        if (itemId == "flocon_enchante")
            return "❄️ Flocon Enchanté";

        return "";
    }

    private void SendSplitCoffre(string username, List<string> items)
    {
        string prefix = "🏆 Coffre de " + username + " | ";
        string current = prefix;

        for (int i = 0; i < items.Count; i++)
        {
            string part = items[i];

            if (current.Length + part.Length + 3 > MaxMessageLength)
            {
                CPH.SendMessage(current.TrimEnd());

                current = "🏆 Coffre suite | " + part;
            }
            else
            {
                if (current != prefix && !current.EndsWith("| "))
                    current += " | ";

                current += part;
            }
        }

        if (!string.IsNullOrWhiteSpace(current))
            CPH.SendMessage(current.TrimEnd());
    }

    private JObject LoadJObject(string path)
    {
        try
        {
            string json = File.ReadAllText(path);

            if (string.IsNullOrWhiteSpace(json))
                return new JObject();

            JObject data = JObject.Parse(json);

            if (data == null)
                return new JObject();

            return data;
        }
        catch
        {
            return null;
        }
    }

    private void SaveJObject(string path, JObject data)
    {
        string folder = Path.GetDirectoryName(path);

        if (!Directory.Exists(folder))
            Directory.CreateDirectory(folder);

        File.WriteAllText(path, data.ToString(Formatting.Indented));
    }

    private void EnsureViewerDefaults(JObject viewer, string username)
    {
        if (viewer == null)
            return;

        if (viewer["username"] == null || viewer["username"].Type == JTokenType.Null)
            viewer["username"] = username;

        EnsureInt(viewer, "xp", 0);
        EnsureInt(viewer, "level", 0);
        EnsureString(viewer, "element", "");
        EnsureInt(viewer, "primogems", 0);
        EnsureInt(viewer, "moras", 0);
        EnsureInt(viewer, "selectedBannerCharacterId", 0);

        JObject particles = GetObject(viewer, "particles");
        EnsureInt(particles, "pyro", 0);
        EnsureInt(particles, "hydro", 0);
        EnsureInt(particles, "cryo", 0);
        EnsureInt(particles, "electro", 0);
        EnsureInt(particles, "anemo", 0);
        EnsureInt(particles, "geo", 0);
        EnsureInt(particles, "dendro", 0);

        GetObject(viewer, "box");
        GetArray(viewer, "team");

        JObject pity = GetObject(viewer, "pity");
        EnsureInt(pity, "pity5", 0);
        EnsureInt(pity, "pity4", 0);

        JObject guarantee = GetObject(viewer, "guarantee");
        EnsureBool(guarantee, "guaranteedFeatured5", false);

        JObject options = GetObject(viewer, "options");
        EnsureString(options, "boxSort", "a");
        EnsureBool(options, "boxSortDescending", false);

        JObject missions = GetObject(viewer, "missions");
        JObject daily = GetObject(missions, "daily");
        EnsureNullOrString(daily, "missionId");
        EnsureNullOrString(daily, "type");
        EnsureInt(daily, "progress", 0);
        EnsureInt(daily, "target", 0);
        EnsureBool(daily, "completed", false);
        EnsureBool(daily, "rewardClaimed", false);
        EnsureNullOrString(daily, "startedAt");
        EnsureNullOrString(daily, "description");

        JObject stats = GetObject(viewer, "stats");
        EnsureInt(stats, "totalMessages", 0);
        EnsureInt(stats, "countedMessages", 0);
        EnsureInt(stats, "totalPulls", 0);
        EnsureInt(stats, "totalFiveStars", 0);
        EnsureInt(stats, "totalFourStars", 0);
        EnsureBool(stats, "lastPullWasFiveStar", false);
        EnsureInt(stats, "fiftyFiftyLostStreak", 0);
        EnsureInt(stats, "fiftyFiftyWon", 0);
        EnsureInt(stats, "fiftyFiftyLost", 0);
        EnsureInt(stats, "totalPrimosEarned", 0);
        EnsureInt(stats, "totalPrimosSpent", 0);
        EnsureInt(stats, "totalMorasEarned", 0);
        EnsureInt(stats, "totalMorasSpent", 0);
        EnsureInt(stats, "level100OverflowRewardsClaimed", 0);

        JObject dates = GetObject(viewer, "dates");
        EnsureNullOrString(dates, "firstSeen");
        EnsureNullOrString(dates, "lastSeen");
        EnsureNullOrString(dates, "lastXpDate");
        EnsureNullOrString(dates, "lastMessageTime");
        EnsureNullOrString(dates, "lastDailyFirstMessageReward");

        GetArray(viewer, "tradeRequests");
        GetArray(viewer, "usedCodes");

        JObject favor = GetObject(viewer, "favor");
        EnsureInt(favor, "daysRemaining", 0);
        EnsureString(favor, "obtainedDate", "");
        EnsureString(favor, "lastClaimDate", "");

        GetObject(viewer, "coffre");

        JObject specialItems = GetObject(viewer, "specialItems");
        EnsureInt(specialItems, "masterlessStellaFortuna", 0);
    }

    private JObject GetObject(JObject parent, string key)
    {
        if (parent[key] == null || parent[key].Type == JTokenType.Null || parent[key].Type != JTokenType.Object)
            parent[key] = new JObject();

        return parent[key] as JObject;
    }

    private JArray GetArray(JObject parent, string key)
    {
        if (parent[key] == null || parent[key].Type == JTokenType.Null || parent[key].Type != JTokenType.Array)
            parent[key] = new JArray();

        return parent[key] as JArray;
    }

    private int ParseInt(JToken token)
    {
        if (token == null || token.Type == JTokenType.Null)
            return 0;

        int value;

        if (int.TryParse(token.ToString(), out value))
            return value;

        return 0;
    }

    private void EnsureInt(JObject obj, string key, int defaultValue)
    {
        if (obj[key] == null || obj[key].Type == JTokenType.Null)
            obj[key] = defaultValue;
    }

    private void EnsureBool(JObject obj, string key, bool defaultValue)
    {
        if (obj[key] == null || obj[key].Type == JTokenType.Null)
            obj[key] = defaultValue;
    }

    private void EnsureString(JObject obj, string key, string defaultValue)
    {
        if (obj[key] == null || obj[key].Type == JTokenType.Null)
            obj[key] = defaultValue;
    }

    private void EnsureNullOrString(JObject obj, string key)
    {
        if (obj[key] == null)
            obj[key] = null;
    }

    private string NormalizeUsername(string username)
    {
        if (string.IsNullOrEmpty(username))
            return "";

        username = username.Trim();

        if (username.StartsWith("@"))
            username = username.Substring(1);

        return username.ToLower();
    }

    private string NormalizeText(string text)
    {
        if (string.IsNullOrEmpty(text))
            return "";

        string value = text.ToLower().Trim();

        value = value.Replace("à", "a");
        value = value.Replace("á", "a");
        value = value.Replace("â", "a");
        value = value.Replace("ä", "a");
        value = value.Replace("ã", "a");
        value = value.Replace("ç", "c");
        value = value.Replace("è", "e");
        value = value.Replace("é", "e");
        value = value.Replace("ê", "e");
        value = value.Replace("ë", "e");
        value = value.Replace("ì", "i");
        value = value.Replace("í", "i");
        value = value.Replace("î", "i");
        value = value.Replace("ï", "i");
        value = value.Replace("ò", "o");
        value = value.Replace("ó", "o");
        value = value.Replace("ô", "o");
        value = value.Replace("ö", "o");
        value = value.Replace("õ", "o");
        value = value.Replace("ù", "u");
        value = value.Replace("ú", "u");
        value = value.Replace("û", "u");
        value = value.Replace("ü", "u");
        value = value.Replace("ÿ", "y");
        value = value.Replace("œ", "oe");
        value = value.Replace("æ", "ae");

        return value;
    }

    private void SendWarning(string message)
    {
        CPH.SendMessage("⚠️ " + message);
    }

    private void SendError(string message)
    {
        CPH.SendMessage("❌ " + message);
    }
}
