/*
CE QUE FAIT LE CODE :

- Gère la commande !banniere
- Lit le fichier genshin_characters.json
- Récupère les personnages actuellement en bannière
- Affiche les 3 personnages 5★ en vedette
- Affiche les 5 personnages 4★ en vedette
- Retourne une erreur si le fichier personnage est introuvable
- Retourne une erreur si aucune bannière n’est encore initialisée
- Lit viewers_data.json en JObject uniquement pour connaître selectedBannerCharacterId
- Ne crée PAS de viewer
- Ne sauvegarde PAS viewers_data.json
- Ne modifie AUCUNE donnée viewer
- Préserve totalement la structure JSON existante
*/

using System;
using System.IO;
using System.Collections.Generic;
using Newtonsoft.Json;
using Newtonsoft.Json.Linq;

public class CPHInline
{
    public bool Execute()
    {
        string folderPath = @"C:\Users\axeld\Documents\Kichnifou\Twitch\Streambot\Data";
        string charactersPath = Path.Combine(folderPath, "genshin_characters.json");
        string viewersPath = Path.Combine(folderPath, "viewers_data.json");

        if (!args.ContainsKey("userName") || args["userName"] == null)
            return false;

        string username = NormalizeUsername(args["userName"].ToString());

        if (!File.Exists(charactersPath))
        {
            CPH.SendMessage("❌ Aucun fichier personnage trouvé.");
            return true;
        }

        CharacterFile characterFile = LoadCharacters(charactersPath);

        if (characterFile == null || characterFile.characters == null)
        {
            CPH.SendMessage("❌ Impossible de lire la liste des personnages.");
            return true;
        }

        JObject viewers = LoadViewersJObject(viewersPath);
        JObject viewer = null;

        if (viewers != null && viewers[username] != null && viewers[username].Type == JTokenType.Object)
            viewer = viewers[username] as JObject;

        List<CharacterData> featured5 = new List<CharacterData>();
        List<CharacterData> featured4 = new List<CharacterData>();

        for (int i = 0; i < characterFile.characters.Count; i++)
        {
            CharacterData character = characterFile.characters[i];

            if (character == null)
                continue;

            if (!character.bannerFeatured)
                continue;

            if (character.rarete == 5)
                featured5.Add(character);
            else if (character.rarete == 4)
                featured4.Add(character);
        }

        if (featured5.Count == 0 && featured4.Count == 0)
        {
            CPH.SendMessage("⚠️ Aucune bannière n’est encore active. Elle sera générée automatiquement quand quelqu’un parlera dans le chat.");
            return true;
        }

        string message = "🎯 Bannières";

        if (!string.IsNullOrEmpty(characterFile.lastBannerUpdate))
        {
            DateTime start;

            if (DateTime.TryParse(characterFile.lastBannerUpdate, out start))
            {
                DateTime end = start.AddDays(6);
                message += $" ({start:dd/MM} -> {end:dd/MM})";
            }
        }

        message += " | ⭐⭐⭐⭐⭐ ";
        message += featured5.Count > 0 ? FormatCharacters(featured5) : "aucun";

        message += " | ⭐⭐⭐⭐ ";
        message += featured4.Count > 0 ? FormatCharacters(featured4) : "aucun";

        if (viewer != null)
        {
            int selectedBannerCharacterId = GetInt(viewer, "selectedBannerCharacterId");

            if (selectedBannerCharacterId == 0 && featured5.Count > 0)
            {
                message += " | Utilise !select nom_du_perso pour choisir ton 5★ ciblé.";
            }
            else
            {
                CharacterData selectedCharacter = FindCharacterById(featured5, selectedBannerCharacterId);

                if (selectedCharacter != null)
                    message += $" | 5★ ciblé : {GetElementEmoji(selectedCharacter.element)} {selectedCharacter.nom}";
            }
        }

        CPH.SendMessage(message);

        return true;
    }

    private CharacterFile LoadCharacters(string path)
    {
        try
        {
            string json = File.ReadAllText(path);
            return JsonConvert.DeserializeObject<CharacterFile>(json);
        }
        catch
        {
            return null;
        }
    }

    private JObject LoadViewersJObject(string path)
    {
        try
        {
            if (!File.Exists(path))
                return null;

            string json = File.ReadAllText(path);

            if (string.IsNullOrWhiteSpace(json))
                return null;

            return JObject.Parse(json);
        }
        catch
        {
            return null;
        }
    }

    private CharacterData FindCharacterById(List<CharacterData> characters, int id)
    {
        if (characters == null)
            return null;

        for (int i = 0; i < characters.Count; i++)
        {
            CharacterData character = characters[i];

            if (character != null && character.id == id)
                return character;
        }

        return null;
    }

    private string FormatCharacters(List<CharacterData> characters)
    {
        List<string> names = new List<string>();

        for (int i = 0; i < characters.Count; i++)
        {
            CharacterData character = characters[i];

            if (character == null)
                continue;

            names.Add($"{GetElementEmoji(character.element)} {character.nom}");
        }

        return string.Join(", ", names.ToArray());
    }

    private int GetInt(JObject obj, string key)
    {
        if (obj == null || obj[key] == null || obj[key].Type == JTokenType.Null)
            return 0;

        int value;

        if (int.TryParse(obj[key].ToString(), out value))
            return value;

        return 0;
    }

    private string GetElementEmoji(string element)
    {
        switch (NormalizeElement(element))
        {
            case "pyro": return "🔥";
            case "hydro": return "💧";
            case "cryo": return "❄️";
            case "electro": return "⚡";
            case "anemo": return "🌪️";
            case "geo": return "☄️";
            case "dendro": return "🌿";
            default: return "";
        }
    }

    private string NormalizeElement(string element)
    {
        if (string.IsNullOrEmpty(element))
            return "";

        string value = element.ToLower().Trim();

        value = value.Replace("é", "e");
        value = value.Replace("è", "e");
        value = value.Replace("ê", "e");
        value = value.Replace("ë", "e");
        value = value.Replace("á", "a");
        value = value.Replace("à", "a");
        value = value.Replace("â", "a");
        value = value.Replace("ä", "a");

        switch (value)
        {
            case "pyro": return "pyro";
            case "hydro": return "hydro";
            case "cryo": return "cryo";
            case "electro": return "electro";
            case "anemo": return "anemo";
            case "geo": return "geo";
            case "dendro": return "dendro";
            default: return value;
        }
    }

    private string NormalizeUsername(string username)
    {
        if (string.IsNullOrEmpty(username))
            return "";

        username = username.Trim();

        if (username.StartsWith("@"))
            username = username.Substring(1);

        return username.ToLower();
    }
}

public class CharacterFile
{
    public string lastBannerUpdate { get; set; }
    public List<CharacterData> characters { get; set; }
}

public class CharacterData
{
    public int id { get; set; }
    public string nom { get; set; }
    public int rarete { get; set; }
    public string element { get; set; }
    public string arme { get; set; }
    public string region { get; set; }
    public string classe { get; set; }
    public bool bannerFeatured { get; set; }
    public string passif { get; set; }
}
