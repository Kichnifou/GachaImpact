/*
CE QUE FAIT LE CODE :

- Gère la commande !banque
- Affiche le solde bancaire du joueur avec !banque
- Permet de déposer des moras avec !banque deposer 10000
- Permet de retirer des moras avec !banque retirer 10000
- Accepte deposer / depose / déposer / dépose
- Accepte retirer / retire / retiré
- Stocke les moras dans viewer.bank.moras
- Ne crée PAS de viewer si le joueur n’existe pas
- Ne reconstruit jamais le viewer complet
- Initialise seulement la section bank si elle manque
*/

using System;
using System.IO;
using Newtonsoft.Json;
using Newtonsoft.Json.Linq;

public class CPHInline
{
    public bool Execute()
    {
        string folderPath = @"C:\Users\axeld\Documents\Kichnifou\Twitch\Streambot\Data";
        string viewersPath = Path.Combine(folderPath, "viewers_data.json");

        if (!args.ContainsKey("userName") || args["userName"] == null)
            return false;

        string username = NormalizeUsername(args["userName"].ToString());
        string input = GetCommandInput();

        if (!File.Exists(viewersPath))
        {
            SendWarning(username + ", ton profil n’existe pas encore.");
            return true;
        }

        JObject viewers = LoadJObject(viewersPath);

        if (viewers == null)
        {
            SendError("Impossible de lire viewers_data.json.");
            return true;
        }

        JObject viewer = viewers[username] as JObject;

        if (viewer == null)
        {
            SendWarning(username + ", ton profil n’existe pas encore.");
            return true;
        }

        EnsureViewerDefaults(viewer);

        if (string.IsNullOrEmpty(input))
        {
            ShowBank(username, viewer);
            SaveJObject(viewersPath, viewers);
            return true;
        }

        string[] parts = input.Split(new char[] { ' ' }, StringSplitOptions.RemoveEmptyEntries);

        if (parts.Length < 2)
        {
            SendWarning("Utilise : !banque deposer 10000 | !banque retirer 10000");
            return true;
        }

        string action = NormalizeText(parts[0]);
        int amount = ParsePositiveInt(parts[1]);

        if (amount <= 0)
        {
            SendWarning("Montant invalide. Exemple : !banque deposer 10000");
            return true;
        }

        if (IsDeposit(action))
        {
            Deposit(username, viewer, amount);
            SaveJObject(viewersPath, viewers);
            return true;
        }

        if (IsWithdraw(action))
        {
            Withdraw(username, viewer, amount);
            SaveJObject(viewersPath, viewers);
            return true;
        }

        SendWarning("Action invalide. Utilise : !banque deposer 10000 | !banque retirer 10000");
        return true;
    }

    private void ShowBank(string username, JObject viewer)
    {
        JObject bank = viewer["bank"] as JObject;
        int bankMoras = GetInt(bank, "moras");
        int moras = GetInt(viewer, "moras");

        CPH.SendMessage(
            "🏦 Banque " + username +
            " : " + bankMoras + " moras | 💰 Portefeuille : " + moras +
            " | 📥 !banque deposer X | 📤 !banque retirer X"
        );
    }

    private void Deposit(string username, JObject viewer, int amount)
    {
        int moras = GetInt(viewer, "moras");

        if (moras < amount)
        {
            SendWarning(username + ", tu n’as pas assez de moras. Solde : " + moras + ".");
            return;
        }

        JObject bank = viewer["bank"] as JObject;

        AddInt(viewer, "moras", -amount);
        AddInt(bank, "moras", amount);

        SendSuccess(username + " dépose " + amount + " moras à la banque. Banque : " + GetInt(bank, "moras") + " | Sur toi : " + GetInt(viewer, "moras"));
    }

    private void Withdraw(string username, JObject viewer, int amount)
    {
        JObject bank = viewer["bank"] as JObject;
        int bankMoras = GetInt(bank, "moras");

        if (bankMoras < amount)
        {
            SendWarning(username + ", tu n’as pas assez de moras en banque. Banque : " + bankMoras + ".");
            return;
        }

        AddInt(bank, "moras", -amount);
        AddInt(viewer, "moras", amount);

        SendSuccess(username + " retire " + amount + " moras de la banque. Banque : " + GetInt(bank, "moras") + " | Sur toi : " + GetInt(viewer, "moras"));
    }

    private void EnsureViewerDefaults(JObject viewer)
    {
        EnsureInt(viewer, "moras", 0);
        EnsureObject(viewer, "bank");

        JObject bank = viewer["bank"] as JObject;
        EnsureInt(bank, "moras", 0);
        EnsureString(bank, "lastInterestDate", "");
    }

    private bool IsDeposit(string action)
    {
        return action == "deposer" || action == "depose";
    }

    private bool IsWithdraw(string action)
    {
        return action == "retire" || action == "retirer" || action == "retiree";
    }

    private string GetCommandInput()
    {
        if (args.ContainsKey("rawInput") && args["rawInput"] != null)
            return args["rawInput"].ToString().Trim();

        if (args.ContainsKey("input0") && args["input0"] != null)
            return args["input0"].ToString().Trim();

        return "";
    }

    private JObject LoadJObject(string path)
    {
        try
        {
            return JObject.Parse(File.ReadAllText(path));
        }
        catch
        {
            return null;
        }
    }

    private void SaveJObject(string path, JObject data)
    {
        File.WriteAllText(path, data.ToString(Formatting.Indented));
    }

    private void EnsureObject(JObject parent, string key)
    {
        if (parent[key] == null || parent[key].Type == JTokenType.Null || parent[key].Type != JTokenType.Object)
            parent[key] = new JObject();
    }

    private void EnsureInt(JObject obj, string key, int defaultValue)
    {
        if (obj[key] == null || obj[key].Type == JTokenType.Null)
            obj[key] = defaultValue;
    }

    private void EnsureString(JObject obj, string key, string defaultValue)
    {
        if (obj[key] == null || obj[key].Type == JTokenType.Null)
            obj[key] = defaultValue;
    }

    private int GetInt(JObject obj, string key)
    {
        if (obj == null || obj[key] == null || obj[key].Type == JTokenType.Null)
            return 0;

        int value;
        if (int.TryParse(obj[key].ToString(), out value))
            return value;

        return 0;
    }

    private void AddInt(JObject obj, string key, int amount)
    {
        obj[key] = GetInt(obj, key) + amount;
    }

    private int ParsePositiveInt(string value)
    {
        int result;
        if (!int.TryParse(value, out result))
            return 0;

        if (result <= 0)
            return 0;

        return result;
    }

    private string NormalizeUsername(string username)
    {
        if (string.IsNullOrEmpty(username))
            return "";

        username = username.Trim();

        if (username.StartsWith("@"))
            username = username.Substring(1);

        return username.ToLower();
    }

    private string NormalizeText(string text)
    {
        if (string.IsNullOrEmpty(text))
            return "";

        string value = text.ToLower().Trim();
        value = value.Replace("é", "e").Replace("è", "e").Replace("ê", "e").Replace("ë", "e");
        return value;
    }

    private void SendSuccess(string message) { CPH.SendMessage("✅ " + message); }
    private void SendWarning(string message) { CPH.SendMessage("⚠️ " + message); }
    private void SendError(string message) { CPH.SendMessage("❌ " + message); }
}
