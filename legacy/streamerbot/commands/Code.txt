/*
CE QUE FAIT LE CODE :

- Gère la commande !code
- Lit viewers_data.json avec JObject
- Lit gift_codes.json avec classes simples
- Ne crée PAS le viewer s’il n’existe pas
- Permet à un viewer d’utiliser un code cadeau une seule fois
- Gère aussi les codes mensuels renouvelables chaque année
- Pour les codes annuels, stocke l’utilisation avec le format CODE-ANNEE
- Vérifie si le code existe
- Vérifie si le code est expiré
- Donne les récompenses configurées :
  - primos
  - moras
  - particules élémentaires
- Met à jour les stats totalPrimosEarned et totalMorasEarned
- Sauvegarde le code utilisé dans usedCodes
- Sauvegarde viewers_data.json sans reconstruire les viewers
- Préserve les données existantes du viewer
*/

using System;
using System.IO;
using System.Collections.Generic;
using Newtonsoft.Json;
using Newtonsoft.Json.Linq;

public class CPHInline
{
    public bool Execute()
    {
        string folderPath = @"C:\Users\axeld\Documents\Kichnifou\Twitch\Streambot\Data";
        string viewersPath = Path.Combine(folderPath, "viewers_data.json");
        string codesPath = Path.Combine(folderPath, "gift_codes.json");

        if (!args.ContainsKey("userName") || args["userName"] == null)
            return false;

        string username = NormalizeUsername(args["userName"].ToString());
        string inputCode = GetCommandInput().Trim().ToUpper();

        if (string.IsNullOrEmpty(inputCode))
        {
            SendWarning($"{username}, utilise : !code NOMDUCODE");
            return true;
        }

        if (!File.Exists(viewersPath))
        {
            SendWarning($"{username}, ton profil n’existe pas encore.");
            return true;
        }

        if (!File.Exists(codesPath))
        {
            SendError("Fichier gift_codes.json introuvable.");
            return true;
        }

        JObject viewers = LoadJObject(viewersPath);

        if (viewers == null)
        {
            SendError("Impossible de lire viewers_data.json.");
            return true;
        }

        JObject viewer = viewers[username] as JObject;

        if (viewer == null)
        {
            SendWarning($"{username}, ton profil n’existe pas encore.");
            return true;
        }

        GiftCodeFile giftCodeFile = LoadGiftCodes(codesPath);

        if (giftCodeFile == null)
        {
            SendError("Impossible de lire gift_codes.json.");
            return true;
        }

        if (giftCodeFile.codes == null)
            giftCodeFile.codes = new List<GiftCodeData>();

        GiftCodeData code = FindGiftCode(giftCodeFile.codes, inputCode);

        if (code == null)
        {
            SendWarning($"{username}, code invalide.");
            return true;
        }

        DateTime now = DateTime.Now;

        if (!IsCodeActive(code, inputCode, now))
            return true;

        EnsureViewerDefaults(viewer, username);

        string usedCodeKey = GetUsedCodeKey(code, inputCode, now);

        if (HasUsedCode(viewer, usedCodeKey))
        {
            SendWarning($"{username}, tu as déjà utilisé le code {inputCode}.");
            return true;
        }

        if (code.rewards == null)
            code.rewards = new GiftCodeRewards();

        if (code.rewards.particles == null)
            code.rewards.particles = new ElementParticles();

        ApplyRewards(viewer, code.rewards);

        JArray usedCodes = viewer["usedCodes"] as JArray;

        if (usedCodes == null)
        {
            usedCodes = new JArray();
            viewer["usedCodes"] = usedCodes;
        }

        usedCodes.Add(usedCodeKey);

        SaveJObject(viewersPath, viewers);

        string rewardsText = BuildRewardsText(viewer, code.rewards);
        string extraMessage = string.IsNullOrEmpty(code.message) ? "" : $" | {code.message}";

        SendSuccess($"{username} a utilisé {inputCode} ! {rewardsText}{extraMessage}");

        return true;
    }

    private bool IsCodeActive(GiftCodeData code, string inputCode, DateTime now)
    {
        if (code == null)
            return false;

        if (code.annuallyRenewable)
        {
            if (code.month < 1 || code.month > 12)
            {
                SendError($"Le code {inputCode} a un mois invalide.");
                return false;
            }

            if (now.Month != code.month)
            {
                SendWarning($"{inputCode} n’est pas disponible ce mois-ci.");
                return false;
            }

            return true;
        }

        DateTime expiresAt;

        if (!DateTime.TryParse(code.expiresAt, out expiresAt))
        {
            SendError($"Le code {inputCode} a une date d’expiration invalide.");
            return false;
        }

        if (now > expiresAt)
        {
            SendWarning($"{inputCode} est expiré.");
            return false;
        }

        return true;
    }

    private string GetUsedCodeKey(GiftCodeData code, string inputCode, DateTime now)
    {
        if (code != null && code.annuallyRenewable)
            return inputCode + "-" + now.Year.ToString();

        return inputCode;
    }

    private string GetCommandInput()
    {
        if (args.ContainsKey("rawInput") && args["rawInput"] != null)
            return args["rawInput"].ToString();

        if (args.ContainsKey("input0") && args["input0"] != null)
            return args["input0"].ToString();

        return "";
    }

    private JObject LoadJObject(string path)
    {
        try
        {
            string json = File.ReadAllText(path);
            return JObject.Parse(json);
        }
        catch
        {
            return null;
        }
    }

    private GiftCodeFile LoadGiftCodes(string path)
    {
        try
        {
            string json = File.ReadAllText(path);
            GiftCodeFile giftCodeFile = JsonConvert.DeserializeObject<GiftCodeFile>(json);

            if (giftCodeFile == null)
                giftCodeFile = new GiftCodeFile();

            if (giftCodeFile.codes == null)
                giftCodeFile.codes = new List<GiftCodeData>();

            return giftCodeFile;
        }
        catch
        {
            return null;
        }
    }

    private void SaveJObject(string path, JObject data)
    {
        File.WriteAllText(path, data.ToString(Formatting.Indented));
    }

    private GiftCodeData FindGiftCode(List<GiftCodeData> codes, string inputCode)
    {
        for (int i = 0; i < codes.Count; i++)
        {
            GiftCodeData code = codes[i];

            if (code == null || string.IsNullOrEmpty(code.code))
                continue;

            if (code.code.Trim().ToUpper() == inputCode)
                return code;
        }

        return null;
    }

    private bool HasUsedCode(JObject viewer, string inputCode)
    {
        JArray usedCodes = viewer["usedCodes"] as JArray;

        if (usedCodes == null)
            return false;

        for (int i = 0; i < usedCodes.Count; i++)
        {
            string usedCode = usedCodes[i] != null ? usedCodes[i].ToString() : "";

            if (string.IsNullOrEmpty(usedCode))
                continue;

            if (usedCode.Trim().ToUpper() == inputCode)
                return true;
        }

        return false;
    }

    private void ApplyRewards(JObject viewer, GiftCodeRewards rewards)
    {
        if (rewards.primogems > 0)
        {
            AddInt(viewer, "primogems", rewards.primogems);
            AddStat(viewer, "totalPrimosEarned", rewards.primogems);
        }

        if (rewards.moras > 0)
        {
            AddInt(viewer, "moras", rewards.moras);
            AddStat(viewer, "totalMorasEarned", rewards.moras);
        }

        if (rewards.particles == null)
            rewards.particles = new ElementParticles();

        AddParticles(viewer, "pyro", rewards.particles.pyro);
        AddParticles(viewer, "hydro", rewards.particles.hydro);
        AddParticles(viewer, "cryo", rewards.particles.cryo);
        AddParticles(viewer, "electro", rewards.particles.electro);
        AddParticles(viewer, "anemo", rewards.particles.anemo);
        AddParticles(viewer, "geo", rewards.particles.geo);
        AddParticles(viewer, "dendro", rewards.particles.dendro);
    }

    private string BuildRewardsText(JObject viewer, GiftCodeRewards rewards)
    {
        List<string> parts = new List<string>();

        if (rewards.primogems > 0)
            parts.Add($"+💠{rewards.primogems} primos ({GetInt(viewer, "primogems")})");

        if (rewards.moras > 0)
            parts.Add($"+🪙{rewards.moras} moras ({GetInt(viewer, "moras")})");

        if (rewards.particles == null)
            rewards.particles = new ElementParticles();

        AddParticleRewardText(parts, viewer, "pyro", rewards.particles.pyro);
        AddParticleRewardText(parts, viewer, "hydro", rewards.particles.hydro);
        AddParticleRewardText(parts, viewer, "cryo", rewards.particles.cryo);
        AddParticleRewardText(parts, viewer, "electro", rewards.particles.electro);
        AddParticleRewardText(parts, viewer, "anemo", rewards.particles.anemo);
        AddParticleRewardText(parts, viewer, "geo", rewards.particles.geo);
        AddParticleRewardText(parts, viewer, "dendro", rewards.particles.dendro);

        if (parts.Count == 0)
            return "aucune récompense";

        return string.Join(" | ", parts.ToArray());
    }

    private void AddParticleRewardText(List<string> parts, JObject viewer, string element, int amount)
    {
        if (amount <= 0)
            return;

        parts.Add($"+{amount} particules {GetElementDisplayName(element)} ({GetParticleTotal(viewer, element)})");
    }

    private void AddParticles(JObject viewer, string element, int amount)
    {
        if (amount <= 0)
            return;

        JObject particles = viewer["particles"] as JObject;

        if (particles == null)
        {
            particles = new JObject();
            viewer["particles"] = particles;
        }

        AddInt(particles, NormalizeElement(element), amount);
    }

    private int GetParticleTotal(JObject viewer, string element)
    {
        JObject particles = viewer["particles"] as JObject;

        if (particles == null)
            return 0;

        return GetInt(particles, NormalizeElement(element));
    }

    private string GetElementDisplayName(string element)
    {
        switch (NormalizeElement(element))
        {
            case "pyro": return "🔥 Pyro";
            case "hydro": return "💧 Hydro";
            case "cryo": return "❄️ Cryo";
            case "electro": return "⚡ Electro";
            case "anemo": return "🌪️ Anemo";
            case "geo": return "☄️ Geo";
            case "dendro": return "🌿 Dendro";
            default: return element;
        }
    }

    private string NormalizeElement(string element)
    {
        if (string.IsNullOrEmpty(element))
            return "";

        string normalized = element.ToLower().Trim();

        switch (normalized)
        {
            case "pyro": return "pyro";
            case "hydro": return "hydro";
            case "cryo": return "cryo";
            case "electro":
            case "électro": return "electro";
            case "anemo":
            case "anémo": return "anemo";
            case "geo":
            case "géo": return "geo";
            case "dendro": return "dendro";
            default: return normalized;
        }
    }

    private void EnsureViewerDefaults(JObject viewer, string username)
    {
        if (viewer["username"] == null || viewer["username"].Type == JTokenType.Null)
            viewer["username"] = username;

        EnsureInt(viewer, "primogems", 0);
        EnsureInt(viewer, "moras", 0);

        EnsureObject(viewer, "particles");
        EnsureObject(viewer, "stats");
        EnsureObject(viewer, "box");
        EnsureArray(viewer, "team");
        EnsureArray(viewer, "tradeRequests");
        EnsureObject(viewer, "pity");
        EnsureObject(viewer, "guarantee");
        EnsureObject(viewer, "options");
        EnsureObject(viewer, "dates");
        EnsureObject(viewer, "missions");
        EnsureArray(viewer, "usedCodes");

        JObject particles = viewer["particles"] as JObject;
        EnsureInt(particles, "pyro", 0);
        EnsureInt(particles, "hydro", 0);
        EnsureInt(particles, "cryo", 0);
        EnsureInt(particles, "electro", 0);
        EnsureInt(particles, "anemo", 0);
        EnsureInt(particles, "geo", 0);
        EnsureInt(particles, "dendro", 0);

        JObject stats = viewer["stats"] as JObject;
        EnsureInt(stats, "totalPrimosEarned", 0);
        EnsureInt(stats, "totalMorasEarned", 0);

        JObject options = viewer["options"] as JObject;
        EnsureString(options, "boxSort", "a");
        EnsureBool(options, "boxSortDescending", false);

        JObject missions = viewer["missions"] as JObject;
        EnsureObject(missions, "daily");
    }

    private void EnsureObject(JObject parent, string key)
    {
        if (parent[key] == null || parent[key].Type == JTokenType.Null || parent[key].Type != JTokenType.Object)
            parent[key] = new JObject();
    }

    private void EnsureArray(JObject parent, string key)
    {
        if (parent[key] == null || parent[key].Type == JTokenType.Null || parent[key].Type != JTokenType.Array)
            parent[key] = new JArray();
    }

    private void EnsureInt(JObject obj, string key, int defaultValue)
    {
        if (obj[key] == null || obj[key].Type == JTokenType.Null)
            obj[key] = defaultValue;
    }

    private void EnsureString(JObject obj, string key, string defaultValue)
    {
        if (obj[key] == null || obj[key].Type == JTokenType.Null)
            obj[key] = defaultValue;
    }

    private void EnsureBool(JObject obj, string key, bool defaultValue)
    {
        if (obj[key] == null || obj[key].Type == JTokenType.Null)
            obj[key] = defaultValue;
    }

    private int GetInt(JObject obj, string key)
    {
        if (obj == null || obj[key] == null || obj[key].Type == JTokenType.Null)
            return 0;

        int value;

        if (int.TryParse(obj[key].ToString(), out value))
            return value;

        return 0;
    }

    private void AddInt(JObject obj, string key, int amount)
    {
        int current = GetInt(obj, key);
        obj[key] = current + amount;
    }

    private void AddStat(JObject viewer, string statName, int amount)
    {
        JObject stats = viewer["stats"] as JObject;

        if (stats == null)
        {
            stats = new JObject();
            viewer["stats"] = stats;
        }

        AddInt(stats, statName, amount);
    }

    private string NormalizeUsername(string username)
    {
        if (string.IsNullOrEmpty(username))
            return "";

        username = username.Trim();

        if (username.StartsWith("@"))
            username = username.Substring(1);

        return username.ToLower();
    }

    private void SendSuccess(string message) { CPH.SendMessage($"✅ {message}"); }
    private void SendWarning(string message) { CPH.SendMessage($"⚠️ {message}"); }
    private void SendError(string message) { CPH.SendMessage($"❌ {message}"); }
}

public class GiftCodeFile
{
    public List<GiftCodeData> codes { get; set; }
}

public class GiftCodeData
{
    public string code { get; set; }
    public string expiresAt { get; set; }
    public int month { get; set; }
    public bool annuallyRenewable { get; set; }
    public GiftCodeRewards rewards { get; set; }
    public string message { get; set; }
}

public class GiftCodeRewards
{
    public int primogems { get; set; }
    public int moras { get; set; }
    public ElementParticles particles { get; set; }
}

public class ElementParticles
{
    public int pyro { get; set; }
    public int hydro { get; set; }
    public int cryo { get; set; }
    public int electro { get; set; }
    public int anemo { get; set; }
    public int geo { get; set; }
    public int dendro { get; set; }
}
