  - !concours

  - !concours spectateur

using System;
using System.IO;
using System.Collections.Generic;
using Newtonsoft.Json.Linq;

public class CPHInline
{
    private const int MaxParticipants = 4;
    private const int MaxSpectators = 10;
    private const int WinScore = 50;
    private const int FirstPlacePrimosReward = 800;
    private const int SecondPlacePrimosReward = 400;
    private const int ThirdPlacePrimosReward = 200;

    private static Random random = new Random();

    public bool Execute()
    {
        string folderPath = @"C:\Users\axeld\Documents\Kichnifou\Twitch\Streambot\Data";
        string viewersPath = Path.Combine(folderPath, "viewers_data.json");
        string c6CharactersPath = Path.Combine(folderPath, "c6_characters.json");
        string contestsPath = Path.Combine(folderPath, "contests_data.json");

        if (!args.ContainsKey("userName") || args["userName"] == null)
            return false;

        string username = NormalizeUsername(args["userName"].ToString());
        string rawInput = GetCommandInput();

        if (!File.Exists(viewersPath))
        {
            SendWarning(username + ", ton profil n’existe pas encore. Écris d’abord un message dans le chat.");
            return true;
        }

        JObject viewers = LoadJObject(viewersPath);

        if (viewers == null)
        {
            SendError("Impossible de lire viewers_data.json.");
            return true;
        }

        JObject viewer = viewers[username] as JObject;

        if (viewer == null)
        {
            SendWarning(username + ", ton profil n’existe pas encore. Écris d’abord un message dans le chat.");
            return true;
        }

        JObject c6Characters = LoadOrCreateJObject(c6CharactersPath);
        JObject contests = LoadOrCreateContestsData(contestsPath);

        EnsureContestsDefaults(contests);

        DateTime now = DateTime.Now;
        ResetDailyThemeIfNeeded(contests, now);
        ClearOldContestIfNeeded(contests, now);

        ParsedInput parsed = ParseInput(rawInput);

        if (string.IsNullOrEmpty(parsed.command))
        {
            ShowHelp();
            return true;
        }

        string command = parsed.command;
        string extra = parsed.extra;

        if (command == "open")
        {
            HandleOpen(contests, c6Characters, username, extra, now, contestsPath);
            return true;
        }

        if (command == "participant" || command == "participer")
        {
            HandleParticipant(contests, c6Characters, username, extra, contestsPath);
            return true;
        }

        if (command == "spectateur" || command == "spectate" || command == "viewer")
        {
            HandleSpectator(contests, username, contestsPath);
            return true;
        }

        if (command == "annuler" || command == "cancel")
        {
            HandleCancel(contests, username, contestsPath);
            return true;
        }

        if (command == "start" || command == "lancer")
        {
            HandleStart(contests, viewers, c6Characters, username, now, contestsPath, viewersPath, c6CharactersPath);
            return true;
        }

        if (command == "basique" || command == "basic")
        {
            HandlePlayerMove(contests, viewers, c6Characters, username, false, contestsPath, viewersPath, c6CharactersPath);
            return true;
        }

        if (command == "risque" || command == "risqué" || command == "risk")
        {
            HandlePlayerMove(contests, viewers, c6Characters, username, true, contestsPath, viewersPath, c6CharactersPath);
            return true;
        }

        HandleSupport(contests, viewers, c6Characters, username, rawInput, contestsPath, viewersPath, c6CharactersPath);
        return true;
    }

    private void ShowHelp()
    {
        SendSuccess("🎪 !concours open NomPerso = ouvrir | participant NomPerso = rejoindre | spectateur = regarder | start = lancer | annuler = annuler avant start | basique/risque = jouer | Nom = soutenir");
    }

    private void HandleOpen(JObject contests, JObject c6Characters, string username, string characterName, DateTime now, string contestsPath)
    {
        JObject current = GetObject(contests, "currentContest");
        string status = GetString(current, "status");

        if (string.IsNullOrEmpty(characterName))
        {
            SendWarning(username + ", indique le perso à inscrire : !concours open NomPerso");
            return;
        }

        if (status == "open" || status == "running" || status == "support")
        {
            SendWarning("Un concours est déjà ouvert ou en cours.");
            return;
        }

        string today = now.ToString("yyyy-MM-dd");

        if (IsDailyLocked(contests, username, today))
        {
            SendWarning(username + ", tu as déjà participé à un concours aujourd’hui.");
            return;
        }

        JObject legend = FindAvailableLegend(c6Characters, contests, username, characterName, today);

        if (legend == null)
        {
            SendWarning(username + ", ce perso n’est pas une légende C6 disponible aujourd’hui.");
            return;
        }

        string theme = GetOrCreateDailyTheme(contests, now);
        ResetCurrentContest(current);

        current["status"] = "open";
        current["organizer"] = username;
        current["theme"] = theme;
        current["createdAt"] = FormatDateTime(now);
        current["startedAt"] = "";
        current["phase"] = "lobby";
        current["round"] = 0;
        current["currentTurnIndex"] = 0;
        current["activePlayer"] = "";
        current["activeSpectator"] = "";

        JArray participants = new JArray();
        participants.Add(CreatePlayerParticipant(username, legend, theme));
        current["participants"] = participants;

        current["spectators"] = new JArray();
        current["turnOrder"] = new JArray();
        current["scores"] = new JObject();

        JObject pendingSupport = new JObject();
        pendingSupport["spectator"] = "";
        pendingSupport["availableTargets"] = new JArray();
        current["pendingSupport"] = pendingSupport;

        SaveJObjectNoBackup(contestsPath, contests);

        CPH.SendMessage("🎪 Concours ouvert par " + username + " ! Thème du jour : " + GetThemeDisplayName(theme) + " | Participants 1/4 | Spectateurs 0/10 | Rejoindre : !concours participant [perso] ou !concours spectateur | Lancer : !concours start");
    }

    private void HandleParticipant(JObject contests, JObject c6Characters, string username, string characterName, string contestsPath)
    {
        JObject current = GetObject(contests, "currentContest");

        if (string.IsNullOrEmpty(characterName))
        {
            SendWarning(username + ", indique le perso à inscrire : !concours participant NomPerso");
            return;
        }

        string status = GetString(current, "status");

        if (status != "open" &&
            status != "running" &&
            status != "support")
        {
            SendWarning("Aucun concours en cours.");
            return;
        }

        JArray participants = GetArray(current, "participants");
        JArray spectators = GetArray(current, "spectators");

        if (participants.Count >= MaxParticipants)
        {
            SendWarning("Le concours a déjà 4 participants.");
            return;
        }

        if (IsParticipant(participants, username))
        {
            SendWarning(username + ", tu participes déjà à ce concours.");
            return;
        }

        if (IsSpectator(spectators, username))
        {
            SendWarning(username + ", tu es déjà spectateur de ce concours.");
            return;
        }

        string today = GetString(contests, "date");

        if (IsDailyLocked(contests, username, today))
        {
            SendWarning(username + ", tu as déjà participé à un concours aujourd’hui.");
            return;
        }

        JObject legend = FindAvailableLegend(c6Characters, contests, username, characterName, today);

        if (legend == null)
        {
            SendWarning(username + ", ce perso n’est pas une légende C6 disponible aujourd’hui.");
            return;
        }

        string theme = GetString(current, "theme");

        participants.Add(CreatePlayerParticipant(username, legend, theme));
        current["participants"] = participants;

        SaveJObjectNoBackup(contestsPath, contests);

        SendSuccess(username + " rejoint le concours avec " + GetElementEmoji(GetString(legend, "element")) + " " + GetString(legend, "name") + " ! Participants " + participants.Count + "/4");
    }

    private void HandleSpectator(JObject contests, string username, string contestsPath)
    {
        JObject current = GetObject(contests, "currentContest");

        if (GetString(current, "status") != "open")
        {
            SendWarning("Aucun concours ouvert aux spectateurs.");
            return;
        }

        JArray participants = GetArray(current, "participants");
        JArray spectators = GetArray(current, "spectators");

        if (IsParticipant(participants, username))
        {
            SendWarning(username + ", tu participes déjà au concours.");
            return;
        }

        if (IsSpectator(spectators, username))
        {
            SendWarning(username + ", tu es déjà spectateur.");
            return;
        }

        if (spectators.Count >= MaxSpectators)
        {
            SendWarning("Le concours a déjà 10 spectateurs.");
            return;
        }

        spectators.Add(username);
        current["spectators"] = spectators;

        SaveJObjectNoBackup(contestsPath, contests);

        SendSuccess(username + " rejoint le public ! Spectateurs " + spectators.Count + "/10");
    }

   private void HandleCancel(JObject contests, string username, string contestsPath)
    {
        JObject current = GetObject(contests, "currentContest");
        string status = GetString(current, "status");

        if (status != "open" && status != "running")
        {
            SendWarning("Aucun concours en cours.");
            return;
        }

        JArray participants = GetArray(current, "participants");
        JArray spectators = GetArray(current, "spectators");

        // 1) Si c’est un participant
        for (int i = 0; i < participants.Count; i++)
        {
            JObject participant = participants[i] as JObject;

            if (participant == null || GetBool(participant, "isBot"))
                continue;

            if (GetString(participant, "username") != username)
                continue;

            if (status == "open")
            {
                participants.RemoveAt(i);
                current["participants"] = participants;
                SaveJObjectNoBackup(contestsPath, contests);
                SendSuccess(username + " quitte le concours.");
                return;
            }

            ReplaceParticipantByBot(current, participant, i);

            if (!HasHumanParticipant(participants))
            {
                ResetCurrentContest(current);
                SaveJObjectNoBackup(contestsPath, contests);
                SendSuccess("Tous les participants ont quitté le concours. Concours annulé automatiquement.");
                return;
            }

            SaveJObjectNoBackup(contestsPath, contests);
            SendSuccess(username + " quitte le concours. Un bot prend sa place avec les mêmes points.");
            return;
        }

        // 2) Si c’est un spectateur
        for (int i = spectators.Count - 1; i >= 0; i--)
        {
            if (spectators[i] != null && spectators[i].ToString() == username)
            {
                spectators.RemoveAt(i);
                current["spectators"] = spectators;

                if (GetString(current, "activeSpectator") == username)
                {
                    current["phase"] = "participant_turn";
                    current["activeSpectator"] = "";
                    GetObject(current, "pendingSupport")["spectator"] = "";
                    GetObject(current, "pendingSupport")["availableTargets"] = new JArray();
                }

                SaveJObjectNoBackup(contestsPath, contests);
                SendSuccess(username + " quitte le public.");
                return;
            }
        }

        // 3) Si c’est l’organisateur
        if (GetString(current, "organizer") != username && !IsContestAdmin(username))
        {
            SendWarning("Seul l’organisateur, Kichnifou, un participant ou un spectateur peut annuler/quitter.");
            return;
        }

        ResetCurrentContest(current);
        SaveJObjectNoBackup(contestsPath, contests);

        SendSuccess("Concours annulé par l’organisateur. Un nouveau concours peut être lancé.");
    }

    private void ReplaceParticipantByBot(JObject current, JObject oldParticipant, int participantIndex)
    {
        string oldId = GetString(oldParticipant, "id");
        string theme = GetString(current, "theme");

        JObject bot = CreateBotParticipant(participantIndex + 1, theme);

        bot["themeStat"] = GetInt(oldParticipant, "themeStat");
        bot["score"] = GetInt(oldParticipant, "score");

        JArray participants = GetArray(current, "participants");
        participants[participantIndex] = bot;
        current["participants"] = participants;

        JArray turnOrder = GetArray(current, "turnOrder");

        for (int i = 0; i < turnOrder.Count; i++)
        {
            if (turnOrder[i] != null && turnOrder[i].ToString() == oldId)
                turnOrder[i] = GetString(bot, "id");
        }

        current["turnOrder"] = turnOrder;

        JObject scores = GetObject(current, "scores");
        scores.Remove(oldId);
        scores[GetString(bot, "id")] = GetInt(bot, "score");
        current["scores"] = scores;

        if (GetString(current, "activePlayer") == oldId)
            current["activePlayer"] = GetString(bot, "id");
    }

    private bool HasHumanParticipant(JArray participants)
    {
        for (int i = 0; i < participants.Count; i++)
        {
            JObject participant = participants[i] as JObject;

            if (participant == null)
                continue;

            if (!GetBool(participant, "isBot"))
                return true;
        }

        return false;
    }

    private bool IsContestAdmin(string username)
    {
        return username == "kichnifou";
    }

    private void HandleStart(JObject contests, JObject viewers, JObject c6Characters, string username, DateTime now, string contestsPath, string viewersPath, string c6CharactersPath)
    {
        JObject current = GetObject(contests, "currentContest");

        if (GetString(current, "status") != "open")
        {
            SendWarning("Aucun concours ouvert à lancer.");
            return;
        }

        if (GetString(current, "organizer") != username)
        {
            SendWarning("Seul l’organisateur peut lancer le concours.");
            return;
        }

        JArray participants = GetArray(current, "participants");
        string theme = GetString(current, "theme");

        while (participants.Count < MaxParticipants)
        {
            participants.Add(CreateBotParticipant(participants.Count + 1, theme));
        }

        current["participants"] = participants;
        current["status"] = "running";
        current["phase"] = "participant_turn";
        current["startedAt"] = FormatDateTime(now);
        current["round"] = 1;

        JObject scores = new JObject();

        for (int i = 0; i < participants.Count; i++)
        {
            JObject participant = participants[i] as JObject;

            if (participant == null)
                continue;

            scores[GetString(participant, "id")] = 0;
            participant["score"] = 0;
        }

        current["scores"] = scores;

        JArray turnOrder = CreateTurnOrder(participants);
        current["turnOrder"] = turnOrder;
        current["currentTurnIndex"] = 0;

        string firstId = turnOrder.Count > 0 ? turnOrder[0].ToString() : "";
        current["activePlayer"] = firstId;

        SaveJObjectNoBackup(contestsPath, contests);

        CPH.SendMessage("🎪 Concours lancé ! Thème : " + GetThemeDisplayName(theme) + " | Objectif : 50 points | Ordre : " + FormatTurnOrder(current) + " | Premier : " + GetParticipantDisplayNameById(current, firstId));

        AdvanceContest(contests, viewers, c6Characters, contestsPath, viewersPath, c6CharactersPath);
    }

    private void HandlePlayerMove(JObject contests, JObject viewers, JObject c6Characters, string username, bool risky, string contestsPath, string viewersPath, string c6CharactersPath)
    {
        JObject current = GetObject(contests, "currentContest");

        if (GetString(current, "status") != "running" || GetString(current, "phase") != "participant_turn")
        {
            SendWarning("Ce n’est pas le moment de jouer un move.");
            return;
        }

        string activeId = GetString(current, "activePlayer");
        JObject participant = FindParticipantById(current, activeId);

        if (participant == null)
        {
            SendError("Participant actif introuvable.");
            return;
        }

        if (GetBool(participant, "isBot"))
        {
            AdvanceContest(contests, viewers, c6Characters, contestsPath, viewersPath, c6CharactersPath);
            return;
        }

        if (GetString(participant, "username") != username)
        {
            SendWarning("C’est au tour de " + GetString(participant, "username") + ".");
            return;
        }

        ApplyMove(current, participant, risky, false);

        if (CheckWinnerAndFinish(contests, viewers, c6Characters, participant, contestsPath, viewersPath, c6CharactersPath))
            return;

        MoveToNextTurnOrSupport(current);
        SaveJObjectNoBackup(contestsPath, contests);

        AdvanceContest(contests, viewers, c6Characters, contestsPath, viewersPath, c6CharactersPath);
    }

    private void HandleSupport(JObject contests, JObject viewers, JObject c6Characters, string username, string targetInput, string contestsPath, string viewersPath, string c6CharactersPath)
    {
        JObject current = GetObject(contests, "currentContest");

        if (GetString(current, "status") != "running" || GetString(current, "phase") != "support_turn")
        {
            SendWarning("Paramètre inconnu. Tape !concours pour voir l’aide.");
            return;
        }

        string activeSpectator = GetString(current, "activeSpectator");

        if (activeSpectator != username)
        {
            SendWarning("C’est à " + activeSpectator + " de soutenir un participant.");
            return;
        }

        JObject target = FindParticipantByName(current, targetInput);

        if (target == null)
        {
            SendWarning("Participant introuvable. Choisis : " + FormatParticipantTargets(current));
            return;
        }

        int bonus = random.Next(1, 4);
        AddScore(current, target, bonus);

        string targetName = GetBool(target, "isBot")
        ? GetString(target, "characterName")
        : GetString(target, "username");

        CPH.SendMessage("🎭 " + username + " soutient " + targetName + " : +" + bonus + " points ! Total : " + GetInt(target, "score") + "/" + WinScore);

        if (CheckWinnerAndFinish(contests, viewers, c6Characters, target, contestsPath, viewersPath, c6CharactersPath))
            return;

        current["phase"] = "participant_turn";
        current["activeSpectator"] = "";
        JObject pendingSupport = GetObject(current, "pendingSupport");
        pendingSupport["spectator"] = "";
        pendingSupport["availableTargets"] = new JArray();

        AddInt(current, "round", 1);
        current["currentTurnIndex"] = 0;

        JArray turnOrder = GetArray(current, "turnOrder");
        if (turnOrder.Count > 0)
            current["activePlayer"] = turnOrder[0].ToString();

        SaveJObjectNoBackup(contestsPath, contests);

        AdvanceContest(contests, viewers, c6Characters, contestsPath, viewersPath, c6CharactersPath);
    }

    private void AdvanceContest(JObject contests, JObject viewers, JObject c6Characters, string contestsPath, string viewersPath, string c6CharactersPath)
    {
        JObject current = GetObject(contests, "currentContest");

        if (GetString(current, "status") != "running")
            return;

        int safety = 0;

        while (safety < 20)
        {
            safety++;

            if (GetString(current, "phase") == "support_turn")
            {
                SaveJObjectNoBackup(contestsPath, contests);
                CPH.SendMessage("🎭 " + GetString(current, "activeSpectator") + ", soutiens un participant : " + FormatParticipantTargets(current) + " | Commande : !concours Nom");
                return;
            }

            string activeId = GetString(current, "activePlayer");
            JObject participant = FindParticipantById(current, activeId);

            if (participant == null)
            {
                MoveToNextTurnOrSupport(current);
                continue;
            }

            if (!GetBool(participant, "isBot"))
            {
                SaveJObjectNoBackup(contestsPath, contests);
               CPH.SendMessage("🎲 Tour de " + GetParticipantCommandName(participant) + " | Commandes : !concours basique ou !concours risque");
                return;
            }

            bool risky = random.Next(1, 101) <= 40;
            ApplyMove(current, participant, risky, true);

            if (CheckWinnerAndFinish(contests, viewers, c6Characters, participant, contestsPath, viewersPath, c6CharactersPath))
                return;

            MoveToNextTurnOrSupport(current);
            SaveJObjectNoBackup(contestsPath, contests);
        }
    }

    private void ApplyMove(JObject current, JObject participant, bool risky, bool isAuto)
    {
        string theme = GetString(current, "theme");
        int statValue = GetInt(participant, "themeStat");
        int basePoints = GetPointsFromStat(statValue);
        int gained = basePoints;
        string actionText = risky ? "risqué" : "basique";

        if (risky)
        {
            int roll = random.Next(1, 4);

            if (roll == 1)
                gained = 0;
            else if (roll == 2)
                gained = basePoints;
            else
                gained = basePoints * 2;
        }

        AddScore(current, participant, gained);

        string prefix = isAuto ? "🤖 " : "🎲 ";
        CPH.SendMessage(prefix + GetParticipantDisplayName(participant) + " fait un move " + actionText + " : +" + gained + " points | Total : " + GetInt(participant, "score") + "/" + WinScore);
    }

    private bool CheckWinnerAndFinish(JObject contests, JObject viewers, JObject c6Characters, JObject participant, string contestsPath, string viewersPath, string c6CharactersPath)
    {
        if (GetInt(participant, "score") < WinScore)
            return false;

        FinishContest(contests, viewers, c6Characters, participant, contestsPath, viewersPath, c6CharactersPath);
        return true;
    }

    private void FinishContest(JObject contests, JObject viewers, JObject c6Characters, JObject winner, string contestsPath, string viewersPath, string c6CharactersPath)
    {
        JObject current = GetObject(contests, "currentContest");
        string theme = GetString(current, "theme");
        string today = GetString(contests, "date");

        JArray participants = GetArray(current, "participants");

        for (int i = 0; i < participants.Count; i++)
        {
            JObject participant = participants[i] as JObject;

            if (participant == null || GetBool(participant, "isBot"))
                continue;

            string username = GetString(participant, "username");
            int characterId = GetInt(participant, "characterId");

            LockDailyParticipant(contests, username, characterId, today);
            AddContestParticipation(c6Characters, username, characterId, theme);

            if (GetString(participant, "id") == GetString(winner, "id"))
                AddContestWin(c6Characters, username, characterId, theme);
        }

        bool winnerIsBot = GetBool(winner, "isBot");
        string titleText = "";

        GiveRankingPrimosRewards(viewers, participants);
        SaveJObjectNoBackup(viewersPath, viewers);

        if (!winnerIsBot)
        {
            string winnerUsername = GetString(winner, "username");
            int winnerCharacterId = GetInt(winner, "characterId");

            titleText = UpgradeTitle(c6Characters, winnerUsername, winnerCharacterId, theme);
        }

        SaveJObjectNoBackup(c6CharactersPath, c6Characters);

        AddHistoryEntry(contests, winner, theme);

        string phrase = GetVictoryPhrase(theme);
        string message = "🏆 " + GetParticipantCommandName(winner) + " remporte le concours " + GetThemeDisplayName(theme) + " ! " + phrase;
        message += " | Classement : " + FormatFinalRanking(participants);

        if (!winnerIsBot)
        {
            message += " | Récompenses : 🥇" + FirstPlacePrimosReward + " 💠 | 🥈" + SecondPlacePrimosReward + " 💠 | 🥉" + ThirdPlacePrimosReward + " 💠";

            if (!string.IsNullOrEmpty(titleText))
                message += " | 👑 " + titleText;
        }

        ResetCurrentContest(current);
        SaveJObjectNoBackup(contestsPath, contests);

        CPH.SendMessage(message);
    }

    private void MoveToNextTurnOrSupport(JObject current)
    {
        JArray turnOrder = GetArray(current, "turnOrder");
        int index = GetInt(current, "currentTurnIndex");
        index++;

        if (index < turnOrder.Count)
        {
            current["currentTurnIndex"] = index;
            current["activePlayer"] = turnOrder[index].ToString();
            current["phase"] = "participant_turn";
            return;
        }

        JArray spectators = GetArray(current, "spectators");

        if (spectators.Count > 0)
        {
            string spectator = spectators[random.Next(spectators.Count)].ToString();

            current["phase"] = "support_turn";
            current["activePlayer"] = "";
            current["activeSpectator"] = spectator;

            JObject pendingSupport = GetObject(current, "pendingSupport");
            pendingSupport["spectator"] = spectator;
            pendingSupport["availableTargets"] = GetSupportTargets(current);
            current["pendingSupport"] = pendingSupport;
            return;
        }

        AddInt(current, "round", 1);
        current["currentTurnIndex"] = 0;

        if (turnOrder.Count > 0)
            current["activePlayer"] = turnOrder[0].ToString();

        current["phase"] = "participant_turn";
    }

    private void GiveRankingPrimosRewards(JObject viewers, JArray participants)
    {
        List<JObject> ranking = GetSortedRanking(participants);

        for (int i = 0; i < ranking.Count; i++)
        {
            JObject participant = ranking[i];

            if (GetBool(participant, "isBot"))
                continue;

            int reward = 0;

            if (i == 0)
                reward = FirstPlacePrimosReward;
            else if (i == 1)
                reward = SecondPlacePrimosReward;
            else if (i == 2)
                reward = ThirdPlacePrimosReward;

            if (reward <= 0)
                continue;

            string username = GetString(participant, "username");
            JObject viewer = viewers[username] as JObject;

            if (viewer == null)
                continue;

            AddInt(viewer, "primogems", reward);
            AddStat(viewer, "totalPrimosEarned", reward);
        }
    }

    private List<JObject> GetSortedRanking(JArray participants)
    {
        List<JObject> list = new List<JObject>();

        for (int i = 0; i < participants.Count; i++)
        {
            JObject participant = participants[i] as JObject;

            if (participant != null)
                list.Add(participant);
        }

        list.Sort(delegate(JObject a, JObject b)
        {
            return GetInt(b, "score").CompareTo(GetInt(a, "score"));
        });

        return list;
    }

    private JObject FindAvailableLegend(JObject c6Characters, JObject contests, string username, string characterName, string today)
    {
        JObject userData = c6Characters[username] as JObject;

        if (userData == null)
            return null;

        JObject characters = userData["characters"] as JObject;

        if (characters == null || characters.Count == 0)
            return null;

        if (!string.IsNullOrEmpty(characterName))
        {
            JObject found = FindLegendByName(characters, characterName);

            if (found == null)
                return null;

            if (IsCharacterLockedToday(contests, username, GetInt(found, "characterId"), today))
                return null;

            return found;
        }

        foreach (KeyValuePair<string, JToken> entry in characters)
        {
            JObject legend = entry.Value as JObject;

            if (legend == null)
                continue;

            int characterId = GetInt(legend, "characterId");

            if (!IsCharacterLockedToday(contests, username, characterId, today))
                return legend;
        }

        return null;
    }

    private JObject FindLegendByName(JObject characters, string input)
    {
        string wanted = NormalizeText(input);

        foreach (KeyValuePair<string, JToken> entry in characters)
        {
            JObject character = entry.Value as JObject;

            if (character == null)
                continue;

            if (NormalizeText(GetString(character, "name")) == wanted)
                return character;
        }

        foreach (KeyValuePair<string, JToken> entry in characters)
        {
            JObject character = entry.Value as JObject;

            if (character == null)
                continue;

            if (NormalizeText(GetString(character, "name")).Contains(wanted))
                return character;
        }

        return null;
    }

    private void ClearOldContestIfNeeded(JObject contests, DateTime now)
    {
        JObject current = GetObject(contests, "currentContest");
        string status = GetString(current, "status");

        if (status == "none")
            return;

        string createdAt = GetString(current, "createdAt");

        if (string.IsNullOrEmpty(createdAt))
            return;

        DateTime parsed;

        if (!DateTime.TryParse(createdAt, out parsed))
            return;

        if (parsed.ToString("yyyy-MM-dd") != now.ToString("yyyy-MM-dd"))
            ResetCurrentContest(current);
    }

    private string UpgradeTitle(JObject c6Characters, string username, int characterId, string theme)
    {
        JObject legend = GetLegendById(c6Characters, username, characterId);

        if (legend == null)
            return "";

        JObject titles = legend["titles"] as JObject;

        if (titles == null)
        {
            titles = new JObject();
            legend["titles"] = titles;
        }

        string currentTitle = GetString(titles, theme);
        string baseName = GetTitleBaseName(theme);
        string nextRank = GetNextTitleRank(currentTitle);

        if (string.IsNullOrEmpty(nextRank))
            return "Titre déjà au rang max";

        string title = BuildTitle(baseName, nextRank);
        titles[theme] = title;

        return "Titre obtenu : " + title;
    }

    private ParsedInput ParseInput(string rawInput)
    {
        ParsedInput parsed = new ParsedInput();
        parsed.command = "";
        parsed.extra = "";

        if (string.IsNullOrWhiteSpace(rawInput))
            return parsed;

        string trimmed = rawInput.Trim();
        int spaceIndex = trimmed.IndexOf(" ");

        if (spaceIndex < 0)
        {
            parsed.command = trimmed.ToLower();
            parsed.extra = "";
            return parsed;
        }

        parsed.command = trimmed.Substring(0, spaceIndex).ToLower().Trim();
        parsed.extra = trimmed.Substring(spaceIndex + 1).Trim();

        return parsed;
    }

    private JObject LoadOrCreateJObject(string path)
    {
        if (!File.Exists(path))
            return new JObject();

        try
        {
            string json = File.ReadAllText(path);

            if (string.IsNullOrWhiteSpace(json))
                return new JObject();

            JObject data = JObject.Parse(json);

            if (data == null)
                return new JObject();

            return data;
        }
        catch
        {
            return new JObject();
        }
    }

    private JObject LoadJObject(string path)
    {
        try
        {
            string json = File.ReadAllText(path);

            if (string.IsNullOrWhiteSpace(json))
                return new JObject();

            return JObject.Parse(json);
        }
        catch
        {
            return null;
        }
    }

    private void SaveJObjectNoBackup(string path, JObject data)
    {
        string folder = Path.GetDirectoryName(path);

        if (!Directory.Exists(folder))
            Directory.CreateDirectory(folder);

        string tempPath = path + ".tmp";
        File.WriteAllText(tempPath, data.ToString(Newtonsoft.Json.Formatting.Indented));

        if (File.Exists(path))
            File.Delete(path);

        File.Move(tempPath, path);
    }

    private JObject GetObject(JObject parent, string key)
    {
        if (parent == null)
            return new JObject();

        JObject obj = parent[key] as JObject;

        if (obj == null)
        {
            obj = new JObject();
            parent[key] = obj;
        }

        return obj;
    }

    private JArray GetArray(JObject parent, string key)
    {
        if (parent == null)
            return new JArray();

        JArray array = parent[key] as JArray;

        if (array == null)
        {
            array = new JArray();
            parent[key] = array;
        }

        return array;
    }

    private void EnsureObject(JObject parent, string key)
    {
        if (parent[key] == null || parent[key].Type == JTokenType.Null || parent[key].Type != JTokenType.Object)
            parent[key] = new JObject();
    }

    private void EnsureArray(JObject parent, string key)
    {
        if (parent[key] == null || parent[key].Type == JTokenType.Null || parent[key].Type != JTokenType.Array)
            parent[key] = new JArray();
    }

    private void EnsureString(JObject obj, string key, string defaultValue)
    {
        if (obj[key] == null || obj[key].Type == JTokenType.Null)
            obj[key] = defaultValue;
    }

    private void EnsureInt(JObject obj, string key, int defaultValue)
    {
        if (obj[key] == null || obj[key].Type == JTokenType.Null)
            obj[key] = defaultValue;
    }

    private int GetInt(JObject obj, string key)
    {
        if (obj == null || obj[key] == null || obj[key].Type == JTokenType.Null)
            return 0;

        int value;

        if (int.TryParse(obj[key].ToString(), out value))
            return value;

        return 0;
    }

    private bool GetBool(JObject obj, string key)
    {
        if (obj == null || obj[key] == null || obj[key].Type == JTokenType.Null)
            return false;

        bool value;

        if (bool.TryParse(obj[key].ToString(), out value))
            return value;

        return false;
    }

    private string GetString(JObject obj, string key)
    {
        if (obj == null || obj[key] == null || obj[key].Type == JTokenType.Null)
            return "";

        return obj[key].ToString();
    }

    private void AddInt(JObject obj, string key, int amount)
    {
        int current = GetInt(obj, key);
        obj[key] = current + amount;
    }

    private string NormalizeUsername(string username)
    {
        if (string.IsNullOrEmpty(username))
            return "";

        username = username.Trim();

        if (username.StartsWith("@"))
            username = username.Substring(1);

        return username.ToLower();
    }

    private string NormalizeElement(string element)
    {
        if (string.IsNullOrEmpty(element))
            return "";

        string normalized = element.ToLower().Trim();

        switch (normalized)
        {
            case "pyro": return "pyro";
            case "hydro": return "hydro";
            case "cryo": return "cryo";
            case "electro":
            case "électro": return "electro";
            case "anemo":
            case "anémo": return "anemo";
            case "geo":
            case "géo": return "geo";
            case "dendro": return "dendro";
            default: return normalized;
        }
    }

    private string GetElementEmoji(string element)
    {
        switch (NormalizeElement(element))
        {
            case "pyro": return "🔥";
            case "hydro": return "💧";
            case "cryo": return "❄️";
            case "electro": return "⚡";
            case "anemo": return "🌪️";
            case "geo": return "☄️";
            case "dendro": return "🌿";
            default: return "⭐";
        }
    }

    private string NormalizeText(string text)
    {
        if (string.IsNullOrEmpty(text))
            return "";

        string value = text.ToLower().Trim();

        value = value.Replace("à", "a");
        value = value.Replace("á", "a");
        value = value.Replace("â", "a");
        value = value.Replace("ä", "a");
        value = value.Replace("ã", "a");

        value = value.Replace("ç", "c");

        value = value.Replace("è", "e");
        value = value.Replace("é", "e");
        value = value.Replace("ê", "e");
        value = value.Replace("ë", "e");

        value = value.Replace("ì", "i");
        value = value.Replace("í", "i");
        value = value.Replace("î", "i");
        value = value.Replace("ï", "i");

        value = value.Replace("ò", "o");
        value = value.Replace("ó", "o");
        value = value.Replace("ô", "o");
        value = value.Replace("ö", "o");
        value = value.Replace("õ", "o");

        value = value.Replace("ù", "u");
        value = value.Replace("ú", "u");
        value = value.Replace("û", "u");
        value = value.Replace("ü", "u");

        value = value.Replace("ÿ", "y");
        value = value.Replace("œ", "oe");
        value = value.Replace("æ", "ae");

        value = value.Replace("'", "");
        value = value.Replace("’", "");
        value = value.Replace("-", "");
        value = value.Replace("_", "");
        value = value.Replace(" ", "");

        return value;
    }

    private string FormatDateTime(DateTime date)
    {
        return date.ToString("yyyy-MM-dd HH:mm:ss");
    }

    private string FormatFinalRanking(JArray participants)
    {
        List<JObject> list = new List<JObject>();

        for (int i = 0; i < participants.Count; i++)
        {
            JObject participant = participants[i] as JObject;

            if (participant != null)
                list.Add(participant);
        }

        list.Sort(delegate(JObject a, JObject b)
        {
            return GetInt(b, "score").CompareTo(GetInt(a, "score"));
        });

        string text = "";

        for (int i = 0; i < list.Count; i++)
        {
            if (i > 0)
                text += " | ";

            text += (i + 1) + ". " + GetParticipantCommandName(list[i]) + " " + GetInt(list[i], "score") + " pts";
        }

        return text;
    }

    private void SendError(string message) { CPH.SendMessage("❌ " + message); }
