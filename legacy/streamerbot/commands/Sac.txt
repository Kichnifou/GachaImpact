/*
CE QUE FAIT LE CODE :

- Gère la commande !sac
- Lit viewers_data.json en JObject
- Ne crée PAS le viewer s’il n’existe pas
- Ne sauvegarde PAS viewers_data.json
- Affiche les primos
- Affiche les moras
- Affiche toutes les particules élémentaires
- Affiche le nombre de voeux possibles avec les primos possédés
- Affiche toujours les particules de l’élément du viewer en premier
- Préserve totalement la structure JSON existante car aucune écriture n’est faite
*/

using System;
using System.IO;
using System.Collections.Generic;
using Newtonsoft.Json.Linq;

public class CPHInline
{
    private const int PullCost = 160;

    public bool Execute()
    {
        string path = @"C:\Users\axeld\Documents\Kichnifou\Twitch\Streambot\Data\viewers_data.json";

        if (!args.ContainsKey("userName") || args["userName"] == null)
            return false;

        string username = NormalizeUsername(args["userName"].ToString());

        if (!File.Exists(path))
        {
            SendWarning($"{username}, ton profil n’existe pas encore.");
            return true;
        }

        JObject viewers = LoadJObject(path);

        if (viewers == null)
        {
            SendWarning($"{username}, impossible de lire les données.");
            return true;
        }

        JObject viewer = viewers[username] as JObject;

        if (viewer == null)
        {
            SendWarning($"{username}, ton profil n’existe pas encore.");
            return true;
        }

        string element = NormalizeElement(GetString(viewer, "element"));
        int primogems = GetInt(viewer, "primogems");
        int moras = GetInt(viewer, "moras");
        int possibleWishes = primogems / PullCost;

        JObject particles = viewer["particles"] as JObject;

        if (particles == null)
            particles = new JObject();

        string particlesDisplay = GetParticlesDisplay(particles, element);

        SendSuccess($"{username}, sac : 💠 {primogems} primos ({possibleWishes} voeux) | 💰 {moras} moras | {particlesDisplay}");

        return true;
    }

    private string GetParticlesDisplay(JObject particles, string mainElement)
    {
        List<string> parts = new List<string>();

        if (!string.IsNullOrEmpty(mainElement))
            AddElementDisplay(parts, particles, mainElement);

        string[] elements = { "pyro", "hydro", "cryo", "electro", "anemo", "geo", "dendro" };

        for (int i = 0; i < elements.Length; i++)
        {
            string element = elements[i];

            if (element != mainElement)
                AddElementDisplay(parts, particles, element);
        }

        return string.Join(" | ", parts.ToArray());
    }

    private void AddElementDisplay(List<string> parts, JObject particles, string element)
    {
        switch (NormalizeElement(element))
        {
            case "pyro":
                parts.Add($"🔥 {GetInt(particles, "pyro")}");
                break;

            case "hydro":
                parts.Add($"💧 {GetInt(particles, "hydro")}");
                break;

            case "cryo":
                parts.Add($"❄️ {GetInt(particles, "cryo")}");
                break;

            case "electro":
                parts.Add($"⚡ {GetInt(particles, "electro")}");
                break;

            case "anemo":
                parts.Add($"🌪️ {GetInt(particles, "anemo")}");
                break;

            case "geo":
                parts.Add($"☄️ {GetInt(particles, "geo")}");
                break;

            case "dendro":
                parts.Add($"🌿 {GetInt(particles, "dendro")}");
                break;
        }
    }

    private JObject LoadJObject(string path)
    {
        try
        {
            string json = File.ReadAllText(path);

            if (string.IsNullOrEmpty(json))
                return null;

            return JObject.Parse(json);
        }
        catch
        {
            return null;
        }
    }

    private int GetInt(JObject obj, string key)
    {
        if (obj == null || obj[key] == null || obj[key].Type == JTokenType.Null)
            return 0;

        int value;

        if (int.TryParse(obj[key].ToString(), out value))
            return value;

        return 0;
    }

    private string GetString(JObject obj, string key)
    {
        if (obj == null || obj[key] == null || obj[key].Type == JTokenType.Null)
            return "";

        return obj[key].ToString();
    }

    private string NormalizeUsername(string username)
    {
        if (string.IsNullOrEmpty(username))
            return "";

        username = username.Trim();

        if (username.StartsWith("@"))
            username = username.Substring(1);

        return username.ToLower();
    }

    private string NormalizeElement(string element)
    {
        if (string.IsNullOrEmpty(element))
            return "";

        string normalized = element.ToLower().Trim();

        switch (normalized)
        {
            case "pyro": return "pyro";
            case "hydro": return "hydro";
            case "cryo": return "cryo";
            case "electro":
            case "électro": return "electro";
            case "anemo":
            case "anémo": return "anemo";
            case "geo":
            case "géo": return "geo";
            case "dendro": return "dendro";
            default: return normalized;
        }
    }

    private void SendSuccess(string message) { CPH.SendMessage($"✅ {message}"); }
    private void SendWarning(string message) { CPH.SendMessage($"⚠️ {message}"); }
}
