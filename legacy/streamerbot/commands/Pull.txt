using System;
using System.IO;
using System.Collections.Generic;

public class CPHInline
{
    private const int PullCost = 160;
    private static Random random = new Random();

    public bool Execute()
    {
        string folderPath = @"C:\Users\axeld\Documents\Kichnifou\Twitch\Streambot\Data";
        string viewersPath = Path.Combine(folderPath, "viewers_data.json");
        string charactersPath = Path.Combine(folderPath, "genshin_characters.json");
        string passivesPath = Path.Combine(folderPath, "element_passives.json");
        string c6CharactersPath = Path.Combine(folderPath, "c6_characters.json");

        if (!args.ContainsKey("userName") || args["userName"] == null)
            return false;

        string username = NormalizeUsername(args["userName"].ToString());

        if (!File.Exists(viewersPath))
        {
            SendWarning($"{username}, ton profil n’existe pas encore. Écris d’abord un message dans le chat.");
            return true;
        }

        if (!File.Exists(charactersPath))
        {
            SendError("Aucun fichier personnage trouvé.");
            return true;
        }

        JObject viewers = LoadJObject(viewersPath);

        if (viewers == null)
        {
            SendError("Impossible de lire viewers_data.json.");
            return true;
        }

        JObject viewer = viewers[username] as JObject;

        if (viewer == null)
        {
            SendWarning($"{username}, ton profil n’existe pas encore. Écris d’abord un message dans le chat.");
            return true;
        }

        CharacterFile characterFile = LoadCharacters(charactersPath);
        ElementPassiveFile passiveFile = LoadPassives(passivesPath);

        if (characterFile == null || characterFile.characters == null)
        {
            SendError("Impossible de lire la liste des personnages.");
            return true;
        }

        EnsureViewerDefaultsJObject(viewer, username);

        List<CharacterData> featured5 = GetFeaturedCharacters(characterFile, 5);
        List<CharacterData> featured4 = GetFeaturedCharacters(characterFile, 4);

        if (featured5.Count < 3)
        {
            SendWarning("La bannière 5★ n’est pas correctement initialisée.");
            return true;
        }

        int selectedBannerCharacterId = GetInt(viewer, "selectedBannerCharacterId");

        if (selectedBannerCharacterId == 0)
        {
            SendWarning($"{username}, tu dois d’abord choisir ton personnage 5★ ciblé avec !select.");
            return true;
        }

        CharacterData selectedCharacter = FindCharacterInListById(featured5, selectedBannerCharacterId);

        if (selectedCharacter == null)
        {
            SendWarning($"{username}, ton personnage sélectionné n’est plus dans la bannière actuelle. Choisis-en un nouveau avec !select.");
            return true;
        }

        int pullCount = GetPullCount();

        if (pullCount <= 0)
        {
            SendWarning($"{username}, nombre de pulls invalide. Exemple : !pull ou !pull 10.");
            return true;
        }

        if (pullCount > 10)
        {
            SendWarning($"{username}, tu peux faire 10 voeux maximum d’un coup.");
            return true;
        }

        int primogems = GetInt(viewer, "primogems");
        int maxPulls = primogems / PullCost;

        if (maxPulls <= 0)
        {
            SendWarning($"{username}, tu n’as pas assez de primos. Il te faut 160 primos pour pull. Tu en as {primogems}.");
            return true;
        }

        if (pullCount > maxPulls)
        {
            SendWarning($"{username}, tu ne peux pas faire {pullCount} pulls. Avec tes {primogems} primos, tu peux faire {maxPulls} pull(s) maximum.");
            return true;
        }

        for (int i = 1; i <= pullCount; i++)
        {
            DoSinglePull(viewer, username, featured5, featured4, selectedCharacter, characterFile, passiveFile, c6CharactersPath, i, pullCount);
            SaveJObject(viewersPath, viewers);
        }

        return true;
    }

    private int GetPullCount()
    {
        string input = GetCommandInput();

        if (string.IsNullOrEmpty(input))
            return 1;

        int count;

        if (!int.TryParse(input, out count))
            return -1;

        return count;
    }

    private string GetCommandInput()
    {
        if (args.ContainsKey("rawInput") && args["rawInput"] != null)
            return args["rawInput"].ToString().Trim();

        if (args.ContainsKey("input0") && args["input0"] != null)
            return args["input0"].ToString().Trim();

        return "";
    }

    private void DoSinglePull(JObject viewer, string username, List<CharacterData> featured5, List<CharacterData> featured4, CharacterData selectedCharacter, CharacterFile characterFile, ElementPassiveFile passiveFile, string c6CharactersPath, int pullIndex, int totalPulls)
    {
        PassiveContext passiveContext = BuildPassiveContext(viewer, characterFile, passiveFile);
        List<string> triggeredPassives = new List<string>();

        AddInt(viewer, "primogems", -PullCost);
        AddStat(viewer, "totalPrimosSpent", PullCost);
        AddStat(viewer, "totalPulls", 1);

        JObject pity = viewer["pity"] as JObject;
        AddInt(pity, "pity5", 1);
        AddInt(pity, "pity4", 1);

        int currentPity5 = GetInt(pity, "pity5");
        int currentPity4 = GetInt(pity, "pity4");

        double fiveStarChance = GetFiveStarChance(currentPity5);
        fiveStarChance += passiveContext.fiveStarChanceBonus;

        bool gotFiveStar = Roll(fiveStarChance);
        bool gotFourStar = Roll(GetFourStarChance(currentPity4));

        string prefix = totalPulls > 1 ? $"[{pullIndex}/{totalPulls}] " : "";
        string specialPrefix = "";

        if (currentPity5 == 74)
            specialPrefix += "⚠️ Tu entres en soft pity... ";

        if (gotFiveStar && currentPity5 >= 2 && currentPity5 <= 35)
            specialPrefix += "🔥 WOW EARLY !! ";

        JObject stats = viewer["stats"] as JObject;

        if (gotFiveStar && GetBool(stats, "lastPullWasFiveStar"))
            specialPrefix += "💥 INCROYABLE BACK-TO-BACK !!! ";

        ApplyTriggeredPassives(viewer, passiveContext, triggeredPassives);

        if (gotFiveStar)
        {
            HandleFiveStarPull(viewer, username, featured5, selectedCharacter, prefix, specialPrefix, triggeredPassives, c6CharactersPath);
            stats["lastPullWasFiveStar"] = true;
        }
        else if (gotFourStar)
        {
            HandleFourStarPull(viewer, username, featured4, prefix, specialPrefix, passiveContext, triggeredPassives, c6CharactersPath);
            stats["lastPullWasFiveStar"] = false;
        }
        else
        {
            HandleSecondaryReward(viewer, username, prefix, specialPrefix, passiveContext, triggeredPassives);
            stats["lastPullWasFiveStar"] = false;
        }

        UpdateDailyMissionProgress(viewer, username, "pulls", 1);
    }

    private void HandleFiveStarPull(JObject viewer, string username, List<CharacterData> featured5, CharacterData selectedCharacter, string prefix, string specialPrefix, List<string> triggeredPassives, string c6CharactersPath)
    {
        JObject pity = viewer["pity"] as JObject;
        JObject guarantee = viewer["guarantee"] as JObject;
        JObject stats = viewer["stats"] as JObject;

        CharacterData obtainedCharacter = selectedCharacter;
        bool wonFiftyFifty = true;
        bool usedGuarantee = false;
        bool usedLostStreakGuarantee = false;
        int obtainedAtPity = GetInt(pity, "pity5");

        if (GetBool(guarantee, "guaranteedFeatured5"))
        {
            obtainedCharacter = selectedCharacter;
            guarantee["guaranteedFeatured5"] = false;
            usedGuarantee = true;
        }
        else if (GetInt(stats, "fiftyFiftyLostStreak") >= 3)
        {
            obtainedCharacter = selectedCharacter;
            usedLostStreakGuarantee = true;
            AddStat(viewer, "fiftyFiftyWon", 1);
            stats["fiftyFiftyLostStreak"] = 0;
        }
        else
        {
            bool win = random.Next(1, 101) <= 50;

            if (win)
            {
                obtainedCharacter = selectedCharacter;
                AddStat(viewer, "fiftyFiftyWon", 1);
                stats["fiftyFiftyLostStreak"] = 0;
            }
            else
            {
                obtainedCharacter = GetOtherFeaturedFiveStar(featured5, selectedCharacter.id);
                wonFiftyFifty = false;
                guarantee["guaranteedFeatured5"] = true;
                AddStat(viewer, "fiftyFiftyLost", 1);
                AddStat(viewer, "fiftyFiftyLostStreak", 1);
            }
        }

        pity["pity5"] = 0;
        AddStat(viewer, "totalFiveStars", 1);

        string boxResult = AddCharacterToBox(viewer, username, obtainedCharacter, c6CharactersPath);

        string fiftyText;

        if (usedGuarantee)
            fiftyText = "garantie utilisée";
        else if (usedLostStreakGuarantee)
            fiftyText = "garantie spéciale après 3 pertes de 50/50 d'affilée";
        else if (wonFiftyFifty)
            fiftyText = "50/50 gagné";
        else
            fiftyText = $"50/50 perdu, prochain 5★ garantie (stacks : {GetInt(stats, "fiftyFiftyLostStreak")}/3)";

        if (usedLostStreakGuarantee)
            specialPrefix += "✨ CAPTURE DE BRILLANCE ! ";

        string message = $"{specialPrefix}{prefix}{username} obtient ⭐⭐⭐⭐⭐ {GetElementEmoji(obtainedCharacter.element)} {obtainedCharacter.nom} ! {fiftyText}. {boxResult} | Pity {obtainedAtPity}";

        string passiveSuffix = FormatTriggeredPassiveSuffix(triggeredPassives);

        if (!string.IsNullOrEmpty(passiveSuffix))
            message += " | " + passiveSuffix;

        if (specialPrefix != "")
            CPH.SendMessage(message);
        else
            SendCharacter(message);
    }

    private void HandleFourStarPull(JObject viewer, string username, List<CharacterData> featured4, string prefix, string specialPrefix, PassiveContext passiveContext, List<string> triggeredPassives, string c6CharactersPath)
    {
        if (featured4.Count == 0)
        {
            HandleSecondaryReward(viewer, username, prefix, specialPrefix, passiveContext, triggeredPassives);
            return;
        }

        CharacterData obtainedCharacter = featured4[random.Next(featured4.Count)];

        JObject pity = viewer["pity"] as JObject;
        pity["pity4"] = 0;

        AddStat(viewer, "totalFourStars", 1);

        string boxResult = AddCharacterToBox(viewer, username, obtainedCharacter, c6CharactersPath);

        string message = $"{specialPrefix}{prefix}{username} obtient ⭐⭐⭐⭐ {GetElementEmoji(obtainedCharacter.element)} {obtainedCharacter.nom} ! {boxResult}";

        string passiveSuffix = FormatTriggeredPassiveSuffix(triggeredPassives);

        if (!string.IsNullOrEmpty(passiveSuffix))
            message += " | " + passiveSuffix;

        if (specialPrefix != "")
            CPH.SendMessage(message);
        else
            SendCharacter(message);
    }

    private void HandleSecondaryReward(JObject viewer, string username, string prefix, string specialPrefix, PassiveContext passiveContext, List<string> triggeredPassives)
    {
        int roll = random.Next(1, 101);

        if (roll <= 50)
        {
            int morasGained = random.Next(5000, 15001);
            morasGained = (int)Math.Round(morasGained * passiveContext.moraMultiplier);

            AddInt(viewer, "moras", morasGained);
            AddStat(viewer, "totalMorasEarned", morasGained);

            string message = $"{specialPrefix}{prefix}{username} obtient +{morasGained} 💰 moras ({GetInt(viewer, "moras")}).";

            string passiveSuffix = FormatTriggeredPassiveSuffix(triggeredPassives);

            if (!string.IsNullOrEmpty(passiveSuffix))
                message += " | " + passiveSuffix;

            SendPullMessage(message);
        }
        else
        {
            string element = GetRandomElement();
            int particlesGained = random.Next(20, 81);
            particlesGained = (int)Math.Round(particlesGained * passiveContext.particleMultiplier);

            AddParticles(viewer, element, particlesGained);
            AddMainElementParticlesEarnedIfNeeded(viewer, element, particlesGained);

            int total = GetParticleTotal(viewer, element);

            string message = $"{specialPrefix}{prefix}{username} obtient +{particlesGained} particules {GetElementDisplayName(element)} ({total}).";

            string passiveSuffix = FormatTriggeredPassiveSuffix(triggeredPassives);

            if (!string.IsNullOrEmpty(passiveSuffix))
                message += " | " + passiveSuffix;

            SendPullMessage(message);
        }
    }

    private PassiveContext BuildPassiveContext(JObject viewer, CharacterFile characterFile, ElementPassiveFile passiveFile)
    {
        PassiveContext context = new PassiveContext();

        context.particleMultiplier = 1;
        context.moraMultiplier = 1;
        context.fiveStarChanceBonus = 0;

        if (viewer == null || passiveFile == null || passiveFile.elements == null)
            return context;

        Dictionary<string, int> elementCounts = CountTeamElements(viewer, characterFile);

        foreach (KeyValuePair<string, int> entry in elementCounts)
        {
            string element = entry.Key;
            int count = entry.Value;

            if (!passiveFile.elements.ContainsKey(element))
                continue;

            ElementPassiveData passiveData = passiveFile.elements[element];

            int maxStacks = passiveData.maxStacks > 0 ? passiveData.maxStacks : 2;
            int stack = count;

            if (stack > maxStacks)
                stack = maxStacks;

            string stackKey = stack.ToString();

            if (passiveData.effects == null || !passiveData.effects.ContainsKey(stackKey))
                continue;

            PassiveEffect effect = passiveData.effects[stackKey];

            ApplyPassiveToContext(context, element, effect);
        }

        return context;
    }

    private void ApplyPassiveToContext(PassiveContext context, string element, PassiveEffect effect)
    {
        if (effect == null || string.IsNullOrEmpty(effect.passiveType))
            return;

        switch (effect.passiveType)
        {
            case "particleMultiplier":
                context.particleMultiplier = effect.value;
                break;

            case "moraMultiplier":
                context.moraMultiplier = effect.value;
                break;

            case "fiveStarChanceBonus":
                context.fiveStarChanceBonus = effect.value;
                break;

            case "randomXpBonus":
                context.randomXpBonusChance = effect.chanceDenominator;
                context.randomXpBonusValue = effect.value;
                context.randomXpBonusMessage = effect.triggerMessage;
                break;

            case "randomPity5Bonus":
                context.randomPity5BonusChance = effect.chanceDenominator;
                context.randomPity5BonusValue = effect.value;
                context.randomPity5BonusMessage = effect.triggerMessage;
                break;

            case "randomPrimoRefund":
                context.randomPrimoRefundChance = effect.chanceDenominator;
                context.randomPrimoRefundValue = effect.value;
                context.randomPrimoRefundMessage = effect.triggerMessage;
                break;

            case "randomResourceBundle":
                context.randomResourceBundleChance = effect.chanceDenominator;
                context.randomResourceBundleRewards = effect.rewards;
                context.randomResourceBundleMessage = effect.triggerMessage;
                break;
        }
    }

    private Dictionary<string, int> CountTeamElements(JObject viewer, CharacterFile characterFile)
    {
        Dictionary<string, int> counts = new Dictionary<string, int>();

        JArray team = viewer["team"] as JArray;

        if (team == null || characterFile == null || characterFile.characters == null)
            return counts;

        for (int i = 0; i < team.Count; i++)
        {
            int characterId = 0;

            if (!int.TryParse(team[i].ToString(), out characterId))
                continue;

            CharacterData character = FindCharacterById(characterFile, characterId);

            if (character == null || string.IsNullOrEmpty(character.element))
                continue;

            string element = NormalizeElement(character.element);

            if (!counts.ContainsKey(element))
                counts[element] = 0;

            counts[element]++;
        }

        return counts;
    }

    private void ApplyTriggeredPassives(JObject viewer, PassiveContext context, List<string> triggeredPassives)
    {
        if (context.randomXpBonusChance > 0 && RollOneIn(context.randomXpBonusChance))
        {
            AddInt(viewer, "xp", (int)Math.Round(context.randomXpBonusValue));
            triggeredPassives.Add(context.randomXpBonusMessage);
        }

        if (context.randomPity5BonusChance > 0 && RollOneIn(context.randomPity5BonusChance))
        {
            JObject pity = viewer["pity"] as JObject;

            int newPity = GetInt(pity, "pity5") + (int)Math.Round(context.randomPity5BonusValue);

            if (newPity > 90)
                newPity = 90;

            pity["pity5"] = newPity;

            triggeredPassives.Add(context.randomPity5BonusMessage);
        }

        if (context.randomPrimoRefundChance > 0 && RollOneIn(context.randomPrimoRefundChance))
        {
            int primoRefund = (int)Math.Round(context.randomPrimoRefundValue);
            AddInt(viewer, "primogems", primoRefund);
            AddStat(viewer, "totalPrimosEarned", primoRefund);
            triggeredPassives.Add(context.randomPrimoRefundMessage);
        }

        if (context.randomResourceBundleChance > 0 && RollOneIn(context.randomResourceBundleChance))
        {
            PassiveRewards rewards = context.randomResourceBundleRewards;

            if (rewards != null)
            {
                AddInt(viewer, "primogems", rewards.primogems);
                AddInt(viewer, "moras", rewards.moras);

                AddStat(viewer, "totalPrimosEarned", rewards.primogems);
                AddStat(viewer, "totalMorasEarned", rewards.moras);

                AddParticles(viewer, "pyro", rewards.particlesEachElement);
                AddParticles(viewer, "hydro", rewards.particlesEachElement);
                AddParticles(viewer, "cryo", rewards.particlesEachElement);
                AddParticles(viewer, "electro", rewards.particlesEachElement);
                AddParticles(viewer, "anemo", rewards.particlesEachElement);
                AddParticles(viewer, "geo", rewards.particlesEachElement);
                AddParticles(viewer, "dendro", rewards.particlesEachElement);
                AddMainElementParticlesEarnedIfNeeded(viewer, "pyro", rewards.particlesEachElement);
                AddMainElementParticlesEarnedIfNeeded(viewer, "hydro", rewards.particlesEachElement);
                AddMainElementParticlesEarnedIfNeeded(viewer, "cryo", rewards.particlesEachElement);
                AddMainElementParticlesEarnedIfNeeded(viewer, "electro", rewards.particlesEachElement);
                AddMainElementParticlesEarnedIfNeeded(viewer, "anemo", rewards.particlesEachElement);
                AddMainElementParticlesEarnedIfNeeded(viewer, "geo", rewards.particlesEachElement);
                AddMainElementParticlesEarnedIfNeeded(viewer, "dendro", rewards.particlesEachElement);
            }

            triggeredPassives.Add(context.randomResourceBundleMessage);
        }
    }

    private void UpdateDailyMissionProgress(JObject viewer, string username, string missionType, int amount)
    {
        JObject missions = viewer["missions"] as JObject;

        if (missions == null)
            return;

        JObject mission = missions["daily"] as JObject;

        if (mission == null)
            return;

        if (string.IsNullOrEmpty(GetString(mission, "missionId")))
            return;

        if (GetString(mission, "startedAt") != DateTime.Now.ToString("yyyy-MM-dd"))
            return;

        if (GetBool(mission, "completed") || GetBool(mission, "rewardClaimed"))
            return;

        if (GetString(mission, "type") != missionType)
            return;

        AddInt(mission, "progress", amount);

        if (GetInt(mission, "progress") >= GetInt(mission, "target"))
        {
            mission["progress"] = GetInt(mission, "target");
            CompleteDailyMission(viewer, username, mission);
        }
    }

    private void CompleteDailyMission(JObject viewer, string username, JObject mission)
    {
        int rewardPrimos = 800;

        AddInt(viewer, "primogems", rewardPrimos);
        AddStat(viewer, "totalPrimosEarned", rewardPrimos);

        mission["completed"] = true;
        mission["rewardClaimed"] = true;

        CPH.SendMessage($"🎯 Mission terminée : {GetString(mission, "description")} | +💠{rewardPrimos} primos ({GetInt(viewer, "primogems")})");
    }

    private string AddCharacterToBox(JObject viewer, string username, CharacterData character, string c6CharactersPath)
    {
        JObject box = viewer["box"] as JObject;

        if (box == null)
        {
            box = new JObject();
            viewer["box"] = box;
        }

        string id = character.id.ToString();

        JObject owned = box[id] as JObject;

        if (owned == null)
        {
            owned = new JObject();
            owned["characterId"] = character.id;
            owned["constellation"] = 0;
            owned["copies"] = 1;
            owned["firstObtainedAt"] = DateTime.Now.ToString("yyyy-MM-dd HH:mm:ss");

            box[id] = owned;

            return "Nouveau personnage : C0.";
        }

        AddInt(owned, "copies", 1);

        int constellation = GetInt(owned, "constellation");

        if (constellation < 6)
        {
            constellation++;
            owned["constellation"] = constellation;

            if (constellation == 6 && character.rarete == 5)
                EnsureC6ContestCharacterExists(username, character, c6CharactersPath);

            return $"Doublon : passe C{constellation}.";
        }

        int refund = character.rarete == 5 ? 80 : 40;

        AddInt(viewer, "primogems", refund);
        AddStat(viewer, "totalPrimosEarned", refund);

        if (character.rarete == 5)
        {
            string c6Progress = UpgradeRandomC6ContestStat(viewer, username, character, c6CharactersPath);

            if (!string.IsNullOrEmpty(c6Progress))
                return $"Déjà C6 : +{refund} primos ({GetInt(viewer, "primogems")}) | {c6Progress}";
        }

        return $"Déjà C6 : remboursement +{refund} primos ({GetInt(viewer, "primogems")}).";
    }


    private string UpgradeRandomC6ContestStat(JObject viewer, string username, CharacterData character, string c6CharactersPath)
    {
        if (character == null || character.rarete != 5)
            return "";

        JObject c6Characters = LoadOrCreateJObject(c6CharactersPath);
        JObject c6Data = EnsureC6ContestCharacterData(c6Characters, username, character);

        JObject stats = c6Data["stats"] as JObject;

        EnsureInt(stats, "strength", 1);
        EnsureInt(stats, "intelligence", 1);
        EnsureInt(stats, "beauty", 1);
        EnsureInt(stats, "charisma", 1);
        EnsureInt(stats, "popularity", 1);

        List<string> availableStats = new List<string>();
        availableStats.Add("strength");
        availableStats.Add("intelligence");
        availableStats.Add("beauty");
        availableStats.Add("charisma");
        availableStats.Add("popularity");

        for (int i = availableStats.Count - 1; i >= 0; i--)
        {
            if (GetInt(stats, availableStats[i]) >= 20)
                availableStats.RemoveAt(i);
        }

        if (availableStats.Count == 0)
        {
            int maxStatsMorasReward = 100000;

            AddInt(viewer, "moras", maxStatsMorasReward);
            AddStat(viewer, "totalMorasEarned", maxStatsMorasReward);

            SaveJObjectNoBackup(c6CharactersPath, c6Characters);

            return "Stats concours déjà max : +100000 moras.";
        }

        string selectedStat = availableStats[random.Next(availableStats.Count)];
        AddInt(stats, selectedStat, 1);

        int newValue = GetInt(stats, selectedStat);

        SaveJObjectNoBackup(c6CharactersPath, c6Characters);

        return "🌟 " + character.nom + " progresse : +1 " + GetContestStatDisplayName(selectedStat) + " (" + newValue + "/20).";
    }

    private JObject EnsureC6ContestCharacterData(JObject c6Characters, string username, CharacterData character)
    {
        JObject userData = c6Characters[username] as JObject;

        if (userData == null)
        {
            userData = new JObject();

            userData["characters"] = new JObject();

        }

        JObject characters = userData["characters"] as JObject;

        if (characters == null)
        {
            characters = new JObject();
            userData["characters"] = characters;
        }

        string idKey = character.id.ToString();
        JObject c6Data = characters[idKey] as JObject;

        if (c6Data == null)
        {
            c6Data = new JObject();
            c6Data["owner"] = username;
            c6Data["characterId"] = character.id;
            c6Data["name"] = character.nom;
            c6Data["rarity"] = character.rarete;

            c6Data["weapon"] = character.arme;
            c6Data["region"] = character.region;
            c6Data["class"] = character.classe;
            c6Data["createdAt"] = DateTime.Now.ToString("yyyy-MM-dd HH:mm:ss");

            JObject stats = new JObject();
            stats["strength"] = 1;
            stats["intelligence"] = 1;
            stats["beauty"] = 1;
            stats["charisma"] = 1;
            stats["popularity"] = 1;
            c6Data["stats"] = stats;

            JObject contestStats = new JObject();
            contestStats["totalContests"] = 0;
            contestStats["totalWins"] = 0;
            contestStats["strengthWins"] = 0;
            contestStats["intelligenceWins"] = 0;
            contestStats["beautyWins"] = 0;
            contestStats["charismaWins"] = 0;
            contestStats["popularityWins"] = 0;
            c6Data["contestStats"] = contestStats;

            JObject titles = new JObject();
            titles["strength"] = "";
            titles["intelligence"] = "";
            titles["beauty"] = "";
            titles["charisma"] = "";
            titles["popularity"] = "";
            c6Data["titles"] = titles;

            characters[idKey] = c6Data;
        }

        return c6Data;
    }

    private JObject LoadOrCreateJObject(string path)
    {
        if (!File.Exists(path))
            return new JObject();

        try
        {
            string json = File.ReadAllText(path);

            if (string.IsNullOrWhiteSpace(json))
                return new JObject();

            JObject data = JObject.Parse(json);

            if (data == null)
                return new JObject();

            return data;
        }
        catch
        {
            return new JObject();
        }
    }

    private void SaveJObjectNoBackup(string path, JObject data)
    {
        string folder = Path.GetDirectoryName(path);

        if (!Directory.Exists(folder))
            Directory.CreateDirectory(folder);

        string tempPath = path + ".tmp";

        File.WriteAllText(tempPath, data.ToString(Formatting.Indented));

        if (File.Exists(path))
            File.Delete(path);

        File.Move(tempPath, path);
    }

    private List<CharacterData> GetFeaturedCharacters(CharacterFile characterFile, int rarity)
    {
        List<CharacterData> result = new List<CharacterData>();

        if (characterFile == null || characterFile.characters == null)
            return result;

        for (int i = 0; i < characterFile.characters.Count; i++)
        {
            CharacterData character = characterFile.characters[i];

            if (character == null)
                continue;

            if (character.rarete == rarity && character.bannerFeatured)
                result.Add(character);
        }

        return result;
    }

    private CharacterData FindCharacterInListById(List<CharacterData> characters, int id)

    }

        }

    private CharacterData GetOtherFeaturedFiveStar(List<CharacterData> featured5, int selectedId)
    {
        List<CharacterData> others = new List<CharacterData>();

        for (int i = 0; i < featured5.Count; i++)
        {
            CharacterData character = featured5[i];

            if (character != null && character.id != selectedId)
                others.Add(character);
        }

        if (others.Count == 0)
            return featured5[0];

        return others[random.Next(others.Count)];
    }

    private double GetFiveStarChance(int pity)
    {
        if (pity >= 90)
            return 100.0;

        if (pity <= 73)
            return 0.6;

        return Math.Min(100.0, 0.6 + ((pity - 73) * 6.0));
    }

    private double GetFourStarChance(int pity)
    {
        if (pity >= 10)
            return 100.0;

        if (pity <= 8)
            return 1.5;

        return Math.Min(100.0, 1.5 + ((pity - 8) * 18.0));
    }

    private bool Roll(double chancePercent)
    {
        double roll = random.NextDouble() * 100.0;
        return roll < chancePercent;
    }

    private bool RollOneIn(int denominator)
    {
        if (denominator <= 1)
            return true;

        return random.Next(1, denominator + 1) == 1;
    }

    private void EnsureObject(JObject parent, string key)
    {
        if (parent[key] == null || parent[key].Type == JTokenType.Null || parent[key].Type != JTokenType.Object)

    }

    {
        if (parent[key] == null || parent[key].Type == JTokenType.Null || parent[key].Type != JTokenType.Array)
            parent[key] = new JArray();
    }

    private ElementPassiveFile LoadPassives(string path)
    {
        if (!File.Exists(path))
            return new ElementPassiveFile();

        try
        {
            string json = File.ReadAllText(path);
            ElementPassiveFile passiveFile = JsonConvert.DeserializeObject<ElementPassiveFile>(json);

            if (passiveFile == null)
                passiveFile = new ElementPassiveFile();

        }
    }

    private int GetInt(JObject obj, string key)
    {
        if (obj == null || obj[key] == null || obj[key].Type == JTokenType.Null)
            return 0;

        int value;

        if (int.TryParse(obj[key].ToString(), out value))
            return value;

        return 0;
    }

    private bool GetBool(JObject obj, string key)
    {
        if (obj == null || obj[key] == null || obj[key].Type == JTokenType.Null)
            return false;

        bool value;

        if (bool.TryParse(obj[key].ToString(), out value))
            return value;

        return false;
    }

    private string GetString(JObject obj, string key)
    {
        if (obj == null || obj[key] == null || obj[key].Type == JTokenType.Null)
            return "";

        return obj[key].ToString();
    }

    private void AddInt(JObject obj, string key, int amount)
    {
        int current = GetInt(obj, key);
        obj[key] = current + amount;
    }

    private void AddStat(JObject viewer, string statName, int amount)
    {
        JObject stats = viewer["stats"] as JObject;

        if (stats == null)
        {
            stats = new JObject();
            viewer["stats"] = stats;
        }

        AddInt(stats, statName, amount);
    }

    private string NormalizeUsername(string username)
    {
        if (string.IsNullOrEmpty(username))
            return "";

        username = username.Trim();

        if (username.StartsWith("@"))
            username = username.Substring(1);

        return username.ToLower();
    }

    {
