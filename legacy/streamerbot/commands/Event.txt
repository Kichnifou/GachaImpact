
/*

  - !event

- Gère la boutique event : primos, moras, collection.
- Gère le stockage permanent du coffre dans viewers_data.json.

    private string eventDataPath = @"C:\Users\axeld\Documents\Kichnifou\Twitch\Streambot\Data\monthly_events_data.json";

    private Random rng = new Random();

    public bool Execute()
    {
        if (!args.ContainsKey("userName") || args["userName"] == null)
            return false;

        string username = NormalizeUsername(args["userName"].ToString());
        string input = GetCommandInput();
        DateTime now = DateTime.Now;
        string today = now.ToString("yyyy-MM-dd");

        EventConfig config = GetEventConfig(now.Month);

        if (config == null)
        {
            SendError("Configuration event introuvable pour ce mois.");
            return true;
        }

        if (string.IsNullOrEmpty(input))
        {
            SendSplit(config.helpMessage);
            return true;
        }

        string[] parts = input.Split(new char[] { ' ' }, StringSplitOptions.RemoveEmptyEntries);

        if (parts.Length <= 0)
        {
            SendSplit(config.helpMessage);
            return true;
        }

        string action = NormalizeText(parts[0]);

        JObject viewers = LoadJObject(viewersPath);

        if (viewers == null)
        {
            SendError("Impossible de lire viewers_data.json.");
            return true;
        }

        JObject eventData = LoadJObject(eventDataPath);

        if (eventData == null)
        {
            SendError("Impossible de lire monthly_events_data.json.");
            return true;
        }

        EnsureEventDataForCurrentMonth(eventData, now);

        JObject viewer = viewers[username] as JObject;

        if (viewer == null)
        {
            SendWarning(username + ", tu dois d'abord exister dans le jeu avant de rejoindre l'event.");
            return true;
        }

        EnsureViewerDefaults(viewer, username);

        if (action == "go")
        {
            HandleGo(username, config, eventData);
            SaveJObject(eventDataPath, eventData);
            return true;
        }

        if (!IsJoined(eventData, username))
        {
            SendWarning(username + ", rejoins d'abord " + config.eventNameNoEmoji + " avec !event go.");
            return true;
        }

        if (action == "sac")
        {
            HandleSac(username, config, eventData);
            return true;
        }

        if (action == "boutique" || action == "shop")
        {
            HandleBoutique(config);
            return true;
        }

        if (action == "top")
        {
            HandleTop(config, eventData);
            return true;
        }

        if (action == "primos" || action == "primo" || action == "primogems")
        {
            HandleBuyPrimos(username, parts, config, viewer, viewers, eventData);
            return true;
        }

        if (action == "moras" || action == "mora")
        {
            HandleBuyMoras(username, parts, config, viewer, viewers, eventData);
            return true;
        }

        if (action == "collection")
        {
            HandleBuyCollection(username, config, viewer, viewers, eventData, now);
            return true;
        }

        if (action == "calendrier")
        {
            HandleCalendar(username, config, eventData, now);
            return true;
        }

        if (action == NormalizeText(config.gameACommand))
        {
            HandleGameA(username, config, viewer, viewers, eventData, now, today);
            return true;
        }

        if (action == NormalizeText(config.gameBCommand))
        {
            HandleGameB(username, input, parts, config, viewers, eventData, today);
            return true;
        }

        if (action == NormalizeText(config.gameCCommand))
        {
            HandleGameC(username, input, config, viewer, viewers, eventData, today);
            return true;
        }

        SendWarning("Commande event invalide. Aide : !event | Boutique : !event primos nombre | !event moras nombre | !event collection");
        return true;
    }

    private void HandleGo(string username, EventConfig config, JObject eventData)
    {
        JObject participants = GetObject(eventData, "participants");
        JObject participant = participants[username] as JObject;

        if (participant != null && GetBool(participant, "joined"))
        {
            SendWarning(username + ", tu participes déjà à " + config.eventNameNoEmoji + ". Aide : !event");
            return;
        }

        participant = EnsureParticipant(eventData, username);
        participant["joined"] = true;
        participant["joinedAt"] = DateTime.Now.ToString("yyyy-MM-dd HH:mm:ss");
        AddInt(participant, "currency", 1);

        string message = config.goMessage.Replace("{username}", username);

        string today = DateTime.Now.ToString("yyyy-MM-dd");
        JObject dayData = EnsureGameBDay(eventData, today);

        if (GetBool(dayData, "found"))
        {
            AddInt(participant, "points", 1);
            AddInt(participant, "currency", 1);

            message += " | 🎁 Le " + config.gameBName + " du jour a déjà été trouvé : +1 point et +1 " + config.currencyEmoji;
        }

        SendSplit(message);
    }

    private void HandleSac(string username, EventConfig config, JObject eventData)
    {
        JObject participant = EnsureParticipant(eventData, username);
        int points = GetInt(participant, "points");
        int currency = GetInt(participant, "currency");
        int next = GetNextMilestone(points);

        string msg;

        if (next <= 0)
            msg = "ℹ️ " + config.eventEmoji + " " + username + " | Points : " + points + " | " + config.currencyEmoji + " : " + currency + " | Palier maximal atteint | Boutique : !event boutique";
        else
            msg = "ℹ️ " + config.eventEmoji + " " + username + " | Points : " + points + " | " + config.currencyEmoji + " : " + currency + " | Prochain palier : " + points + "/" + next + " | Boutique : !event boutique";

        SendSplit(msg);
    }

    private void HandleBoutique(EventConfig config)
    {
        string msg = "🛒 Boutique " + config.eventNameNoEmoji + " : ";
        msg += "1" + config.currencyEmoji + " = 160 💠 Primos | ";
        msg += "1" + config.currencyEmoji + " = 20000 💰 Moras | ";
        msg += config.collectionDisplay + " = 80" + config.currencyEmoji + " | ";
        msg += "Acheter : !event primos N / !event moras N / !event collection";

        SendSplit(msg);
    }

    private void HandleTop(EventConfig config, JObject eventData)
    {
        JObject participants = GetObject(eventData, "participants");
        List<TopEntry> top = new List<TopEntry>();

        foreach (var prop in participants.Properties())
        {
            JObject p = prop.Value as JObject;
            if (p == null || !GetBool(p, "joined"))
                continue;

            TopEntry entry = new TopEntry();
            entry.username = prop.Name;
            entry.points = GetInt(p, "points");
            top.Add(entry);
        }

        for (int i = 0; i < top.Count - 1; i++)
        {
            for (int j = i + 1; j < top.Count; j++)
            {
                if (top[j].points > top[i].points)
                {
                    TopEntry tmp = top[i];
                    top[i] = top[j];
                    top[j] = tmp;
                }
            }
        }

        if (top.Count == 0)
        {
            SendWarning("Aucun participant pour l'event du mois.");
            return;
        }

        string msg = "🏆 " + config.eventNameNoEmoji + " | ";
        int max = top.Count;
        if (max > 10)
            max = 10;

        for (int i = 0; i < max; i++)
        {
            if (i > 0)
                msg += " | ";

            msg += (i + 1) + "." + top[i].username + " " + top[i].points + "pts";
        }

        SendSplit(msg);
    }

    private void HandleBuyPrimos(string username, string[] parts, EventConfig config, JObject viewer, JObject viewers, JObject eventData)
    {
        if (parts.Length < 2)
        {
            SendWarning("Achat invalide. Boutique : !event primos nombre | !event moras nombre | !event collection");
            return;
        }

        int count = ParsePositiveInt(parts[1]);

        if (count <= 0)
        {
            SendWarning("Achat invalide. Boutique : !event primos nombre | !event moras nombre | !event collection");
            return;
        }

        JObject participant = EnsureParticipant(eventData, username);
        int currency = GetInt(participant, "currency");

        if (currency < count)
        {
            SendWarning(username + ", pas assez de " + config.currencyEmoji + ". Solde : " + currency + ".");
            return;
        }

        int amount = count * 160;
        AddInt(participant, "currency", -count);
        AddInt(viewer, "primogems", amount);
        AddStat(viewer, "totalPrimosEarned", amount);

        SaveJObject(eventDataPath, eventData);
        SaveJObject(viewersPath, viewers);

        SendSplit("✅ " + username + " échange " + count + config.currencyEmoji + " contre +💠" + amount + " primos (" + GetInt(viewer, "primogems") + ").");
    }

    private void HandleBuyCollection(string username, EventConfig config, JObject viewer, JObject viewers, JObject eventData, DateTime now)
    {
        JObject participant = EnsureParticipant(eventData, username);
        int currency = GetInt(participant, "currency");

        if (currency < 80)
        {
            SendWarning(username + ", pas assez de " + config.currencyEmoji + ". Solde : " + currency + ".");
            return;
        }

        JObject purchases = GetObject(eventData, "collectionPurchases");
        string yearKey = now.Year.ToString();
        JObject yearPurchases = GetObject(purchases, yearKey);
        JArray userPurchases = GetArray(yearPurchases, username);

        if (ArrayContainsString(userPurchases, config.collectionItemId))
        {
            SendWarning(username + ", tu as déjà obtenu " + config.collectionDisplay + " cette année.");
            return;
        }

        AddInt(participant, "currency", -80);
        userPurchases.Add(config.collectionItemId);

        JObject coffre = GetObject(viewer, "coffre");
        AddInt(coffre, config.collectionItemId, 1);

        SaveJObject(eventDataPath, eventData);
        SaveJObject(viewersPath, viewers);

        SendSplit("✅ " + username + " ajoute " + config.collectionDisplay + " à son coffre ! Voir : !coffre");
    }

    private void HandleGameA(string username, EventConfig config, JObject viewer, JObject viewers, JObject eventData, DateTime now, string today)
    {
        JObject participant = EnsureParticipant(eventData, username);
        JObject daily = EnsureDaily(participant, today);

        if (GetBool(daily, "gameASuccess"))
        {
            SendWarning(config.gameAAlready.Replace("{username}", username));
            return;
        }

        JArray windows = GetDailyWindows(eventData, username, today);

        bool inWindow = false;
        string windowText = BuildWindowsText(windows);

        for (int i = 0; i < windows.Count; i++)
        {
            JObject w = windows[i] as JObject;
            if (w == null)
                continue;

            string start = GetString(w, "start");
            string end = GetString(w, "end");

            TimeSpan startTime;
            TimeSpan endTime;

            if (!TimeSpan.TryParse(start, out startTime))
                continue;

            if (!TimeSpan.TryParse(end, out endTime))
                continue;

            TimeSpan nowTime = new TimeSpan(now.Hour, now.Minute, 0);

            if (nowTime >= startTime && nowTime <= endTime)
                inWindow = true;
        }

        if (!inWindow)
        {
            string msg = config.gameAOutside.Replace("{username}", username);
            msg = msg.Replace("{window1}", GetWindowDisplay(windows, 0));
            msg = msg.Replace("{window2}", GetWindowDisplay(windows, 1));
            msg = msg.Replace("{window3}", GetWindowDisplay(windows, 2));
            SendSplit(msg);
            SaveJObject(eventDataPath, eventData);
            return;
        }

        int roll = rng.Next(1, 6);

        if (roll != 1)
        {
            SendSplit(config.gameAFail.Replace("{username}", username));
            SaveJObject(eventDataPath, eventData);
            return;
        }

        daily["gameASuccess"] = true;
        AwardEventReward(username, config, viewer, viewers, eventData, 1, 1, true);

        JObject refreshedParticipant = EnsureParticipant(eventData, username);
        string success = config.gameASuccess.Replace("{username}", username);
        success = success.Replace("{totalPoints}", GetInt(refreshedParticipant, "points").ToString());
        success = success.Replace("{totalCurrency}", GetInt(refreshedParticipant, "currency").ToString());
        SendSplit(success);

        SaveJObject(eventDataPath, eventData);
        SaveJObject(viewersPath, viewers);
    }

    private void HandleGameB(string username, string input, string[] parts, EventConfig config, JObject viewers, JObject eventData, string today)
    {
        JObject dayData = EnsureGameBDay(eventData, today);

        if (parts.Length < 2)
        {
            JArray testedCodesHelp = GetArray(dayData, "testedCodes");
            SendSplit(BuildGameBHelpMessage(config, testedCodesHelp));
            SaveJObject(eventDataPath, eventData);
            return;
        }

        string code = parts[1].Trim();

        if (code.Length != 5)
        {
            SendWarning("Format invalide. Utilise : !event " + config.gameBCommand + " 01101");
            return;
        }

        for (int i = 0; i < code.Length; i++)
        {
            if (code[i] != '0' && code[i] != '1')
            {
                SendWarning("Les coordonnées doivent contenir exactement 5 chiffres composés uniquement de 0 et de 1. Exemple : !event " + config.gameBCommand + " 01101");
                return;
            }
        }

        if (GetBool(dayData, "found"))
        {
            string msg = config.gameBAlreadyFound.Replace("{winningCode}", GetString(dayData, "winningCode"));
            SendSplit(msg);
            return;
        }

        JArray testedCodes = GetArray(dayData, "testedCodes");

        if (ArrayContainsString(testedCodes, code))
        {
            SendWarning(config.gameBAlreadyTried);
            return;
        }

        JObject participant = EnsureParticipant(eventData, username);
        JObject daily = EnsureDaily(participant, today);
        int attempts = GetInt(daily, "gameBAttempts");

        if (attempts >= 3)
        {
            SendWarning(username + ", tu as déjà utilisé tes 3 essais du jour. Reviens demain !");
            return;
        }

        attempts++;
        daily["gameBAttempts"] = attempts;
        testedCodes.Add(code);

        string winningCode = GetString(dayData, "winningCode");

        if (code == winningCode)
        {
            dayData["found"] = true;
            dayData["foundBy"] = username;

            JObject participants = GetObject(eventData, "participants");

            foreach (var prop in participants.Properties())
            {
                JObject p = prop.Value as JObject;
                if (p == null || !GetBool(p, "joined"))
                    continue;

                string targetName = prop.Name;
                JObject targetViewer = viewers[targetName] as JObject;

                if (targetViewer != null)
                {
                    EnsureViewerDefaults(targetViewer, targetName);
                    AwardEventReward(targetName, config, targetViewer, viewers, eventData, 1, 1, true);
                }
                else
                {
                    AddInt(p, "points", 1);
                    AddInt(p, "currency", 1);
                }
            }

            SaveJObject(eventDataPath, eventData);
            SaveJObject(viewersPath, viewers);

            string foundMsg = config.gameBFound.Replace("{username}", username).Replace("{code}", code);
            SendSplit(foundMsg);
            return;
        }

        SaveJObject(eventDataPath, eventData);

        int triesLeft = 3 - attempts;
        string fail = config.gameBFail.Replace("{username}", username).Replace("{code}", code).Replace("{triesLeft}", triesLeft.ToString());
        SendSplit(fail);
    }

    private void HandleGameC(string username, string input, EventConfig config, JObject viewer, JObject viewers, JObject eventData, string today)
    {
        string raw = input.Trim();
        int firstSpace = raw.IndexOf(' ');

        if (firstSpace < 0 || firstSpace >= raw.Length - 1)
        {
            SendSplit(config.gameCHelp);
            return;
        }

        string rest = raw.Substring(firstSpace + 1).Trim();
        int firstQuote = rest.IndexOf('"');
        int lastQuote = rest.LastIndexOf('"');

        if (firstQuote < 0 || lastQuote <= firstQuote)
        {
            SendWarning("Le message doit être entre guillemets. Exemple : !event " + config.gameCCommand + " pseudo \"" + config.gameCExample + "\"");
            return;
        }

        string targetPart = rest.Substring(0, firstQuote).Trim();
        string message = rest.Substring(firstQuote + 1, lastQuote - firstQuote - 1).Trim();
        string target = NormalizeUsername(targetPart);

        if (string.IsNullOrEmpty(target) || string.IsNullOrEmpty(message))
        {
            SendWarning("Format invalide. Utilise : !event " + config.gameCCommand + " pseudo \"message\"");
            return;
        }

        if (target == username)
        {
            SendWarning(config.gameCSelf.Replace("{username}", username));
            return;
        }

        JObject targetViewer = viewers[target] as JObject;

        if (targetViewer == null)
        {
            SendWarning("Destinataire introuvable. Le viewer doit déjà exister dans le jeu.");
            return;
        }

        JObject participant = EnsureParticipant(eventData, username);
        JObject daily = EnsureDaily(participant, today);

        if (GetBool(daily, "gameCSent"))
        {
            SendWarning(config.gameCAlready.Replace("{username}", username));
            return;
        }

        daily["gameCSent"] = true;

        JObject messages = GetObject(eventData, "messages");
        JArray targetMessages = GetArray(messages, target);
        JObject msg = new JObject();
        msg["sender"] = username;
        msg["text"] = message;
        msg["read"] = false;
        msg["type"] = config.gameCCommand;
        msg["createdAt"] = DateTime.Now.ToString("yyyy-MM-dd HH:mm:ss");
        targetMessages.Add(msg);

        AwardEventReward(username, config, viewer, viewers, eventData, 1, 1, true);

        JObject refreshedParticipant = EnsureParticipant(eventData, username);
        string success = config.gameCSuccess.Replace("{username}", username).Replace("{target}", target);
        success = success.Replace("{totalPoints}", GetInt(refreshedParticipant, "points").ToString());
        success = success.Replace("{totalCurrency}", GetInt(refreshedParticipant, "currency").ToString());

        SaveJObject(eventDataPath, eventData);
        SaveJObject(viewersPath, viewers);
        SendSplit(success);
    }

    private void HandleCalendar(string username, EventConfig config, JObject eventData, DateTime now)
    {
        if (now.Month != 12 || now.Day < 1 || now.Day > 25)
        {
            SendWarning("Le calendrier de Noël n'est disponible que du 1er au 25 décembre.");
            return;
        }

        JObject calendar = GetObject(eventData, "calendar");
        JObject userCal = GetObject(calendar, username);
        JArray openedDays = GetArray(userCal, "openedDays");

        if (ArrayContainsInt(openedDays, now.Day))
        {
            SendWarning(username + ", tu as déjà ouvert la case du jour. Reviens demain !");
            return;
        }

        int reward = now.Day == 25 ? 50 : rng.Next(1, 6);
        openedDays.Add(now.Day);

        JObject participant = EnsureParticipant(eventData, username);
        AddInt(participant, "currency", reward);

        SaveJObject(eventDataPath, eventData);

        string msg = GetCalendarMessage(now.Day, username, reward, config.currencyEmoji, GetInt(participant, "currency"));
        SendSplit(msg);
    }

    private void AwardEventReward(string username, EventConfig config, JObject viewer, JObject viewers, JObject eventData, int points, int currency, bool checkMilestones)
    {
        JObject participant = EnsureParticipant(eventData, username);
        AddInt(participant, "points", points);
        AddInt(participant, "currency", currency);

        if (checkMilestones)
            CheckMilestones(username, config, viewer, participant);
    }

    private void CheckMilestones(string username, EventConfig config, JObject viewer, JObject participant)
    {
        int points = GetInt(participant, "points");
        JArray claimed = GetArray(participant, "milestonesClaimed");
        int[] milestones = new int[] { 10, 20, 30, 40, 50, 60, 70, 80 };

        for (int i = 0; i < milestones.Length; i++)
        {
            int milestone = milestones[i];

            if (points < milestone)
                continue;

            if (ArrayContainsInt(claimed, milestone))
                continue;

            claimed.Add(milestone);

            if (milestone == 10)
            {
                string element = GetRandomElement();
                JObject particles = GetObject(viewer, "particles");
                AddInt(particles, element, 500);
                AddMainElementParticlesEarnedIfNeeded(viewer, element, 500);
                string display = GetElementDisplayName(element);
                SendSplit("✅ " + config.currencyEmoji + " " + username + " atteint le palier des 10 points de " + config.eventNameNoEmoji + " ! Gain : +500 particules " + display + " (" + GetInt(particles, element) + ")");
            }
            else if (milestone == 20)
            {
                AddInt(participant, "currency", 1);
                SendSplit("✅ " + config.currencyEmoji + " Déjà 20 points pour " + username + " ! Gain : +1 " + config.currencyEmoji + " (" + GetInt(participant, "currency") + ")");
            }
            else if (milestone == 30)
            {
                string element = GetString(viewer, "element");
                if (string.IsNullOrEmpty(element))
                    element = GetRandomElement();

                element = NormalizeText(element);
                JObject particles = GetObject(viewer, "particles");
                AddInt(particles, element, 500);
                AddMainElementParticlesEarnedIfNeeded(viewer, element, 500);
                string display = GetElementDisplayName(element);
                SendSplit("✅ " + config.eventEmoji + " " + username + " atteint le palier des 30 points de " + config.eventNameNoEmoji + " ! Gain : +500 particules " + display + " (" + GetInt(particles, element) + ")");
            }
            else if (milestone == 40)
            {
                AddInt(participant, "currency", 2);
                SendSplit("✅ " + config.currencyEmoji + " " + username + " atteint le palier des 40 points ! Gain : +2 " + config.currencyEmoji + " (" + GetInt(participant, "currency") + ")");
            }
            else if (milestone == 50)
            {
                AddInt(viewer, "moras", 50000);
                AddStat(viewer, "totalMorasEarned", 50000);
                SendSplit("✅ ⭐ Impressionnant ! " + username + " atteint le palier des 50 points ! Gain : +50000 moras (" + GetInt(viewer, "moras") + ")");
            }
            else if (milestone == 60)
            {
                AddInt(participant, "currency", 5);
                SendSplit("✅ " + config.currencyEmoji + " " + username + " atteint le palier des 60 points ! Gain : +5 " + config.currencyEmoji + " (" + GetInt(participant, "currency") + ")");
            }
            else if (milestone == 70)
            {
                AddInt(viewer, "primogems", 1600);
                AddStat(viewer, "totalPrimosEarned", 1600);
                SendSplit("✅ ⭐ Incroyable participation ! " + username + " atteint le palier des 70 points ! Gain : +1600 primos (" + GetInt(viewer, "primogems") + ")");
            }
            else if (milestone == 80)

    private JArray GetDailyWindows(JObject eventData, string username, string date)
    {
        JObject dailyWindows = GetObject(eventData, "dailyWindows");
        JObject userWindows = GetObject(dailyWindows, username);
        JArray windows = userWindows[date] as JArray;

        if (windows == null)
        {
            windows = GenerateWindows();
            userWindows[date] = windows;
        }

        return windows;
    }

    private JArray GenerateWindows()
    {
        List<WindowRange> list = new List<WindowRange>();

        AddWindowInRange(list, 7 * 60, 12 * 60);
        AddWindowInRange(list, 12 * 60, 18 * 60);
        AddWindowInRange(list, 18 * 60, 23 * 60);

        JArray result = new JArray();

        for (int i = 0; i < list.Count; i++)
        {
            JObject w = new JObject();
            w["start"] = MinuteToTime(list[i].startMinute);
            w["end"] = MinuteToTime(list[i].endMinute);
            result.Add(w);
        }

        return result;
    }

    private void AddWindowInRange(List<WindowRange> list, int rangeStartMinute, int rangeEndMinute)
    {
        int duration = 60;
        int latestStart = rangeEndMinute - duration;

        if (latestStart < rangeStartMinute)
            latestStart = rangeStartMinute;

        WindowRange wr = new WindowRange();
        wr.startMinute = rng.Next(rangeStartMinute, latestStart + 1);
        wr.endMinute = wr.startMinute + duration;

        list.Add(wr);
    }

    private string MinuteToTime(int minute)
    {
        int h = minute / 60;
        int m = minute % 60;
        return h.ToString("00") + ":" + m.ToString("00");
    }

    private string GetWindowDisplay(JArray windows, int index)
    {
        if (windows == null || index < 0 || index >= windows.Count)
            return "??h?? à ??:??";

        JObject w = windows[index] as JObject;

        if (w == null)
            return "??h?? à ??:??";

        string start = GetString(w, "start").Replace(":", "h");
        string end = GetString(w, "end").Replace(":", "h");

        return start + " à " + end;
    }

    private string BuildWindowsText(JArray windows)
    {
        return GetWindowDisplay(windows, 0) + " | " + GetWindowDisplay(windows, 1) + " | " + GetWindowDisplay(windows, 2);
    }

    private string FormatRemainingCodes(JArray testedCodes)
    {
        List<string> remaining = new List<string>();

        for (int i = 0; i < 32; i++)
        {
            string code = Convert.ToString(i, 2).PadLeft(5, '0');

            if (!ArrayContainsString(testedCodes, code))
                remaining.Add(code);
        }

        if (remaining.Count == 0)
            return "aucune";

        return string.Join(", ", remaining.ToArray());
    }

    private string BuildGameBHelpMessage(EventConfig config, JArray testedCodes)
    {
        return "ℹ️ Combinaisons restantes : " + FormatRemainingCodes(testedCodes)
            + " | Utilise : !event " + config.gameBCommand + " 01101"
            + " | Une seule combinaison est correcte parmi 32 possibilités"
            + " | 3 essais par joueur et par jour.";
    }

    private string GenerateBinaryCode()
    {
        string code = "";

        for (int i = 0; i < 5; i++)
            code += rng.Next(0, 2).ToString();

        return code;
    }

    private int GetNextMilestone(int points)
    {
        int[] milestones = new int[] { 10, 20, 30, 40, 50, 60, 70, 80 };

        for (int i = 0; i < milestones.Length; i++)
        {
            if (points < milestones[i])
                return milestones[i];
        }

        return 0;
    }

    private string GetRandomElement()
    {
        string[] elements = new string[] { "pyro", "hydro", "cryo", "electro", "anemo", "geo", "dendro" };
        return elements[rng.Next(0, elements.Length)];
    }

    private string GetElementDisplayName(string element)
    {
        element = NormalizeText(element);

        if (element == "pyro") return "🔥 Pyro";
        if (element == "hydro") return "💧 Hydro";
        if (element == "cryo") return "❄️ Cryo";
        if (element == "electro") return "⚡ Electro";
        if (element == "anemo") return "🌪️ Anemo";
        if (element == "geo") return "☄️ Geo";
        if (element == "dendro") return "🌿 Dendro";

        return element;
    }

    private void AddMainElementParticlesEarnedIfNeeded(JObject viewer, string element, int amount)
    {
        string mainElement = NormalizeText(GetString(viewer, "element"));
        string gainedElement = NormalizeText(element);

        if (string.IsNullOrEmpty(mainElement))
            return;

        if (mainElement != gainedElement)
            return;

        AddStat(viewer, "totalMainElementParticlesEarned", amount);
    }

    private string GetCommandInput()
    {

    private JObject LoadJObject(string path)
    {
        try
        {
            if (!File.Exists(path))
                return new JObject();

            string json = File.ReadAllText(path);

            if (string.IsNullOrWhiteSpace(json))
                return new JObject();

            JObject data = JObject.Parse(json);

            if (data == null)
                return new JObject();

            return data;
        }
        catch
        {
            return null;
        }
    }

    private void SaveJObject(string path, JObject data)
    {
        string folder = Path.GetDirectoryName(path);

        if (!Directory.Exists(folder))
            Directory.CreateDirectory(folder);

        File.WriteAllText(path, data.ToString(Formatting.Indented));
    }

    private JObject GetObject(JObject parent, string key)
    {
        if (parent[key] == null || parent[key].Type == JTokenType.Null || parent[key].Type != JTokenType.Object)
            parent[key] = new JObject();

        return parent[key] as JObject;
    }

    private JArray GetArray(JObject parent, string key)
    {
        if (parent[key] == null || parent[key].Type == JTokenType.Null || parent[key].Type != JTokenType.Array)
            parent[key] = new JArray();

        return parent[key] as JArray;
    }

    private int GetInt(JObject obj, string key)
    {
        if (obj == null || obj[key] == null || obj[key].Type == JTokenType.Null)
            return 0;

        int value;
        if (int.TryParse(obj[key].ToString(), out value))
            return value;

        return 0;
    }

    private string GetString(JObject obj, string key)
    {
        if (obj == null || obj[key] == null || obj[key].Type == JTokenType.Null)
            return "";

        return obj[key].ToString();
    }

    private bool GetBool(JObject obj, string key)
    {
        if (obj == null || obj[key] == null || obj[key].Type == JTokenType.Null)
            return false;

        bool value;
        if (bool.TryParse(obj[key].ToString(), out value))
            return value;

        return false;
    }

    private void AddInt(JObject obj, string key, int amount)
    {
        int current = GetInt(obj, key);
        obj[key] = current + amount;
    }

    private void AddStat(JObject viewer, string statName, int amount)
    {
        JObject stats = GetObject(viewer, "stats");
        AddInt(stats, statName, amount);
    }

    private void EnsureInt(JObject obj, string key, int defaultValue)
    {
        if (obj[key] == null || obj[key].Type == JTokenType.Null)
            obj[key] = defaultValue;
    }

    private void EnsureBool(JObject obj, string key, bool defaultValue)
    {
        if (obj[key] == null || obj[key].Type == JTokenType.Null)
            obj[key] = defaultValue;
    }

    private void EnsureString(JObject obj, string key, string defaultValue)
    {
        if (obj[key] == null || obj[key].Type == JTokenType.Null)
            obj[key] = defaultValue;
    }

    private void EnsureNullOrString(JObject obj, string key)
    {
        if (obj[key] == null)
            obj[key] = null;
    }

    private bool ArrayContainsString(JArray array, string value)
    {
        if (array == null)
            return false;

        for (int i = 0; i < array.Count; i++)
        {
            if (array[i] != null && array[i].ToString() == value)
                return true;
        }

        return false;
    }

    private bool ArrayContainsInt(JArray array, int value)
    {
        if (array == null)
            return false;

        for (int i = 0; i < array.Count; i++)
        {
            int current;
            if (array[i] != null && int.TryParse(array[i].ToString(), out current))
            {
                if (current == value)
                    return true;
            }
        }

        return false;
    }

    private int ParsePositiveInt(string value)
    {
        if (string.IsNullOrEmpty(value))
            return 0;

        int result;
        if (!int.TryParse(value.Trim(), out result))
            return 0;

        if (result <= 0)
            return 0;

        return result;
    }

    private string NormalizeUsername(string username)
    {
        if (string.IsNullOrEmpty(username))
            return "";

        username = username.Trim();

        if (username.StartsWith("@"))
            username = username.Substring(1);

        return username.ToLower();
    }

    private string NormalizeText(string text)
    {
        if (string.IsNullOrEmpty(text))
            return "";

        string value = text.ToLower().Trim();
        value = value.Replace("à", "a"); value = value.Replace("á", "a"); value = value.Replace("â", "a"); value = value.Replace("ä", "a"); value = value.Replace("ã", "a");
        value = value.Replace("ç", "c");
        value = value.Replace("è", "e"); value = value.Replace("é", "e"); value = value.Replace("ê", "e"); value = value.Replace("ë", "e");
        value = value.Replace("ì", "i"); value = value.Replace("í", "i"); value = value.Replace("î", "i"); value = value.Replace("ï", "i");
        value = value.Replace("ò", "o"); value = value.Replace("ó", "o"); value = value.Replace("ô", "o"); value = value.Replace("ö", "o"); value = value.Replace("õ", "o");
        value = value.Replace("ù", "u"); value = value.Replace("ú", "u"); value = value.Replace("û", "u"); value = value.Replace("ü", "u");
        value = value.Replace("ÿ", "y"); value = value.Replace("œ", "oe"); value = value.Replace("æ", "ae");
        return value;
    }

    private void SendWarning(string message)
    {
        CPH.SendMessage("⚠️ " + message);
    }

    private void SendError(string message)
    {
        CPH.SendMessage("❌ " + message);
    }

    private void SendSplit(string message)
    {
        if (string.IsNullOrEmpty(message))
            return;

        if (message.Length <= 450)
        {
            CPH.SendMessage(message);
            return;
        }

        string remaining = message;

        while (remaining.Length > 450)
        {
            int cut = remaining.LastIndexOf(" | ", 450);

            if (cut <= 0)
                cut = 450;

            string part = remaining.Substring(0, cut).Trim();
            CPH.SendMessage(part);
            remaining = remaining.Substring(cut).Trim();

            if (remaining.StartsWith("|"))
                remaining = remaining.Substring(1).Trim();
        }

        if (remaining.Length > 0)
            CPH.SendMessage(remaining);
    }
}
