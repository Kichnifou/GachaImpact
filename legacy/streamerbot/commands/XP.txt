    private bool TryGetMessageData(out string username, out string message)
    {
        username = "";
        message = "";

        if (!args.ContainsKey("userName") || args["userName"] == null)
            return false;

        username = NormalizeUsername(args["userName"].ToString());

        if (args.ContainsKey("message") && args["message"] != null)
            message = args["message"].ToString();

        return true;
    }

    private void EnsureDataFolderExists(string folderPath)
    {
        if (!Directory.Exists(folderPath))
            Directory.CreateDirectory(folderPath);
    }

    private JObject LoadViewersJObject(string path)
    {
        if (!File.Exists(path))
            return new JObject();

        try
        {
            string json = File.ReadAllText(path);

            if (string.IsNullOrWhiteSpace(json))
                return new JObject();

            JObject viewers = JObject.Parse(json);

            if (viewers == null)
                return new JObject();

            return viewers;
        }
        catch (Exception ex)
        {
            CPH.LogError("Erreur lecture viewers_data.json : " + ex.Message);
            return null;
        }
    }

    private void SaveJObjectSafe(string path, JObject data)
    {
        string folder = Path.GetDirectoryName(path);

        if (!Directory.Exists(folder))
            Directory.CreateDirectory(folder);

        File.WriteAllText(path, data.ToString(Formatting.Indented));
    }

        }

        return viewer;
    }

    private void CheckWeeklyBanner(string folderPath)
    {
        string charactersPath = Path.Combine(folderPath, "genshin_characters.json");

        if (!File.Exists(charactersPath))
            return;

        JObject characterFile = LoadJObject(charactersPath);

        if (characterFile == null)
            return;

        JArray characters = characterFile["characters"] as JArray;

        if (characters == null)
            return;

        string currentWeek = GetCurrentWeekId();
        string lastBannerUpdate = GetString(characterFile, "lastBannerUpdate");

        bool noBanner = string.IsNullOrEmpty(lastBannerUpdate);
        bool weekChanged = lastBannerUpdate != currentWeek;

        if (!noBanner && !weekChanged)
            return;

        JArray previousBannerFeaturedIds = GetCurrentFeaturedIds(characters);
        characterFile["previousBannerFeaturedIds"] = previousBannerFeaturedIds;

        for (int i = 0; i < characters.Count; i++)
        {
            JObject character = characters[i] as JObject;

            if (character != null)
                character["bannerFeatured"] = false;
        }

        List<JObject> fiveStars = new List<JObject>();
        List<JObject> fourStars = new List<JObject>();

        for (int i = 0; i < characters.Count; i++)
        {
            JObject character = characters[i] as JObject;

            if (character == null)
                continue;

            int rarity = GetInt(character, "rarete");

            if (rarity == 5)
                fiveStars.Add(character);
            else if (rarity == 4)
                fourStars.Add(character);
        }

        RemoveCharactersByIds(fiveStars, previousBannerFeaturedIds);
        RemoveCharactersByIds(fourStars, previousBannerFeaturedIds);

        PickRandomCharacters(fiveStars, 3);
        ApplyVoteBannerCharacter(folderPath,characterFile,characters,previousBannerFeaturedIds);
        PickRandomCharacters(fourStars, 6);

        characterFile["lastBannerUpdate"] = currentWeek;

        SaveJObjectSafe(charactersPath, characterFile);

        CPH.LogInfo("Nouvelle bannière hebdomadaire générée automatiquement.");
    }

    private string GetCurrentWeekId()
    {
        DateTime now = DateTime.Now;
        DateTime startOfWeek = now.Date.AddDays(-(int)now.DayOfWeek + (int)DayOfWeek.Monday);

        if (now.DayOfWeek == DayOfWeek.Sunday)
            startOfWeek = now.Date.AddDays(-6);

        return startOfWeek.ToString("yyyy-MM-dd");
    }

    private JArray GetCurrentFeaturedIds(JArray characters)
    {
        JArray ids = new JArray();

        if (characters == null)
            return ids;

        for (int i = 0; i < characters.Count; i++)
        {
            JObject character = characters[i] as JObject;

            if (character == null)
                continue;

            bool featured = GetBool(character, "bannerFeatured");

            if (featured)
                ids.Add(GetInt(character, "id"));
        }

        return ids;
    }

    private void RemoveCharactersByIds(List<JObject> characters, JArray excludedIds)
    {
        if (characters == null || excludedIds == null)
            return;

        for (int i = characters.Count - 1; i >= 0; i--)
        {
            JObject character = characters[i];

            if (character == null)
                continue;

            int id = GetInt(character, "id");

            for (int j = 0; j < excludedIds.Count; j++)
            {
                int excludedId = 0;

                if (int.TryParse(excludedIds[j].ToString(), out excludedId) && id == excludedId)
                {
                    characters.RemoveAt(i);
                    break;
                }
            }
        }
    }

    private void PickRandomCharacters(List<JObject> characters, int count)
    {
        int picked = 0;

        while (picked < count && characters.Count > 0)
        {
            int index = random.Next(characters.Count);
            characters[index]["bannerFeatured"] = true;
            characters.RemoveAt(index);
            picked++;
        }
    }

    private void TryCountGiveawayMessage(
    string giveawayPath,
    JObject viewer,
    string username,
    string message)
    {
        if (viewer == null)
            return;

        username = NormalizeUsername(username);

        // Kichnifou est le créateur + le bot parle sous ce pseudo
        if (username == "kichnifou")
            return;

        // Niveau 2 minimum
        if (GetInt(viewer, "level") < 2)
            return;

        // Un vrai message doit exister
        if (string.IsNullOrWhiteSpace(message))
            return;

        if (!File.Exists(giveawayPath))
            return;

        JObject giveaway = LoadOrCreateJObject(giveawayPath);

        if (giveaway == null)
            return;

        // Comptage uniquement entre !giveaway open et !giveaway close
        if (GetString(giveaway, "status") != "open")
            return;

        JObject messageCounts = giveaway["messageCounts"] as JObject;

        if (messageCounts == null)
        {
            messageCounts = new JObject();
            giveaway["messageCounts"] = messageCounts;
        }

        AddInt(messageCounts, username, 1);

        SaveJObjectNoBackup(giveawayPath, giveaway);
    }

    private void ExpireOldTradeRequests(JObject viewers, string username, DateTime now)
    {
        JObject viewer = viewers[username] as JObject;

        if (viewer == null)
            return;

        JArray tradeRequests = viewer["tradeRequests"] as JArray;

        if (tradeRequests == null || tradeRequests.Count == 0)
            return;

        string today = now.ToString("yyyy-MM-dd");
        List<JObject> expiredRequests = new List<JObject>();

        for (int i = 0; i < tradeRequests.Count; i++)
        {
            JObject request = tradeRequests[i] as JObject;

            if (request == null)
                continue;

            string createdAt = GetString(request, "createdAt");

            if (!string.IsNullOrEmpty(createdAt) && !createdAt.StartsWith(today))
                expiredRequests.Add(request);
        }

        if (expiredRequests.Count == 0)
            return;

        for (int i = 0; i < expiredRequests.Count; i++)
        {
            RemoveTradeRequestFromBothUsers(viewers, username, expiredRequests[i]);
        }

        CPH.LogInfo("Trades expirés nettoyés pour " + username + " : " + expiredRequests.Count);
    }

    private void RemoveTradeRequestFromBothUsers(JObject viewers, string username, JObject request)
    {
        string otherUser = GetString(request, "otherUser");

        RemoveMatchingTradeRequest(viewers, username, otherUser);

        if (!string.IsNullOrEmpty(otherUser))
            RemoveMatchingTradeRequest(viewers, otherUser, username);
    }

    private void RemoveMatchingTradeRequest(JObject viewers, string user, string otherUser)
    {
        if (string.IsNullOrEmpty(user))
            return;

        JObject viewer = viewers[user] as JObject;

        if (viewer == null)
            return;

        JArray requests = viewer["tradeRequests"] as JArray;

        if (requests == null)
            return;

        for (int i = requests.Count - 1; i >= 0; i--)
        {
            JObject request = requests[i] as JObject;

            if (request == null)
                continue;

            if (GetString(request, "otherUser") == otherUser)
                requests.RemoveAt(i);
        }
    }

    private void ExpireDailyMission(JObject viewer, DateTime now)
    {
        JObject missions = viewer["missions"] as JObject;

        if (missions == null)
            return;

        JObject daily = missions["daily"] as JObject;

        if (daily == null)
            return;

        string missionId = GetString(daily, "missionId");

        if (string.IsNullOrEmpty(missionId))
            return;

        string today = now.ToString("yyyy-MM-dd");

        if (GetString(daily, "startedAt") != today)
        {
            missions["daily"] = CreateEmptyDailyMissionJObject();
            CPH.LogInfo("Mission journalière expirée pour " + GetString(viewer, "username"));
        }
    }

    private void TryGiveDailyFirstMessageReward(JObject viewer, string username, string message, DateTime now)
    {
        if (IsCommand(message))
            return;

        if (IsSystemMessage(message))
            return;

        if (GetInt(viewer, "level") < 2)
            return;

        string element = NormalizeElement(GetString(viewer, "element"));

        if (string.IsNullOrEmpty(element))
            return;

        JObject dates = viewer["dates"] as JObject;
        string today = now.ToString("yyyy-MM-dd");

        if (GetString(dates, "lastDailyFirstMessageReward") == today)
            return;

        int primosReward = 160;
        int particlesReward = 160;
        int morasReward = 10000;

        AddInt(viewer, "primogems", primosReward);
        AddInt(viewer, "moras", morasReward);
        AddParticles(viewer, element, particlesReward);
        AddMainElementParticlesEarnedIfNeeded(viewer, element, particlesReward);

        AddStat(viewer, "totalPrimosEarned", primosReward);
        AddStat(viewer, "totalMorasEarned", morasReward);

    private void TryGiveFavorDailyReward(JObject viewer, string username, string message, DateTime now)

            return;

            return;

        daysRemaining--;

        if (daysRemaining < 0)
            daysRemaining = 0;

        favor["daysRemaining"] = daysRemaining;
        favor["lastClaimDate"] = today;

        CPH.SendMessage(GetRandomFavorMessage(username, primosReward, daysRemaining, GetInt(viewer, "primogems")));
    }

    private bool IsInXpCooldown(JObject viewer, DateTime now)
    {
        JObject dates = viewer["dates"] as JObject;
        DateTime lastMsgTime;

        if (DateTime.TryParse(GetString(dates, "lastMessageTime"), out lastMsgTime))
            return (now - lastMsgTime).TotalSeconds < XpCooldownSeconds;

        return false;
    }

    private void UpdateDailyMissionProgress(JObject viewer, string username, string missionType, int amount, DateTime now)
    {
        JObject missions = viewer["missions"] as JObject;

        if (missions == null)
            return;

        JObject daily = missions["daily"] as JObject;

        if (daily == null)
            return;

        if (string.IsNullOrEmpty(GetString(daily, "missionId")))
            return;

        if (GetString(daily, "startedAt") != now.ToString("yyyy-MM-dd"))
            return;

        if (GetBool(daily, "completed") || GetBool(daily, "rewardClaimed"))
            return;

        if (GetString(daily, "type") != missionType)
            return;

        AddInt(daily, "progress", amount);

        if (GetInt(daily, "progress") >= GetInt(daily, "target"))
        {
            daily["progress"] = GetInt(daily, "target");
            CompleteDailyMission(viewer, username, daily);
        }
    }

    private void CheckLevelUp(JObject viewer, string username)
    {
        int oldLevel = GetInt(viewer, "level");
        int newLevel = CalculateLevel(GetInt(viewer, "xp"));

        if (newLevel > oldLevel)
        {
            List<LevelRewards> rewardsList = new List<LevelRewards>();

            for (int level = oldLevel + 1; level <= newLevel; level++)
            {
                rewardsList.Add(GiveLevelRewards(viewer, level));
            }

            viewer["level"] = newLevel;

            LevelRewards lastRewards = rewardsList[rewardsList.Count - 1];

            string levelMessage = "🎉 " + username + " passe niveau " + newLevel + " ! ";
            levelMessage += "+💠" + lastRewards.primos + " primos (" + GetInt(viewer, "primogems") + ") | ";
            levelMessage += "+💰" + lastRewards.moras + " moras (" + GetInt(viewer, "moras") + ")";

            if (newLevel >= 5 && lastRewards.particlesMain > 0)
            {
                int mainParticleTotal = GetParticleTotal(viewer, lastRewards.particlesMainElement);
                levelMessage += " | +" + lastRewards.particlesMain + " particules " + GetElementDisplayName(lastRewards.particlesMainElement) + " (" + mainParticleTotal + ")";
            }

            if (newLevel >= 10 && lastRewards.particlesOther > 0)
            {
                int otherParticleTotal = GetParticleTotal(viewer, lastRewards.particlesOtherElement);
                levelMessage += " | +" + lastRewards.particlesOther + " particules " + GetElementDisplayName(lastRewards.particlesOtherElement) + " (" + otherParticleTotal + ")";
            }

            if (newLevel == 1 && string.IsNullOrEmpty(NormalizeElement(GetString(viewer, "element"))))
                levelMessage += " | Pour continuer à obtenir de l'expérience, tu dois choisir ton élément !";

            CPH.SendMessage(levelMessage);
            SendLevelTutorials(oldLevel, newLevel);
        }

        CheckMaxLevelOverflowRewards(viewer, username);
    }

    private LevelRewards GiveLevelRewards(JObject viewer, int level)
    {
        LevelRewards rewards = new LevelRewards();

        int primosGained = 800;
        int morasGained = 10000;    

        AddInt(viewer, "primogems", primosGained);
        AddInt(viewer, "moras", morasGained);

        AddStat(viewer, "totalPrimosEarned", primosGained);
        AddStat(viewer, "totalMorasEarned", morasGained);

        rewards.primos = primosGained;
        rewards.moras = morasGained;

        string mainElement = NormalizeElement(GetString(viewer, "element"));

        if (level >= 5 && !string.IsNullOrEmpty(mainElement))
        {
            int particlesGained = 80;

            AddParticles(viewer, mainElement, particlesGained);
            AddMainElementParticlesEarnedIfNeeded(viewer, mainElement, particlesGained);

            rewards.particlesMain = particlesGained;
            rewards.particlesMainElement = mainElement;
        }

        if (level >= 10)
        {
            string otherElement = GetRandomElementExcept(mainElement);
            int particlesGained = 40;

            AddParticles(viewer, otherElement, particlesGained);

            rewards.particlesOther = particlesGained;
            rewards.particlesOtherElement = otherElement;
        }

        return rewards;
    }

    private void AddMainElementParticlesEarnedIfNeeded(JObject viewer, string element, int amount)
    {
        string mainElement = NormalizeElement(GetString(viewer, "element"));
        string gainedElement = NormalizeElement(element);

        if (string.IsNullOrEmpty(mainElement))
            return;

        if (mainElement != gainedElement)
            return;

        AddStat(viewer, "totalMainElementParticlesEarned", amount);
    }

    private void CheckMaxLevelOverflowRewards(JObject viewer, string username)
    {
        if (GetInt(viewer, "level") < MaxLevel)
            return;

        JObject stats = viewer["stats"] as JObject;

        int currentSteps = CalculateMaxLevelOverflowSteps(GetInt(viewer, "xp"));
        int claimedSteps = GetInt(stats, "level100OverflowRewardsClaimed");

        if (currentSteps <= claimedSteps)
            return;

        int rewardsToGive = currentSteps - claimedSteps;
        LevelRewards totalRewards = new LevelRewards();

        for (int i = 0; i < rewardsToGive; i++)
        {
            LevelRewards rewards = GiveLevelRewards(viewer, MaxLevel);

            totalRewards.primos += rewards.primos;
            totalRewards.moras += rewards.moras;
            totalRewards.particlesMain += rewards.particlesMain;
            totalRewards.particlesMainElement = rewards.particlesMainElement;
            totalRewards.particlesOther += rewards.particlesOther;
            totalRewards.particlesOtherElement = rewards.particlesOtherElement;
        }

        stats["level100OverflowRewardsClaimed"] = currentSteps;

        string overflowMessage = "🎉 " + username + " gagne une récompense niveau max ! ";
        overflowMessage += "+💠" + totalRewards.primos + " primos (" + GetInt(viewer, "primogems") + ") | ";
        overflowMessage += "+💰" + totalRewards.moras + " moras (" + GetInt(viewer, "moras") + ")";

        if (totalRewards.particlesMain > 0)
        {
            int mainParticleTotal = GetParticleTotal(viewer, totalRewards.particlesMainElement);
            overflowMessage += " | +" + totalRewards.particlesMain + " particules " + GetElementDisplayName(totalRewards.particlesMainElement) + " (" + mainParticleTotal + ")";
        }

        if (totalRewards.particlesOther > 0)
        {
            int otherParticleTotal = GetParticleTotal(viewer, totalRewards.particlesOtherElement);
            overflowMessage += " | +" + totalRewards.particlesOther + " particules " + GetElementDisplayName(totalRewards.particlesOtherElement) + " (" + otherParticleTotal + ")";
        }

        if (rewardsToGive > 1)
            overflowMessage += " | x" + rewardsToGive;

        CPH.SendMessage(overflowMessage);
    }

    private int CalculateLevel(int xp)
    {
        int level = xp / XpPerLevel;

        if (level > MaxLevel)
            level = MaxLevel;

        return level;
    }

    private int CalculateMaxLevelOverflowSteps(int xp)
    {
        int maxLevelXp = MaxLevel * XpPerLevel;

        if (xp <= maxLevelXp)
            return 0;

        return (xp - maxLevelXp) / XpPerLevel;
    }

    private void AddParticles(JObject viewer, string element, int amount)
    {
        JObject particles = viewer["particles"] as JObject;

        if (particles == null)
        {
            particles = new JObject();
            viewer["particles"] = particles;
        }

        string normalized = NormalizeElement(element);

        if (string.IsNullOrEmpty(normalized))
            return;

        AddInt(particles, normalized, amount);
    }

    private int GetParticleTotal(JObject viewer, string element)
    {
        JObject particles = viewer["particles"] as JObject;

        if (particles == null)
            return 0;

        return GetInt(particles, NormalizeElement(element));
    }

    private string GetRandomElementExcept(string excludedElement)
    {
        string normalizedExcluded = NormalizeElement(excludedElement);

        List<string> elements = new List<string>();
        elements.Add("pyro");
        elements.Add("hydro");
        elements.Add("cryo");
        elements.Add("electro");
        elements.Add("anemo");
        elements.Add("geo");
        elements.Add("dendro");

        for (int i = elements.Count - 1; i >= 0; i--)
        {
            if (elements[i] == normalizedExcluded)
                elements.RemoveAt(i);
        }

        if (elements.Count == 0)
            return "pyro";

        return elements[random.Next(elements.Count)];
    }

    private string GetElementDisplayName(string element)
    {
        switch (NormalizeElement(element))
        {
            case "pyro": return "🔥 Pyro";
            case "hydro": return "💧 Hydro";
            case "cryo": return "❄️ Cryo";
            case "electro": return "⚡ Electro";
            case "anemo": return "🌪️ Anemo";
            case "geo": return "☄️ Geo";
            case "dendro": return "🌿 Dendro";
            default: return element;
        }
    }

    private string NormalizeElement(string element)
    {
        if (string.IsNullOrEmpty(element))
            return "";

        string normalized = element.ToLower().Trim();

        switch (normalized)
        {
            case "pyro": return "pyro";
            case "hydro": return "hydro";
            case "cryo": return "cryo";
            case "electro":
            case "électro": return "electro";
            case "anemo":
            case "anémo": return "anemo";
            case "geo":
            case "géo": return "geo";
            case "dendro": return "dendro";
            default: return normalized;
        }
    }

    private string NormalizeUsername(string username)
    {
        if (string.IsNullOrEmpty(username))
            return "";

        username = username.Trim();

        if (username.StartsWith("@"))
            username = username.Substring(1);

        return username.ToLower();
    }

    private string FormatDateTime(DateTime date)

    }

    private JObject LoadJObject(string path)
    {
        try
        {
            string json = File.ReadAllText(path);
            return JObject.Parse(json);
        }
        catch (Exception ex)
        {
            CPH.LogError("Erreur lecture JSON : " + ex.Message);
            return null;
        }
    }

    private void EnsureObject(JObject parent, string key)
    {
        if (parent[key] == null || parent[key].Type == JTokenType.Null || parent[key].Type != JTokenType.Object)
            parent[key] = new JObject();
    }

    private void EnsureArray(JObject parent, string key)
    {
        if (parent[key] == null || parent[key].Type == JTokenType.Null || parent[key].Type != JTokenType.Array)
            parent[key] = new JArray();
    }

    private void EnsureInt(JObject obj, string key, int defaultValue)
    {
        if (obj[key] == null || obj[key].Type == JTokenType.Null)
            obj[key] = defaultValue;
    }

    private void EnsureBool(JObject obj, string key, bool defaultValue)
    {
        if (obj[key] == null || obj[key].Type == JTokenType.Null)
            obj[key] = defaultValue;
    }

    private void EnsureString(JObject obj, string key, string defaultValue)
    {
        if (obj[key] == null || obj[key].Type == JTokenType.Null)
            obj[key] = defaultValue;
    }

    private void EnsureNullableString(JObject obj, string key)
    {
        if (obj[key] == null)
            obj[key] = null;
    }

    private int GetInt(JObject obj, string key)
    {
        if (obj == null || obj[key] == null || obj[key].Type == JTokenType.Null)
            return 0;

        int value;

        if (int.TryParse(obj[key].ToString(), out value))
            return value;

        return 0;
    }

    private bool GetBool(JObject obj, string key)
    {
        if (obj == null || obj[key] == null || obj[key].Type == JTokenType.Null)
            return false;

        bool value;

        if (bool.TryParse(obj[key].ToString(), out value))
            return value;

        return false;
    }

    private string GetString(JObject obj, string key)
    {
        if (obj == null || obj[key] == null || obj[key].Type == JTokenType.Null)
            return "";

        return obj[key].ToString();
    }

    private void AddInt(JObject obj, string key, int amount)
    {
        int current = GetInt(obj, key);
        obj[key] = current + amount;
    }

    private void AddStat(JObject viewer, string statName, int amount)
    {
        JObject stats = viewer["stats"] as JObject;

        if (stats == null)
        {
            stats = new JObject();
            viewer["stats"] = stats;
        }

        AddInt(stats, statName, amount);
    }

    private void EnsureMonthlyEventDataDefaults(JObject data, DateTime now)
    {
        int currentYear = now.Year;
        int currentMonth = now.Month;

        int savedYear = GetInt(data, "year");
        int savedMonth = GetInt(data, "month");

        if (savedYear != currentYear || savedMonth != currentMonth)
        {
            data.RemoveAll();
            data["year"] = currentYear;
            data["month"] = currentMonth;
            data["participants"] = new JObject();
            data["dailyWindows"] = new JObject();
            data["gameB"] = new JObject();
            data["messages"] = new JObject();
            data["calendar"] = new JObject();
            data["collectionPurchases"] = new JObject();

            JObject monthlyDraw = new JObject();
            monthlyDraw["drawDone"] = false;
            monthlyDraw["winner"] = "";
            data["monthlyDraw"] = monthlyDraw;

            return;
        }

        EnsureInt(data, "year", currentYear);
        EnsureInt(data, "month", currentMonth);
        EnsureObject(data, "participants");
        EnsureObject(data, "dailyWindows");
        EnsureObject(data, "gameB");
        EnsureObject(data, "messages");
        EnsureObject(data, "calendar");
        EnsureObject(data, "collectionPurchases");
        EnsureObject(data, "monthlyDraw");

        JObject monthlyDrawObj = data["monthlyDraw"] as JObject;
        EnsureBool(monthlyDrawObj, "drawDone", false);
        EnsureString(monthlyDrawObj, "winner", "");
    }

    private bool TrySendPendingEventMessages(JObject eventData, string username)
    {
        if (eventData == null)
            return false;

        JObject messages = eventData["messages"] as JObject;

        if (messages == null)
            return false;

        JArray userMessages = messages[username] as JArray;

        if (userMessages == null || userMessages.Count == 0)
            return false;

        List<string> parts = new List<string>();
        string prefix = GetCurrentEventSocialReceivePrefix(DateTime.Now.Month)
            + " pour toi " + username;

        for (int i = 0; i < userMessages.Count; i++)
        {
            JObject msg = userMessages[i] as JObject;

            if (msg == null)
                continue;

            string sender = GetString(msg, "sender");
            string text = GetString(msg, "text");

            if (string.IsNullOrEmpty(sender) || string.IsNullOrEmpty(text))
                continue;

            parts.Add("de la part de " + sender + " : \"" + text + "\"");
        }

        if (parts.Count == 0)
        {
            messages.Remove(username);
            return true;
        }

        SendSplitEventMessages(prefix, parts);

        messages.Remove(username);

        return true;
    }

    private void SendSplitEventMessages(string prefix, List<string> parts)
    {
        int maxLength = 450;
        string current = prefix + " ";

        for (int i = 0; i < parts.Count; i++)
        {
            string part = parts[i];

            if (current.Length + part.Length + 3 > maxLength)
            {
                CPH.SendMessage(current.TrimEnd());
                current = prefix + " " + part;
            }
            else
            {
                if (current != prefix + " " && !current.EndsWith(" "))
                    current += " | ";

                current += part;
            }
        }

        if (!string.IsNullOrWhiteSpace(current))
            CPH.SendMessage(current.TrimEnd());
    }

    private JObject LoadOrCreateJObject(string path)
    {
        if (!File.Exists(path))
            return new JObject();

        try
        {
            string json = File.ReadAllText(path);

            if (string.IsNullOrWhiteSpace(json))
                return new JObject();

            return JObject.Parse(json);
        }
        catch
        {
            return new JObject();
        }
    }

    private void ApplyVoteBannerCharacter(string folderPath,JObject characterFile,JArray characters,JArray previousBannerFeaturedIds)
    {

    private JObject PickWeightedVoteCharacter(JArray characters, JObject votes, JArray previousBannerFeaturedIds)
    {
        List<JObject> candidates = new List<JObject>();
        List<int> weights = new List<int>();

        foreach (KeyValuePair<string, JToken> entry in votes)
        {
            JObject voteEntry = entry.Value as JObject;

            if (voteEntry == null)
                continue;

            int characterId = GetInt(voteEntry, "characterId");
            int voteCount = GetInt(voteEntry, "votes");

            if (characterId <= 0 || voteCount <= 0)
                continue;

            JObject character = FindCharacterByIdInArray(characters, characterId);

            if (character == null)
                continue;

            if (GetInt(character, "rarete") != 5)
                continue;

            if (GetBool(character, "bannerFeatured"))
                continue;

            if (IsIdInArray(previousBannerFeaturedIds, characterId))
                continue;

            candidates.Add(character);
            weights.Add(voteCount);
        }

        if (candidates.Count == 0)
            return null;

        int totalWeight = 0;

        for (int i = 0; i < weights.Count; i++)
            totalWeight += weights[i];

        if (totalWeight <= 0)
            return null;

        int roll = random.Next(1, totalWeight + 1);
        int cumulative = 0;

        for (int i = 0; i < candidates.Count; i++)
        {
            cumulative += weights[i];

            if (roll <= cumulative)
                return candidates[i];
        }

        return candidates[0];
    }

    private JObject PickRandomVoteFallbackCharacter(JArray characters, JArray previousBannerFeaturedIds)
    {
        List<JObject> candidates = new List<JObject>();

        for (int i = 0; i < characters.Count; i++)
        {
            JObject character = characters[i] as JObject;

            if (character == null)
                continue;

            int characterId = GetInt(character, "id");

            if (GetInt(character, "rarete") != 5)
                continue;

            if (GetBool(character, "bannerFeatured"))
                continue;

            if (IsIdInArray(previousBannerFeaturedIds, characterId))
                continue;

            candidates.Add(character);
        }

        if (candidates.Count == 0)
            return null;

        return candidates[random.Next(candidates.Count)];
    }

    private void CheckLongMissions(JObject viewer, string username, JObject missionsFile, JObject charactersFile, JObject friendshipsData)
    {
        if (viewer == null || missionsFile == null)
            return;

        JObject ranks = missionsFile["ranks"] as JObject;

        if (ranks == null)
            return;

        JObject longMissions = viewer["longMissions"] as JObject;

        if (longMissions == null)
            return;

        JObject categories = longMissions["categories"] as JObject;

        if (categories == null)
            return;

        List<string> categoryKeys = new List<string>();

        foreach (KeyValuePair<string, JToken> entry in categories)
            categoryKeys.Add(entry.Key);

        for (int i = 0; i < categoryKeys.Count; i++)
        {
            string category = categoryKeys[i];
            JObject categoryData = categories[category] as JObject;

            if (categoryData == null)
                continue;

            if (!GetBool(categoryData, "active"))
                continue;

            string rank = GetString(categoryData, "activeRank");
            string missionId = GetString(categoryData, "activeMissionId");

            JObject mission = FindLongMissionById(missionsFile, rank, missionId);

            if (mission == null)
                continue;

            FixMainParticlesMissionBaselineIfNeeded(viewer, categoryData, mission);

            int currentValue = GetLongMissionCurrentValue(viewer, username, GetString(mission, "type"), charactersFile, friendshipsData);
            EnsureLongMissionBaseline(categoryData, currentValue, GetString(mission, "type"));

            int baseline = GetInt(categoryData, "baselineValue");
            int progress = currentValue - baseline;

            if (progress < 0)
                progress = 0;

            int target = GetInt(mission, "target");

            if (progress > target)
                progress = target;

            categoryData["progress"] = progress;

            if (progress >= target)
                CompleteLongMission(viewer, username, missionsFile, mission, categoryData, currentValue);
        }

        CheckUnlockZMissions(viewer, username, missionsFile);
        CheckActiveZMissions(viewer, username, missionsFile, charactersFile, friendshipsData);
    }

    private void EnsureLongMissionBaseline(JObject categoryData, int currentValue, string missionType)
    {
        if (categoryData["baselineValue"] != null && categoryData["baselineValue"].Type != JTokenType.Null)
            return;

        int baseline = currentValue;

        if (missionType == "messages" && baseline > 0)
            baseline--;

        categoryData["baselineValue"] = baseline;
    }

    private void CompleteLongMission(JObject viewer, string username, JObject missionsFile, JObject mission, JObject categoryData, int currentValue)
    {
        string rank = GetString(categoryData, "activeRank");

        JArray completedRanks = categoryData["completedRanks"] as JArray;

        if (completedRanks == null)
        {
            completedRanks = new JArray();
            categoryData["completedRanks"] = completedRanks;
        }

        if (!ArrayContainsString(completedRanks, rank))
            completedRanks.Add(rank);

        categoryData["active"] = false;

        int reward = GetLongMissionReward(missionsFile, rank);

        AddInt(viewer, "primogems", reward);
        AddStat(viewer, "totalPrimosEarned", reward);

        CPH.SendMessage("✅ Mission " + rank + " terminée : " + GetString(mission, "name") + " ! +💠" + reward + " primos (" + GetInt(viewer, "primogems") + ")");

        string nextRank = GetNextLongMissionRank(rank);

        if (!string.IsNullOrEmpty(nextRank))
            AutoAcceptNextLongMission(viewer, username, missionsFile, mission, categoryData, nextRank, currentValue);
    }

    private void AutoAcceptNextLongMission(JObject viewer, string username, JObject missionsFile, JObject previousMission, JObject categoryData, string nextRank, int currentValue)

        categoryData["active"] = true;
        categoryData["activeRank"] = nextRank;
        categoryData["activeMissionId"] = GetString(nextMission, "id");

        JArray acceptedRanks = categoryData["acceptedRanks"] as JArray;

        if (acceptedRanks == null)
        {
            acceptedRanks = new JArray();
            categoryData["acceptedRanks"] = acceptedRanks;
        }

        if (!ArrayContainsString(acceptedRanks, nextRank))
            acceptedRanks.Add(nextRank);

        int target = GetInt(nextMission, "target");
        int baseline = GetInt(categoryData, "baselineValue");
        int progress = currentValue - baseline;

        if (progress < 0)
            progress = 0;

        if (progress > target)
            progress = target;

        categoryData["progress"] = progress;

        CPH.SendMessage("🎯 " + username + " débloque automatiquement [" + nextRank + "] " + GetString(nextMission, "name") + " : " + progress + "/" + target + " " + GetString(nextMission, "progressLabel") + ".");
    }

    private void CheckUnlockZMissions(JObject viewer, string username, JObject missionsFile)
    {
        JObject longMissions = viewer["longMissions"] as JObject;

        if (longMissions == null)
            return;

        if (GetBool(longMissions, "completedAllMainRanksAnnounced"))
            return;

        if (!AreAllMainLongMissionsCompleted(viewer, missionsFile))
            return;

        longMissions["unlockedZ"] = true;
        longMissions["completedAllMainRanksAnnounced"] = true;

        CPH.SendMessage("🏆 INCROYABLE " + username + " ! Toutes les missions B, A et S sont terminées ! Les missions Z sont maintenant débloquées avec !mission Z !");
    }

    private void ApplyDailyBankInterest(JObject viewer, string message, DateTime now)
    {
        if (IsCommand(message))
            return;

        if (IsSystemMessage(message))
            return;

        JObject bank = viewer["bank"] as JObject;

        if (bank == null)
            return;

        string today = now.ToString("yyyy-MM-dd");

        if (GetString(bank, "lastInterestDate") == today)
            return;

        int bankMoras = GetInt(bank, "moras");

        if (bankMoras > 0)
        {
            int interest = bankMoras * 3 / 100;

            if (interest > 0)
            {
                AddInt(bank, "moras", interest);
                AddStat(viewer, "totalMorasEarned", interest);
            }
        }

        bank["lastInterestDate"] = today;
    }

    private void CheckActiveZMissions(JObject viewer, string username, JObject missionsFile, JObject charactersFile, JObject friendshipsData)
    {
        JObject longMissions = viewer["longMissions"] as JObject;

        if (longMissions == null)
            return;

        if (!GetBool(longMissions, "unlockedZ"))
            return;

        JObject z = longMissions["z"] as JObject;

        if (z == null)
            return;

        foreach (KeyValuePair<string, JToken> entry in z)
        {
            JObject zMissionData = entry.Value as JObject;

            if (zMissionData == null)
                continue;

            if (!GetBool(zMissionData, "active"))
                continue;

            if (GetBool(zMissionData, "completed"))
                continue;

            JObject mission = FindLongMissionById(missionsFile, "Z", entry.Key);

            if (mission == null)
                continue;

            int currentValue = GetLongMissionCurrentValue(viewer, username, GetString(mission, "type"), charactersFile, friendshipsData);
            int target = GetInt(mission, "target");

            zMissionData["progress"] = currentValue;

            if (currentValue >= target)
            {
                zMissionData["completed"] = true;
                zMissionData["active"] = false;

                int reward = GetLongMissionReward(missionsFile, "Z");

                AddInt(viewer, "primogems", reward);
                AddStat(viewer, "totalPrimosEarned", reward);

                CPH.SendMessage("🏆 Mission Z terminée : " + GetString(mission, "name") + " ! +💠" + reward + " primos (" + GetInt(viewer, "primogems") + ")");
            }
        }
    }

    private void FixMainParticlesMissionBaselineIfNeeded(JObject viewer, JObject categoryData, JObject mission)
    {
        if (GetString(mission, "type") != "mainParticlesEarned")
            return;

        if (GetBool(categoryData, "usesCumulativeMainParticles"))
            return;

        JObject stats = viewer["stats"] as JObject;

        if (stats == null)
            return;

        EnsureInt(stats, "totalMainElementParticlesEarned", 0);

        int currentCumulative = GetInt(stats, "totalMainElementParticlesEarned");
        int currentProgress = GetInt(categoryData, "progress");

        categoryData["baselineValue"] = currentCumulative - currentProgress;

        if (GetInt(categoryData, "baselineValue") < 0)
            categoryData["baselineValue"] = 0;

        categoryData["usesCumulativeMainParticles"] = true;
    }

    private int GetLongMissionCurrentValue(JObject viewer, string username, string type, JObject charactersFile, JObject friendshipsData)
    {
        JObject stats = viewer["stats"] as JObject;

        switch (type)
        {
            case "messages":
                return GetInt(stats, "countedMessages");

            case "pulls":
                return GetInt(stats, "totalPulls");

            case "characters4":
                return CountOwnedCharactersByRarity(viewer, charactersFile, 4);

            case "characters5":
                return CountOwnedCharactersByRarity(viewer, charactersFile, 5);

            case "morasEarned":
                return GetInt(stats, "totalMorasEarned");

            case "mainParticlesEarned":
                return GetInt(stats, "totalMainElementParticlesEarned");

            case "expeditions":
                return GetInt(stats, "totalExpeditionsCompleted");

            case "combatWins":
                return GetInt(stats, "totalCombatWins");

			case "manualCombatWins":
    			return GetInt(stats, "totalManualCombatWins");

            case "friendHeartsSent":
                return GetInt(stats, "totalFriendHeartsSent");

            case "c6FiveStars":
                return CountC6Characters(viewer);

            case "perfectFriendship":
                return HasPerfectFriendship(username, friendshipsData) ? 1 : 0;

            case "level100":
                return GetInt(viewer, "level");

            default:
                return 0;
        }
    }
