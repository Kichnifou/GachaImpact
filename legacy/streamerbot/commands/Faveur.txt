/*
CE QUE FAIT LE CODE :

- Gère la commande !faveur
- Affiche l’état de la Faveur de l’Astre du viewer
- Accepte :
  - !faveur
  - !faveur pseudo
  - !faveur @pseudo
- Lit viewers_data.json en JObject
- Ne modifie pas viewers_data.json
- Ne crée pas de viewer
- Initialise uniquement la lecture de la section favor si elle existe
- Affiche :
  - aucune faveur active
  - faveur active avec jours restants
  - maximum atteint si 180 jours
- Compatible Streamer.bot
- Sans LINQ
*/

using System;
using System.IO;
using Newtonsoft.Json.Linq;

public class CPHInline
{
    private const int FavorMaxDays = 180;

    public bool Execute()
    {
        string path = @"C:\Users\axeld\Documents\Kichnifou\Twitch\Streambot\Data\viewers_data.json";

        if (!args.ContainsKey("userName") || args["userName"] == null)
            return false;

        string username = NormalizeUsername(args["userName"].ToString());
        string input = GetCommandInput();

        string target = username;

        if (!string.IsNullOrEmpty(input))
            target = NormalizeUsername(input);

        if (!File.Exists(path))
        {
            SendWarning("Aucune donnée viewer trouvée.");
            return true;
        }

        JObject viewers = LoadJObject(path);

        if (viewers == null)
        {
            SendError("Impossible de lire viewers_data.json.");
            return true;
        }

        JObject viewer = viewers[target] as JObject;

        if (viewer == null)
        {
            SendWarning(username + ", joueur introuvable.");
            return true;
        }

        JObject favor = viewer["favor"] as JObject;

        if (favor == null)
        {
            CPH.SendMessage("🌙 " + target + " | Aucune Faveur de l'Astre active");
            return true;
        }

        int daysRemaining = GetInt(favor, "daysRemaining");

        if (daysRemaining <= 0)
        {
            CPH.SendMessage("🌙 " + target + " | Aucune Faveur de l'Astre active");
            return true;
        }

        if (daysRemaining >= FavorMaxDays)
        {
            CPH.SendMessage("🌙 " + target + " | Faveur de l'Astre | ⏳ " + FavorMaxDays + " jour(s) restants | 👑 Maximum atteint");
            return true;
        }

        CPH.SendMessage("🌙 " + target + " | Faveur de l'Astre | ⏳ " + daysRemaining + " jour(s) restants | 🎁 +💠800 primos/jour");

        return true;
    }

    private string GetCommandInput()
    {
        if (args.ContainsKey("rawInput") && args["rawInput"] != null)
            return args["rawInput"].ToString().Trim();

        if (args.ContainsKey("input0") && args["input0"] != null)
            return args["input0"].ToString().Trim();

        return "";
    }

    private JObject LoadJObject(string path)
    {
        try
        {
            string json = File.ReadAllText(path);

            if (string.IsNullOrWhiteSpace(json))
                return new JObject();

            JObject data = JObject.Parse(json);

            if (data == null)
                return new JObject();

            return data;
        }
        catch
        {
            return null;
        }
    }

    private int GetInt(JObject obj, string key)
    {
        if (obj == null || obj[key] == null || obj[key].Type == JTokenType.Null)
            return 0;

        int value;

        if (int.TryParse(obj[key].ToString(), out value))
            return value;

        return 0;
    }

    private string NormalizeUsername(string username)
    {
        if (string.IsNullOrEmpty(username))
            return "";

        username = username.Trim();

        if (username.StartsWith("@"))
            username = username.Substring(1);

        return username.ToLower();
    }

    private void SendWarning(string message)
    {
        CPH.SendMessage("⚠️ " + message);
    }

    private void SendError(string message)
    {
        CPH.SendMessage("❌ " + message);
    }
}
