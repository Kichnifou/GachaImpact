    {

    private void ShowCurrentTeamEvaluation(string username, JObject viewer, JObject combatData, JObject combatConfig, CharacterFile characterFile, string today)
    {
        List<int> teamIds = GetViewerTeamIds(viewer);
        List<string> validationErrors = ValidateTeam(viewer, teamIds, characterFile, today);

        if (validationErrors.Count > 0)
        {
            SendWarning(username + ", évaluation impossible : " + string.Join(" | ", validationErrors.ToArray()));
            return;
        }

        JArray enemyTeam = GetArray(combatData, "enemyTeam");

        double chance = CalculateWinChance(viewer, teamIds, enemyTeam, combatConfig, characterFile);
        string teamText = FormatTeamDisplayFromIds(teamIds, viewer, characterFile, true);

        SendInfo("⚔️ Évaluation combat | Chance de victoire : " + FormatPercent(chance) + " | Team : " + teamText);
    }

    private void RunCombat(string viewersPath, JObject viewers, JObject viewer, string username, List<int> teamIds, bool isAuto, JObject combatData, JObject combatConfig, CharacterFile characterFile, string today)
    {
        if (GetString(GetObject(viewer, "combat"), "lastWinDate") == today)
        {
            SendWarning(username + ", tu as déjà gagné ton combat aujourd’hui. Reviens demain !");
            SaveJObject(viewersPath, viewers);
            return;
        }

        List<string> validationErrors = ValidateTeam(viewer, teamIds, characterFile, today);

        if (validationErrors.Count > 0)
        {
            SendWarning(username + ", combat impossible : " + string.Join(" | ", validationErrors.ToArray()));
            SaveJObject(viewersPath, viewers);
            return;
        }

        JArray enemyTeam = GetArray(combatData, "enemyTeam");

        if (enemyTeam.Count < TeamSize)
        {
            SendError("Team ennemie invalide. Réessaie !combat pour régénérer le combat du jour.");
            SaveJObject(viewersPath, viewers);
            return;
        }

        double chance = CalculateWinChance(viewer, teamIds, enemyTeam, combatConfig, characterFile);
        bool win = random.NextDouble() * 100.0 < chance;

        JObject combat = GetObject(viewer, "combat");
        combat["lastFightDate"] = today;
        combat["lastResult"] = win ? "win" : "lose";

        AddStat(viewer, "totalCombatFights", 1);

        if (win)
        {
            combat["lastWinDate"] = today;
            AddInt(viewer, "primogems", WinPrimos);
            AddInt(viewer, "moras", WinMoras);
            AddStat(viewer, "totalPrimosEarned", WinPrimos);
			AddStat(viewer, "totalMorasEarned", WinMoras);
			AddStat(viewer, "totalCombatWins", 1);

			// Compte uniquement les victoires faites avec !combat go.
			// !combat auto appelle RunCombat avec isAuto = true.
			if (!isAuto)
				AddStat(viewer, "totalManualCombatWins", 1);

			AddCharacterStat(combat, "characterWins", teamIds);

            SaveJObject(viewersPath, viewers);

            string teamText = FormatTeamDisplayFromIds(teamIds, viewer, characterFile, true);
            string modeText = isAuto ? "Team auto temporaire" : "Team";
            SendSplit("✅ ⚔️ " + username + " gagne le combat ! Chance de victoire : " + FormatPercent(chance) + " | " + modeText + " : " + teamText + " | Gain : +💠" + WinPrimos + " (" + GetInt(viewer, "primogems") + "), +💰" + WinMoras + " (" + GetInt(viewer, "moras") + ")");
            return;
        }

        AddStat(viewer, "totalCombatLosses", 1);
        AddCharacterStat(combat, "characterLosses", teamIds);
        LockLostCharacters(viewer, today, teamIds);

        SaveJObject(viewersPath, viewers);

        string lockedText = FormatTeamDisplayFromIds(teamIds, viewer, characterFile, true);
        string autoText = isAuto ? "Team auto temporaire utilisée. " : "";
        SendSplit("❌ ⚔️ " + username + " perd le combat. Chance de victoire : " + FormatPercent(chance) + " | " + autoText + "Personnages bloqués jusqu’à demain : " + lockedText);
    }

    private List<string> ValidateTeam(JObject viewer, List<int> teamIds, CharacterFile characterFile, string today)
    {
        List<string> errors = new List<string>();

        if (teamIds == null || teamIds.Count < TeamSize)
        {
            errors.Add("ta team doit contenir 4 personnages");
            return errors;
        }

        if (teamIds.Count > TeamSize)
            errors.Add("ta team contient trop de personnages");

        List<int> unique = new List<int>();
        List<string> lockedNames = new List<string>();

        for (int i = 0; i < teamIds.Count && i < TeamSize; i++)
        {
            int characterId = teamIds[i];

            if (characterId <= 0)
            {
                errors.Add("personnage invalide dans la team");
                continue;
            }

            if (ContainsInt(unique, characterId))
            {
                errors.Add("doublon dans la team");
                continue;
            }

            unique.Add(characterId);

            CharacterData character = FindCharacterById(characterFile, characterId);

            if (character == null)
            {
                errors.Add("personnage introuvable : " + characterId);
                continue;
            }

            if (!ViewerOwnsCharacter(viewer, characterId))
            {
                errors.Add("tu ne possèdes pas " + character.nom);
                continue;
            }

            if (IsCharacterLockedToday(viewer, characterId, today))
                lockedNames.Add(GetElementEmoji(character.element) + " " + character.nom);
        }

        if (lockedNames.Count > 0)
            errors.Add("change " + string.Join(", ", lockedNames.ToArray()) + " déjà KO aujourd’hui");

        return errors;
    }

    private double CalculateWinChance(JObject viewer, List<int> playerTeam, JArray enemyTeam, JObject combatConfig, CharacterFile characterFile)
    {
        double chance = BaseWinChance;

        for (int i = 0; i < playerTeam.Count && i < TeamSize; i++)
        {
            CharacterData playerCharacter = FindCharacterById(characterFile, playerTeam[i]);

            if (playerCharacter == null)
                continue;

            string playerElement = NormalizeElement(playerCharacter.element);

            for (int j = 0; j < enemyTeam.Count && j < TeamSize; j++)
            {
                int enemyId = ToInt(enemyTeam[j]);
                CharacterData enemyCharacter = FindCharacterById(characterFile, enemyId);

                if (enemyCharacter == null)
                    continue;

                string enemyElement = NormalizeElement(enemyCharacter.element);
                int relation = GetElementRelation(playerElement, enemyElement, combatConfig);
                chance += relation * ElementBonus;
            }

            int constellation = GetCharacterConstellation(viewer, playerCharacter.id);

            if (constellation < 0)
                constellation = 0;

            if (constellation > 6)
                constellation = 6;

            if (playerCharacter.rarete == 5)
                chance += constellation * FiveStarConstellationBonus;
            else
                chance += constellation * FourStarConstellationBonus;
        }

        if (chance < MinWinChance)
            chance = MinWinChance;

        if (chance > MaxWinChance)
            chance = MaxWinChance;

        return chance;
    }

    private List<int> BuildBestAutoTeam(JObject viewer, JObject combatData, JObject combatConfig, CharacterFile characterFile, string today)
    {
        List<CombatCandidate> candidates = new List<CombatCandidate>();
        JObject box = GetObject(viewer, "box");
        JArray enemyTeam = GetArray(combatData, "enemyTeam");

        foreach (KeyValuePair<string, JToken> entry in box)
        {
            JObject owned = entry.Value as JObject;

            if (owned == null)
                continue;

            int characterId = GetInt(owned, "characterId");

            if (characterId <= 0)
            {
                int parsed;
                if (int.TryParse(entry.Key, out parsed))
                    characterId = parsed;
            }

            if (characterId <= 0)
                continue;

            if (IsCharacterLockedToday(viewer, characterId, today))
                continue;

            CharacterData character = FindCharacterById(characterFile, characterId);

            if (character == null)
                continue;

            CombatCandidate candidate = new CombatCandidate();
            candidate.characterId = characterId;
            candidate.score = CalculateAutoScore(viewer, character, enemyTeam, combatConfig, characterFile);
            candidates.Add(candidate);
        }

        candidates.Sort(delegate(CombatCandidate a, CombatCandidate b)
        {
            int scoreCompare = b.score.CompareTo(a.score);

            if (scoreCompare != 0)
                return scoreCompare;

            return a.characterId.CompareTo(b.characterId);
        });

        List<int> result = new List<int>();

        for (int i = 0; i < candidates.Count && result.Count < TeamSize; i++)
            result.Add(candidates[i].characterId);

        return result;
    }

    private double CalculateAutoScore(JObject viewer, CharacterData character, JArray enemyTeam, JObject combatConfig, CharacterFile characterFile)
    {
        double score = 0;
        string element = NormalizeElement(character.element);

        for (int i = 0; i < enemyTeam.Count && i < TeamSize; i++)
        {
            CharacterData enemy = FindCharacterById(characterFile, ToInt(enemyTeam[i]));

            if (enemy == null)
                continue;

            score += GetElementRelation(element, NormalizeElement(enemy.element), combatConfig) * 10;
        }

        int constellation = GetCharacterConstellation(viewer, character.id);

        if (constellation < 0)
            constellation = 0;

        if (constellation > 6)
            constellation = 6;

        if (character.rarete == 5)
            score += 20 + (constellation * 3);
        else
            score += 10 + (constellation * 1.5);

        return score;
    }

    private int GetElementRelation(string attackerElement, string defenderElement, JObject combatConfig)
    {
        attackerElement = NormalizeElement(attackerElement);
        defenderElement = NormalizeElement(defenderElement);

        if (string.IsNullOrEmpty(attackerElement) || string.IsNullOrEmpty(defenderElement))
            return 0;

        JObject rule = combatConfig[attackerElement] as JObject;

        if (rule == null)
            return 0;

        JArray strongAgainst = rule["strongAgainst"] as JArray;
        JArray weakAgainst = rule["weakAgainst"] as JArray;

        if (ArrayContainsString(strongAgainst, defenderElement))
            return 1;

        if (ArrayContainsString(weakAgainst, defenderElement))
            return -1;

        return 0;
    }

    private void EnsureMonthlyBossData(JObject data, DateTime now)
    {
        if (data == null)
            return;

        JObject globalStats = GetObject(data, "globalStats");
        EnsureInt(globalStats, "totalBossesGenerated", 0);
        EnsureInt(globalStats, "totalBossesDefeated", 0);
        EnsureInt(globalStats, "totalDamage", 0);
        EnsureInt(globalStats, "totalAttacks", 0);
        EnsureInt(globalStats, "highestHit", 0);
        EnsureString(globalStats, "highestHitPlayer", "");

        GetArray(data, "history");

        string currentMonth = now.ToString("yyyy-MM");
        JObject currentBoss = data["currentBoss"] as JObject;

        if (currentBoss == null || GetString(currentBoss, "month") != currentMonth)
        {
            if (currentBoss != null && !string.IsNullOrEmpty(GetString(currentBoss, "month")))
                ArchiveCurrentBoss(data, currentBoss);

            data["currentBoss"] = CreateMonthlyBoss(now);
            AddInt(globalStats, "totalBossesGenerated", 1);
        }
        else
        {
            EnsureMonthlyBossDefaults(currentBoss);
        }
    }

    private JObject CreateMonthlyBoss(DateTime now)
    {
        JObject boss = new JObject();
        string month = now.ToString("yyyy-MM");

        int minHp = (int)Math.Round(BossBaseHp * (1.0 - BossHpVariation));
        int maxHp = (int)Math.Round(BossBaseHp * (1.0 + BossHpVariation));
        int rawHp = random.Next(minHp, maxHp + 1);

        // Arrondi à 10 000 pour garder des PV lisibles.
        int maxBossHp = (int)(Math.Round(rawHp / 10000.0) * 10000.0);

        string[] resistances = new string[]
        {
            "pyro", "hydro", "cryo", "electro", "anemo", "geo", "dendro"
        };

        boss["month"] = month;
        boss["name"] = GetBossNameForMonth(now.Month);
        boss["maxHp"] = maxBossHp;
        boss["currentHp"] = maxBossHp;
        boss["resistance"] = resistances[random.Next(resistances.Length)];
        boss["defeated"] = false;
        boss["defeatedAt"] = "";
        boss["finalBlowBy"] = "";
        boss["totalDamage"] = 0;
        boss["totalAttacks"] = 0;
        boss["rewardsDistributed"] = false;
        boss["participants"] = new JObject();
        boss["createdAt"] = now.ToString("yyyy-MM-dd HH:mm:ss");

        return boss;
    }

    private void EnsureMonthlyBossDefaults(JObject boss)
    {
        EnsureString(boss, "month", "");
        EnsureString(boss, "name", "Boss inconnu");
        EnsureInt(boss, "maxHp", BossBaseHp);
        EnsureInt(boss, "currentHp", GetInt(boss, "maxHp"));
        EnsureString(boss, "resistance", "pyro");
        EnsureBool(boss, "defeated", false);
        EnsureString(boss, "defeatedAt", "");
        EnsureString(boss, "finalBlowBy", "");
        EnsureInt(boss, "totalDamage", 0);
        EnsureInt(boss, "totalAttacks", 0);
        EnsureBool(boss, "rewardsDistributed", false);
        GetObject(boss, "participants");
        EnsureString(boss, "createdAt", "");
    }

    private void ArchiveCurrentBoss(JObject data, JObject currentBoss)
    {
        JArray history = GetArray(data, "history");

        for (int i = 0; i < history.Count; i++)
        {
            JObject archived = history[i] as JObject;

            if (archived != null && GetString(archived, "month") == GetString(currentBoss, "month"))
                return;
        }

        history.Add(currentBoss.DeepClone());
    }

    private string GetBossNameForMonth(int month)

    private void RunMonthlyBossAttack(
        string viewersPath,
        string monthlyBossPath,
        JObject viewers,
        JObject viewer,
        string username,
        JObject data,
        CharacterFile characterFile,
        string today)
    {
        JObject boss = data["currentBoss"] as JObject;

        if (boss == null)
        {
            SendError("Boss mensuel introuvable.");
            return;
        }

        if (GetBool(boss, "defeated") || GetInt(boss, "currentHp") <= 0)
        {
            SendWarning(
                username + ", " + GetString(boss, "name") +
                " a déjà été vaincu. Attends le mois prochain pour affronter un nouveau Boss."
            );
            return;
        }

        JObject participants = GetObject(boss, "participants");
        JObject participant = participants[username] as JObject;

        if (participant != null && GetString(participant, "lastAttackDate") == today)
        {
            SendWarning(username + ", tu as déjà attaqué le Boss aujourd’hui. Reviens demain !");
            return;
        }

        List<int> teamIds = GetViewerTeamIds(viewer);
        List<string> validationErrors = ValidateBossTeam(viewer, teamIds, characterFile);

        if (validationErrors.Count > 0)
        {
            SendWarning(username + ", attaque Boss impossible : " + string.Join(" | ", validationErrors.ToArray()));
            return;
        }

        string resistance = GetString(boss, "resistance");
        int calculatedDamage = CalculateBossTeamDamage(viewer, teamIds, characterFile, resistance);
        int currentHp = GetInt(boss, "currentHp");
        int appliedDamage = calculatedDamage;

        if (appliedDamage > currentHp)
            appliedDamage = currentHp;

        if (appliedDamage < 0)
            appliedDamage = 0;

        int remainingHp = currentHp - appliedDamage;

        if (remainingHp < 0)
            remainingHp = 0;

        boss["currentHp"] = remainingHp;
        AddInt(boss, "totalDamage", appliedDamage);
        AddInt(boss, "totalAttacks", 1);

        if (participant == null)
        {
            participant = new JObject();
            participant["damage"] = 0;
            participant["attacks"] = 0;
            participant["lastAttackDate"] = "";
            participant["highestHit"] = 0;
            participants[username] = participant;
        }

        AddInt(participant, "damage", appliedDamage);
        AddInt(participant, "attacks", 1);
        participant["lastAttackDate"] = today;

        if (appliedDamage > GetInt(participant, "highestHit"))
            participant["highestHit"] = appliedDamage;

        JObject stats = GetObject(viewer, "stats");

        AddInt(stats, "totalBossDamage", appliedDamage);
        AddInt(stats, "totalBossAttacks", 1);

        if (appliedDamage > GetInt(stats, "highestBossHit"))
            stats["highestBossHit"] = appliedDamage;

        string bossMonth = GetString(boss, "month");

        if (GetString(stats, "lastBossParticipationMonth") != bossMonth)
        {
            AddInt(stats, "bossesParticipated", 1);
            stats["lastBossParticipationMonth"] = bossMonth;
        }

        JObject globalStats = GetObject(data, "globalStats");
        AddInt(globalStats, "totalDamage", appliedDamage);
        AddInt(globalStats, "totalAttacks", 1);

        if (appliedDamage > GetInt(globalStats, "highestHit"))
        {
            globalStats["highestHit"] = appliedDamage;
            globalStats["highestHitPlayer"] = username;
        }

        string teamText = FormatTeamDisplayFromIds(teamIds, viewer, characterFile, true);

        if (remainingHp > 0)
        {
            SaveJObject(viewersPath, viewers);
            SaveJObject(monthlyBossPath, data);

            SendSplit(
                "⚔️ " + username + " inflige " + appliedDamage + " DMG à " +
                GetString(boss, "name") + " grâce à sa team [" + teamText + "]" +
                " | ❤️ PV restants : " + remainingHp + "/" + GetInt(boss, "maxHp")
            );
            return;
        }

        boss["defeated"] = true;
        boss["defeatedAt"] = DateTime.Now.ToString("yyyy-MM-dd HH:mm:ss");
        boss["finalBlowBy"] = username;

        AddInt(stats, "totalBossFinalBlows", 1);
        AddInt(globalStats, "totalBossesDefeated", 1);

        DistributeBossCommunityRewards(viewers, boss);

        SaveJObject(viewersPath, viewers);
        SaveJObject(monthlyBossPath, data);

        SendSplit(
            "⚔️ " + username + " inflige " + appliedDamage + " DMG à " +
            GetString(boss, "name") + " grâce à sa team [" + teamText + "]" +
            " | ❤️ PV restants : 0/" + GetInt(boss, "maxHp")
        );

        SendSplit(
            "🏆 " + GetString(boss, "name") +
            " a été vaincu ! Tous les participants remportent +💠" +
            BossRewardPrimos + " primos et +💰" + BossRewardMoras +
            " moras ! Un nouveau Boss apparaîtra le mois prochain."
        );
    }

    private List<string> ValidateBossTeam(JObject viewer, List<int> teamIds, CharacterFile characterFile)
    {
        List<string> errors = new List<string>();

        if (teamIds == null || teamIds.Count != TeamSize)
        {
            errors.Add("ta team doit contenir exactement 4 personnages");
            return errors;
        }

        List<int> unique = new List<int>();

        for (int i = 0; i < teamIds.Count; i++)
        {
            int characterId = teamIds[i];

            if (characterId <= 0)
            {
                errors.Add("personnage invalide dans la team");
                continue;
            }

            if (ContainsInt(unique, characterId))
            {
                errors.Add("doublon dans la team");
                continue;
            }

            unique.Add(characterId);

            CharacterData character = FindCharacterById(characterFile, characterId);

            if (character == null)
            {
                errors.Add("personnage introuvable : " + characterId);
                continue;
            }

            if (!ViewerOwnsCharacter(viewer, characterId))
                errors.Add("tu ne possèdes pas " + character.nom);
        }

        return errors;
    }

    private int CalculateBossTeamDamage(
        JObject viewer,
        List<int> teamIds,
        CharacterFile characterFile,
        string bossResistance)
    {
        int totalDamage = 0;

        for (int i = 0; i < teamIds.Count && i < TeamSize; i++)
        {
            CharacterData character = FindCharacterById(characterFile, teamIds[i]);

            if (character == null)
                continue;

            int constellation = GetCharacterConstellation(viewer, character.id);

            if (constellation < 0)
                constellation = 0;

            if (constellation > 6)
                constellation = 6;

            int damage;

            if (character.rarete == 5)
                damage = FiveStarBossBaseDamage + (constellation * FiveStarBossConstellationDamage);
            else
                damage = FourStarBossBaseDamage + (constellation * FourStarBossConstellationDamage);

            if (NormalizeElement(character.element) == NormalizeElement(bossResistance))
                damage /= 2;

            totalDamage += damage;
        }

        return totalDamage;
    }

    private void DistributeBossCommunityRewards(JObject viewers, JObject boss)
    {
        if (GetBool(boss, "rewardsDistributed"))
            return;

        JObject participants = GetObject(boss, "participants");

        foreach (KeyValuePair<string, JToken> entry in participants)
        {
            string participantUsername = NormalizeUsername(entry.Key);
            JObject participantData = entry.Value as JObject;

            if (participantData == null || GetInt(participantData, "damage") <= 0)
                continue;

            JObject participantViewer = viewers[participantUsername] as JObject;

            if (participantViewer == null)
                continue;

            EnsureViewerDefaults(participantViewer, participantUsername);

            AddInt(participantViewer, "primogems", BossRewardPrimos);
            AddInt(participantViewer, "moras", BossRewardMoras);
            AddStat(participantViewer, "totalPrimosEarned", BossRewardPrimos);
            AddStat(participantViewer, "totalMorasEarned", BossRewardMoras);
            AddStat(participantViewer, "bossesDefeated", 1);
        }

        boss["rewardsDistributed"] = true;
    }

    private void EnsureCombatDataForToday(JObject combatData, string today, CharacterFile characterFile)
    {
        string currentDate = GetString(combatData, "date");
        JArray enemyTeam = combatData["enemyTeam"] as JArray;

        if (currentDate != today || enemyTeam == null || enemyTeam.Count < TeamSize)
        {
            combatData.RemoveAll();
            combatData["date"] = today;
            combatData["enemyTeam"] = GenerateEnemyTeam(characterFile);
        }
    }

    private JArray GenerateEnemyTeam(CharacterFile characterFile)
    {
        JArray result = new JArray();
        List<CharacterData> pool = new List<CharacterData>();

        if (characterFile != null && characterFile.characters != null)
        {
            for (int i = 0; i < characterFile.characters.Count; i++)
            {
                CharacterData c = characterFile.characters[i];

                if (c == null)
                    continue;

                if (c.id <= 0)
                    continue;

                if (string.IsNullOrEmpty(NormalizeElement(c.element)))
                    continue;

                pool.Add(c);
            }
        }

        while (result.Count < TeamSize && pool.Count > 0)
        {
            int index = random.Next(pool.Count);
            result.Add(pool[index].id);
            pool.RemoveAt(index);
        }

        return result;
    }

    private bool ViewerOwnsCharacter(JObject viewer, int characterId)
    {
        JObject box = GetObject(viewer, "box");
        string idKey = characterId.ToString();

        if (box[idKey] != null && box[idKey].Type == JTokenType.Object)
            return true;

        foreach (KeyValuePair<string, JToken> entry in box)
        {
            JObject owned = entry.Value as JObject;

            if (owned != null && GetInt(owned, "characterId") == characterId)
                return true;
        }

        return false;
    }

    private int GetCharacterConstellation(JObject viewer, int characterId)
    {
        JObject box = GetObject(viewer, "box");
        string idKey = characterId.ToString();
        JObject ownedByKey = box[idKey] as JObject;

        if (ownedByKey != null)
            return GetInt(ownedByKey, "constellation");

        foreach (KeyValuePair<string, JToken> entry in box)
        {
            JObject owned = entry.Value as JObject;

            if (owned != null && GetInt(owned, "characterId") == characterId)
                return GetInt(owned, "constellation");
        }

        return 0;
    }

    }

    private string FormatTeamDisplayFromIds(List<int> ids, JObject viewer, CharacterFile characterFile, bool includeConstellation)
    {
        List<string> texts = new List<string>();

        for (int i = 0; i < ids.Count && i < TeamSize; i++)
        {
            CharacterData character = FindCharacterById(characterFile, ids[i]);

            if (character == null)
                continue;

            string text = GetElementEmoji(character.element) + " " + character.nom;

            if (includeConstellation && viewer != null)
                text += " (C" + GetCharacterConstellation(viewer, character.id) + ")";
            else
                text += " (" + character.rarete + "★)";

            texts.Add(text);
        }

        if (texts.Count == 0)
            return "aucune";

        return string.Join(" - ", texts.ToArray());
    }

    private CharacterData FindCharacterById(CharacterFile characterFile, int characterId)
    {

        {
            CharacterData character = characterFile.characters[i];

            if (character != null && character.id == characterId)
                return character;
        }

        return null;
    }

    private JObject LoadJObject(string path)
    {
        try
        {
            if (!File.Exists(path))
                return new JObject();

            string json = File.ReadAllText(path);

            if (string.IsNullOrWhiteSpace(json))
                return new JObject();

            JObject data = JObject.Parse(json);

            if (data == null)
                return new JObject();

            return data;
        }
        catch
        {
            return null;
        }
    }

    private void SaveJObject(string path, JObject data)
    {
        string folder = Path.GetDirectoryName(path);

        if (!Directory.Exists(folder))
            Directory.CreateDirectory(folder);

        File.WriteAllText(path, data.ToString(Formatting.Indented));
    }

    private CharacterFile LoadCharacters(string path)
    {
        try
        {
            string json = File.ReadAllText(path);
            CharacterFile characterFile = JsonConvert.DeserializeObject<CharacterFile>(json);

            if (characterFile == null)
                characterFile = new CharacterFile();

            if (characterFile.characters == null)
                characterFile.characters = new List<CharacterData>();

            return characterFile;
        }
        catch
        {
            return null;
        }
    }

    private JObject GetObject(JObject parent, string key)
    {
        if (parent[key] == null || parent[key].Type == JTokenType.Null || parent[key].Type != JTokenType.Object)
            parent[key] = new JObject();

        return parent[key] as JObject;
    }

    private JArray GetArray(JObject parent, string key)
    {
        if (parent[key] == null || parent[key].Type == JTokenType.Null || parent[key].Type != JTokenType.Array)
            parent[key] = new JArray();

        return parent[key] as JArray;
    }

    private int GetInt(JObject obj, string key)
    {
        if (obj == null || obj[key] == null || obj[key].Type == JTokenType.Null)
            return 0;

        int value;
        if (int.TryParse(obj[key].ToString(), out value))
            return value;

        return 0;
    }

    private string GetString(JObject obj, string key)
    {
        if (obj == null || obj[key] == null || obj[key].Type == JTokenType.Null)
            return "";

        return obj[key].ToString();
    }

    private bool GetBool(JObject obj, string key)
    {
        if (obj == null || obj[key] == null || obj[key].Type == JTokenType.Null)
            return false;

        bool value;
        if (bool.TryParse(obj[key].ToString(), out value))
            return value;

        return false;
    }

    private void AddInt(JObject obj, string key, int amount)
    {
        int current = GetInt(obj, key);
        obj[key] = current + amount;
    }

    private void AddStat(JObject viewer, string statName, int amount)
    {
        JObject stats = GetObject(viewer, "stats");
        AddInt(stats, statName, amount);
    }

    private void EnsureInt(JObject obj, string key, int defaultValue)
    {
        if (obj[key] == null || obj[key].Type == JTokenType.Null)
            obj[key] = defaultValue;
    }

    private void EnsureBool(JObject obj, string key, bool defaultValue)
    {
        if (obj[key] == null || obj[key].Type == JTokenType.Null)
            obj[key] = defaultValue;
    }

    private void EnsureString(JObject obj, string key, string defaultValue)
    {
        if (obj[key] == null || obj[key].Type == JTokenType.Null)
            obj[key] = defaultValue;
    }

    private void EnsureNullOrString(JObject obj, string key)
    {
        if (obj[key] == null)
            obj[key] = null;
    }

    private bool ArrayContainsString(JArray array, string value)
    {
        if (array == null)
            return false;

        for (int i = 0; i < array.Count; i++)
        {
            if (array[i] != null && NormalizeElement(array[i].ToString()) == NormalizeElement(value))
                return true;
        }

        return false;
    }

    private bool ArrayContainsInt(JArray array, int value)
    {
        if (array == null)
            return false;

        for (int i = 0; i < array.Count; i++)
        {
            if (ToInt(array[i]) == value)
                return true;
        }

        return false;
    }

    private bool ContainsInt(List<int> list, int value)
    {
        if (list == null)
            return false;

        for (int i = 0; i < list.Count; i++)
        {
            if (list[i] == value)
                return true;
        }

        return false;
    }

    private int ToInt(JToken token)
    {
        if (token == null || token.Type == JTokenType.Null)
            return 0;

        int value;
        if (int.TryParse(token.ToString(), out value))
            return value;

        return 0;
    }

    private string GetCommandInput()
    {
        if (args.ContainsKey("rawInput") && args["rawInput"] != null)
            return args["rawInput"].ToString().Trim();

        if (args.ContainsKey("input0") && args["input0"] != null)
            return args["input0"].ToString().Trim();

        return "";
    }

    private string NormalizeUsername(string username)
    {
        if (string.IsNullOrEmpty(username))
            return "";

        username = username.Trim();

        if (username.StartsWith("@"))
            username = username.Substring(1);

        return username.ToLower();
    }

    private string NormalizeText(string text)
    {
        if (string.IsNullOrEmpty(text))
            return "";

        string value = text.ToLower().Trim();
        value = value.Replace("à", "a"); value = value.Replace("á", "a"); value = value.Replace("â", "a"); value = value.Replace("ä", "a"); value = value.Replace("ã", "a");
        value = value.Replace("ç", "c");
        value = value.Replace("è", "e"); value = value.Replace("é", "e"); value = value.Replace("ê", "e"); value = value.Replace("ë", "e");
        value = value.Replace("ì", "i"); value = value.Replace("í", "i"); value = value.Replace("î", "i"); value = value.Replace("ï", "i");
        value = value.Replace("ò", "o"); value = value.Replace("ó", "o"); value = value.Replace("ô", "o"); value = value.Replace("ö", "o"); value = value.Replace("õ", "o");
        value = value.Replace("ù", "u"); value = value.Replace("ú", "u"); value = value.Replace("û", "u"); value = value.Replace("ü", "u");
        value = value.Replace("ÿ", "y"); value = value.Replace("œ", "oe"); value = value.Replace("æ", "ae");
        return value;
    }

    private string NormalizeElement(string element)
    {
        if (string.IsNullOrEmpty(element))
            return "";

        string normalized = NormalizeText(element);

        switch (normalized)
        {
            case "pyro": return "pyro";
            case "hydro": return "hydro";
            case "cryo": return "cryo";
            case "electro": return "electro";
            case "anemo": return "anemo";
            case "geo": return "geo";
            case "dendro": return "dendro";
            default: return "";
        }
    }

    private string GetElementEmoji(string element)
    {
        switch (NormalizeElement(element))
        {
            case "pyro": return "🔥";
            case "hydro": return "💧";
            case "cryo": return "❄️";
            case "electro": return "⚡";
            case "anemo": return "🌪️";
            case "geo": return "☄️";
            case "dendro": return "🌿";
            default: return "";
        }
    }

    private string FormatPercent(double value)
    {
        return value.ToString("0.#", System.Globalization.CultureInfo.InvariantCulture) + "%";
    }

    private void SendSplit(string message)
    {
        if (string.IsNullOrEmpty(message))
            return;

        if (message.Length <= MaxMessageLength)
        {
            CPH.SendMessage(message);
            return;
        }

        string remaining = message;

        while (remaining.Length > MaxMessageLength)
        {
            int cut = remaining.LastIndexOf(" | ", MaxMessageLength);

            if (cut <= 0)
                cut = MaxMessageLength;

            string part = remaining.Substring(0, cut).Trim();
            CPH.SendMessage(part);
            remaining = remaining.Substring(cut).Trim();

            if (remaining.StartsWith("|"))
                remaining = remaining.Substring(1).Trim();
        }

        if (remaining.Length > 0)
            CPH.SendMessage(remaining);
    }

    private void SendInfo(string message) { CPH.SendMessage("ℹ️ " + message); }
    private void SendWarning(string message) { CPH.SendMessage("⚠️ " + message); }
    private void SendError(string message) { CPH.SendMessage("❌ " + message); }
}
